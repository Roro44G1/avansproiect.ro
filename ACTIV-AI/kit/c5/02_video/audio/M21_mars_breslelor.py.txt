"""M21 – fundal muzical pentru „Breslele defilează” (64 s).

Temă originală, compusă și sintetizată în cod pentru animația M21 (artefact-model ACTIV AI),
în stil de marș: tobe (toba mică, toba mare, talgere), alămuri sintetizate (melodia, acordurile
pe contratimp, tuba) și fifă (fluier înalt). Nu citează și nu imită o melodie existentă;
nu are titlu de piesă reală.

Tempo 120 BPM, 4/4 (o măsură = 2 s), Si bemol major (trio în Mi bemol major).
Secțiunile urmăresc scenele video:
  0,0 s  rafală de tobă mică + semnal de alămuri     – titlul (9 noiembrie 1376)
  4,0 s  tema A (alămuri, apoi și fifa)              – muzicanții și primele 4 steaguri numite
 20,0 s  trio (fifă legato, alămuri moi)             – steagurile cu semnul întrebării
 28,0 s  acord de dominantă + motiv „întrebător”     – ultimele steaguri
 30,0 s  pauză de tobe                               – cartonul „Alte bresle atestate…”
 32,0 s  tema A, tutti (cu clopoței)                 – a doua paradă
 48,0 s  cadență
 50,0 s  patru acorduri care urcă (19 → 26 → 34 → 40) – contorul
 58,0 s  cadență plagală, încet                       – desființarea (1872 / 1875)
 60,0 s  acord final, stins                           – sursele

Totul este sintetizat din formule (oscilatoare cu bandă limitată, sinteză modală, zgomot filtrat) –
fără eșantioane. Folosește biblioteca comună /home/claude/expo/04_audio/_src/audiolib.py.

Rulare: python3 M21_mars_breslelor.py  -> M21_mars_breslelor.mp3 (în același folder)
"""
import os
import sys

import numpy as np

HERE = os.path.dirname(os.path.abspath(__file__))
LIB = "/home/claude/expo/04_audio/_src"
if LIB not in sys.path:
    sys.path.insert(0, LIB)

from audiolib import (add, additive, bp, env_adsr, env_ar, finalize, hp, hz, kick, lp, make_ir,
                      match_lufs, modal, ns, report, reverb, rng_of, saw, stereo, t_axis, vibrato)

BPM = 120
BEAT = 60.0 / BPM          # 0,5 s
BAR = 4 * BEAT             # 2,0 s
DUR = 64.0
OUT = os.path.join(HERE, "M21_mars_breslelor.mp3")


def b2t(bar, beat=0.0):
    return bar * BAR + beat * BEAT


# ------------------------------------------------------------------ instrumente
def brass(f, dur, rng, bright=1.0):
    """Alamă sintetizată: două dinți de fierăstrău ușor dezacordați, „scoop” de ton la atac,
    strălucire care se deschide repede (amestec între un filtru închis și unul deschis)."""
    n = ns(dur + 0.12)
    t = t_axis(n)
    scoop = 2 ** ((-0.35 * np.exp(-t / 0.03)) / 12)
    vib = 2 ** ((0.07 * np.clip((t - 0.3) / 0.3, 0, 1) * np.sin(2 * np.pi * 5.2 * t + rng.uniform(0, 6))) / 12)
    fr = f * scoop * vib
    x = 0.5 * saw(fr * 2 ** (0.05 / 12)) + 0.5 * saw(fr * 2 ** (-0.05 / 12), phase0=0.37)
    eb = (1 - np.exp(-t / 0.035)) * (0.55 + 0.45 * np.exp(-t / 0.25))
    y = lp(x, 650) * (1 - eb) + lp(x, 650 + 3600 * bright) * eb
    return y * env_adsr(n, 0.025, 0.12, 0.8, 0.09)


def fife(f, dur, rng):
    """Fifă: sinus cu puține armonice, suflu filtrat, „chiff” la atac, vibrato întârziat."""
    n = ns(dur + 0.08)
    t = t_axis(n)
    fr = vibrato(n, f, rate=5.8, depth_semitones=0.12, delay=0.18, rng=rng)
    x = additive(fr, [1.0, 0.22, 0.09, 0.03])
    breath = bp(rng.standard_normal(n), f * 0.8, min(f * 3.0, 15000)) * 0.06
    chiff = np.exp(-t / 0.03) * bp(rng.standard_normal(n), 1500, 6000) * 0.18
    return (x + breath + chiff) * env_adsr(n, 0.018, 0.06, 0.85, 0.06)


def tuba(f, dur, rng):
    n = ns(dur + 0.1)
    t = t_axis(n)
    x = 0.6 * lp(saw(np.full(n, f), phase0=rng.uniform()), 480) + 0.7 * np.sin(2 * np.pi * f * t)
    return x * env_adsr(n, 0.02, 0.1, 0.7, 0.08)


def snare(rng, vel=1.0, dur=0.3):
    n = ns(dur)
    body = modal([185, 330, 460], [1.0, 0.5, 0.25], [0.09, 0.06, 0.04], dur, rng=rng, attack=0.0005)
    nz = bp(rng.standard_normal(n), 1800, 8000) * env_ar(n, 0.002, 0.2)
    y = 0.6 * body + 0.9 * nz
    return vel * y / (np.max(np.abs(y)) + 1e-12)


def rim(rng):
    x = modal([1650, 2900, 4100], [1.0, 0.5, 0.2], [0.05, 0.03, 0.02], 0.08, rng=rng, attack=0.0003)
    return x / (np.max(np.abs(x)) + 1e-12)


def bass_drum(rng):
    x = kick(dur=0.7, f_hi=95, f_lo=46, pitch_tau=0.04, t60=0.55, click=0.04, rng=rng)
    n = len(x)
    x = x + 0.35 * lp(rng.standard_normal(n), 180) * env_ar(n, 0.002, 0.25)
    return x / (np.max(np.abs(x)) + 1e-12)


def cymbal(rng, dur=2.2):
    n = ns(dur)
    nz = hp(rng.standard_normal(n), 4200) * env_ar(n, 0.002, dur * 0.8)
    metal = modal([3150, 4370, 5230, 6810, 7940, 9120], [1, .8, .7, .6, .5, .4],
                  [1.2, 1.0, .9, .8, .7, .6], dur, rng=rng, beat_hz=3)
    y = 0.8 * nz / (np.max(np.abs(nz)) + 1e-12) + 0.25 * metal / (np.max(np.abs(metal)) + 1e-12)
    return y / (np.max(np.abs(y)) + 1e-12)


def glock(f, rng):
    x = modal([f, f * 2.76, f * 5.4, f * 8.93], [1.0, 0.38, 0.16, 0.07], [1.3, 0.6, 0.3, 0.18], 1.6,
              rng=rng, attack=0.001)
    return x / (np.max(np.abs(x)) + 1e-12)


# ------------------------------------------------------------------ materialul muzical
# fiecare măsură = 4 timpi: (notă sau None, durată în timpi)
THEME_A = [
    [("Bb4", 1), ("D5", .75), ("C5", .25), ("Bb4", 1), ("F5", 1)],
    [("G5", .75), ("F5", .25), ("Eb5", .5), ("D5", .5), ("C5", 2)],
    [("A4", 1), ("C5", .75), ("Bb4", .25), ("A4", 1), ("Eb5", 1)],
    [("D5", .75), ("C5", .25), ("Bb4", .5), ("A4", .5), ("Bb4", 1), (None, 1)],
    [("Bb4", 1), ("D5", .75), ("C5", .25), ("Bb4", 1), ("F5", 1)],
    [("Bb5", 1.5), ("A5", .5), ("G5", .5), ("F5", .5), ("Eb5", 1)],
    [("D5", .75), ("Eb5", .25), ("F5", .5), ("D5", .5), ("C5", .75), ("D5", .25), ("Eb5", 1)],
    [("D5", 1), ("C5", 1), ("Bb4", 1), (None, 1)],
]
CH_A = [("Bb", "Bb"), ("Eb", "F7"), ("F7", "F7"), ("F7", "Bb"), ("Bb", "Bb"), ("Eb", "Eb"), ("Bb", "F7"), ("Bb", "Bb")]
TRIO = [
    [("G5", 2), ("Bb5", 1), ("Eb6", 1)],
    [("D6", 1.5), ("C6", .5), ("Bb5", 2)],
    [("Ab5", 1), ("C6", 1), ("F6", 1.5), ("Eb6", .5)],
    [("D6", 1), ("Bb5", 1), ("Eb6", 2)],
]
CH_T = [("Eb", "Eb"), ("Bb7", "Bb7"), ("Fm7", "Fm7"), ("Bb7", "Eb")]
for ph in THEME_A + TRIO:
    assert abs(sum(d for _, d in ph) - 4) < 1e-9

CH = {"Bb": ["D4", "F4", "Bb4"], "Eb": ["Eb4", "G4", "Bb4"], "F7": ["Eb4", "F4", "A4"], "F": ["F4", "A4", "C5"],
      "Bb7": ["D4", "Ab4", "Bb4"], "Fm7": ["Eb4", "F4", "Ab4"]}
ROOT = {"Bb": "Bb1", "Eb": "Eb2", "F7": "F2", "F": "F2", "Bb7": "Bb1", "Fm7": "F2"}
FIFTH = {"Bb": "F2", "Eb": "Bb1", "F7": "C2", "F": "C2", "Bb7": "F2", "Fm7": "C2"}


def render(seed=1376):
    rng = rng_of(seed)
    lead, fif, pah, pad, tub, snr, bdr, cym, glk = (stereo(DUR) for _ in range(9))

    def phrase(buf, ph, bar, inst, gain, pan, transpose=0, legato=0.92):
        pos = 0.0
        for nm, d in ph:
            if nm is not None:
                x = inst(hz(nm, transpose), d * BEAT * legato, rng)
                add(buf, x, b2t(bar, pos) + rng.uniform(-0.003, 0.003), gain=gain * rng.uniform(0.9, 1.0), pan=pan)
            pos += d

    def chord(buf, name, t0, dur, gain, bright=0.6, pan=0.0, top=None):
        notes = CH[name] + ([top] if top else [])
        for k, nm in enumerate(notes):
            add(buf, brass(hz(nm), dur, rng, bright), t0, gain=gain / len(notes) ** 0.5,
                pan=pan + (k - 1) * 0.18)

    def oompah(bar, chords, lvl=1.0, pah_on=True):
        for half, ch in enumerate(chords):
            t0 = b2t(bar, 2 * half)
            same = half == 1 and chords[0] == chords[1]
            add(tub, tuba(hz(FIFTH[ch] if same else ROOT[ch]), BEAT * 0.8, rng), t0, gain=lvl)
            if pah_on:
                chord(pah, ch, t0 + BEAT, BEAT * 0.42, 0.8 * lvl, bright=0.45)

    def drums_march(bar, lvl=1.0, piatti=True):
        pattern = [(0, 1.0), (.75, .45), (1, .7), (1.5, .55), (2, .95), (2.75, .45), (3, .7), (3.25, .4), (3.5, .6)]
        for beat, g in pattern:
            t = b2t(bar, beat)
            if beat in (0, 2):  # „flam”: o lovitură de grație înainte
                add(snr, snare(rng, 0.35), t - 0.025, gain=lvl * 0.5, pan=-0.15)
            add(snr, snare(rng), t, gain=lvl * g * rng.uniform(0.9, 1.0), pan=-0.15)
        for beat in (0, 2):
            add(bdr, bass_drum(rng), b2t(bar, beat), gain=lvl * (1.0 if beat == 0 else 0.85))
            if piatti:
                add(cym, cymbal(rng, 0.6), b2t(bar, beat), gain=lvl * 0.35, pan=0.3)

    def roll(t0, t1, g0, g1, step=0.04):
        k = 0
        for t in np.arange(t0, t1, step):
            u = (t - t0) / max(1e-6, t1 - t0)
            add(snr, snare(rng, 1.0, 0.18), t, gain=lerp(g0, g1, u) * (0.85 if k % 2 else 1.0), pan=-0.15)
            k += 1

    def crash(t, g=1.0):
        add(cym, cymbal(rng, 2.4), t, gain=g, pan=0.25)

    # ---- intro: rafală de tobă și semnal de alămuri
    roll(0.0, 1.95, 0.15, 0.95)
    add(bdr, bass_drum(rng), 0.0, gain=0.6)
    fanfare = [("F4", .5), ("Bb4", .5), ("D5", .5), ("F5", .5), ("Bb5", 2)]
    phrase(lead, fanfare, 1, lambda f, d, r: brass(f, d, r, 1.0), 1.0, 0.0, legato=0.85)
    chord(pad, "Bb", b2t(1, 2), BEAT * 2, 0.9, bright=0.7)
    add(tub, tuba(hz("Bb1"), BEAT * 1.8, rng), b2t(1, 2), gain=1.0)
    add(bdr, bass_drum(rng), b2t(1, 2), gain=1.0)
    crash(b2t(1, 2), 0.9)
    for beat, g in ((3.5, 0.7), (3.75, 0.9)):
        add(snr, snare(rng), b2t(1, beat), gain=g, pan=-0.15)

    # ---- tema A (măsurile 2–9): alămuri; fifa se alătură în a doua jumătate
    for k in range(8):
        bar = 2 + k
        phrase(lead, THEME_A[k], bar, lambda f, d, r: brass(f, d, r, 1.0), 1.0, 0.05)
        if k >= 4:
            phrase(fif, THEME_A[k], bar, fife, 0.8, -0.25, transpose=12)
        oompah(bar, CH_A[k])
        drums_march(bar, 0.9 if k < 4 else 1.0)
    crash(b2t(2), 0.8)
    crash(b2t(6), 0.6)

    # ---- trio (măsurile 10–13): fifă legato, acorduri lungi și moi, tobe ușoare
    for k in range(4):
        bar = 10 + k
        phrase(fif, TRIO[k], bar, fife, 1.0, -0.1, legato=0.98)
        for half, ch in enumerate(CH_T[k]):
            if half == 0 or CH_T[k][1] != CH_T[k][0]:
                L = BAR if CH_T[k][1] == CH_T[k][0] else BAR / 2
                chord(pad, ch, b2t(bar, 2 * half), L * 0.95, 0.75, bright=0.25)
        oompah(bar, CH_T[k], 0.75, pah_on=False)
        for beat in (1, 3):
            add(snr, rim(rng), b2t(bar, beat), gain=0.55, pan=-0.2)
        add(bdr, bass_drum(rng), b2t(bar), gain=0.7)
    # măsura 14: dominanta (Fa 7) + motiv „întrebător” la fifă, rafală care crește
    chord(pad, "F7", b2t(14), BAR * 0.95, 0.85, bright=0.5)
    add(tub, tuba(hz("F2"), BAR * 0.9, rng), b2t(14), gain=0.9)
    phrase(fif, [("C6", .5), ("F6", .5), ("A6", 1), (None, 2)], 14, fife, 0.95, -0.1)
    roll(b2t(14, 2), b2t(15) - 0.03, 0.2, 1.0)

    # ---- măsura 15: pauză de tobe (cartonul „Alte bresle…”)
    crash(b2t(15), 1.0)
    brk = [(0, 1.0), (.25, .4), (.5, .55), (.75, .8), (1, .5), (1.25, .4), (1.5, .9), (2, 1.0), (2.25, .45),
           (2.5, .8), (2.75, .45), (3, .95), (3.25, .5)]
    for beat, g in brk:
        add(snr, snare(rng), b2t(15, beat), gain=g, pan=-0.15)
    for beat in (0, 1.5, 2, 3):
        add(bdr, bass_drum(rng), b2t(15, beat), gain=0.9)
    add(lead, brass(hz("F4"), BEAT * 0.4, rng), b2t(15, 3.5), gain=0.8)
    add(lead, brass(hz("F4"), BEAT * 0.2, rng), b2t(15, 3.75), gain=0.8)

    # ---- tema A tutti (măsurile 16–23): alămuri + fifă + clopoței
    for k in range(8):
        bar = 16 + k
        phrase(lead, THEME_A[k], bar, lambda f, d, r: brass(f, d, r, 1.0), 1.0, 0.05)
        phrase(fif, THEME_A[k], bar, fife, 0.85, -0.25, transpose=12)
        phrase(glk, THEME_A[k], bar, lambda f, d, r: glock(f, r), 0.6, 0.35, transpose=12)
        oompah(bar, CH_A[k], 1.05)
        drums_march(bar, 1.0)
    crash(b2t(16), 1.0)
    crash(b2t(20), 0.7)

    # ---- măsura 24: cadență „ta – ta – taaa”
    for beat, L, g in ((0, .35, .9), (1, .35, .9), (2, 1.6, 1.0)):
        t0 = b2t(24, beat)
        chord(pad, "Bb", t0, L, 1.1 * g, bright=0.8, top="Bb5" if beat == 2 else "F5")
        add(tub, tuba(hz("Bb1"), L, rng), t0, gain=1.0)
        add(bdr, bass_drum(rng), t0, gain=g)
        add(snr, snare(rng), t0, gain=g, pan=-0.15)
    crash(b2t(24, 2), 0.9)

    # ---- măsurile 25–28: contorul 19 → 26 → 34 → 40 (acorduri care urcă)
    steps = [("Bb", "D5", "Bb1"), ("Eb", "Eb5", "Eb2"), ("F", "F5", "F2"), ("Bb", "Bb5", "Bb1")]
    for k, (ch, top, root) in enumerate(steps):
        t0 = b2t(25 + k)
        g = 0.85 + 0.08 * k
        chord(pad, ch, t0, BEAT * 3, 1.1 * g, bright=0.75, top=top)
        add(lead, brass(hz(top), BEAT * 3, rng, 1.0), t0, gain=0.8 * g)
        add(tub, tuba(hz(root), BEAT * 2.5, rng), t0, gain=1.0)
        add(bdr, bass_drum(rng), t0, gain=g)
        crash(t0, 0.5 + 0.15 * k)
        if k < 3:
            roll(t0 + BEAT * 3, t0 + BAR - 0.03, 0.2, 0.75)
    for j in range(20):  # tril de fifă pe ultimul acord
        add(fif, fife(hz("Bb6" if j % 2 == 0 else "C7"), 0.06, rng), b2t(28) + j * 0.065, gain=0.55, pan=-0.2)

    # ---- măsura 29: cadență plagală, încet (desființarea)
    chord(pad, "Eb", b2t(29), BEAT * 2, 0.55, bright=0.2)
    add(tub, tuba(hz("Eb2"), BEAT * 1.8, rng), b2t(29), gain=0.6)
    chord(pad, "Bb", b2t(29, 2), BEAT * 2.2, 0.55, bright=0.2)
    add(tub, tuba(hz("Bb1"), BEAT * 2, rng), b2t(29, 2), gain=0.6)
    add(bdr, bass_drum(rng), b2t(29), gain=0.45)
    add(fif, fife(hz("F5"), BAR * 0.95, rng), b2t(29), gain=0.45, pan=-0.1)

    # ---- măsurile 30–31: acord final, stins
    chord(pad, "Bb", b2t(30), BAR * 1.8, 0.6, bright=0.25, top="D5")
    add(tub, tuba(hz("Bb1"), BAR * 1.6, rng), b2t(30), gain=0.55)
    for j, nm in enumerate(("Bb5", "D6", "F6", "Bb6")):
        add(glk, glock(hz(nm), rng), b2t(30, j * 0.5), gain=0.5, pan=-0.3 + 0.2 * j)
    add(snr, rim(rng), b2t(31), gain=0.3, pan=-0.2)

    ir = make_ir(t60=1.3, predelay=0.014, seed=76)
    # filtrare blândă a straturilor strălucitoare (tranzitorii foarte înalte produc vârfuri la recodarea AAC din MP4)
    cym, glk, snr = lp(cym, 9000), lp(glk, 8000), lp(snr, 9000)
    stems = [(lead, -17.0, 0.16), (fif, -20.5, 0.22), (pah, -24.0, 0.15), (pad, -23.0, 0.2), (tub, -20.0, 0.05),
             (snr, -21.0, 0.14), (bdr, -22.5, 0.08), (cym, -29.0, 0.2), (glk, -26.0, 0.25)]
    mix = None
    for x, target, wet in stems:
        y = reverb(match_lufs(hp(x, 30), target), ir, wet=wet)
        mix = y if mix is None else mix + y
    return lp(mix, 13000)


def lerp(a, b, u):
    return a + (b - a) * u


if __name__ == "__main__":
    mix = render()
    out = finalize(mix, OUT, "M21 – temă originală în stil de marș pentru Breslele defilează",
                   "Temă originală, sintetizată în cod (tobe, alămuri sintetizate, fifă) – ACTIV AI",
                   target_lufs=-14.0, fin=0.02, fout=2.0, ceiling_db=-2.6, max_tp=-2.0)  # rezervă pentru recodarea AAC din MP4
    report(out, "M21 final")
