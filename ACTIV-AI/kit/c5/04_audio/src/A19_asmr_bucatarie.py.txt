def ev(name, t, lo, hi, dur, ref=None):
    """Eveniment pentru verificare; ref = fereastra de fond (s) cu care se compară, dacă stratul
    e prezent până la final (altfel: mediana benzii pe tot fișierul)."""

# ------------------------------------------------------------------ utilitare
def nrm(x):
    return x / (np.max(np.abs(x)) + 1e-12)


def smooth_env(n, att, rel):
    e = np.ones(n)
    a, r = max(1, min(n // 2, ns(att))), max(1, min(n // 2, ns(rel)))
    e[:a] = np.sin(np.linspace(0, np.pi / 2, a)) ** 2
    e[n - r:] *= np.cos(np.linspace(0, np.pi / 2, r)) ** 2
    return e


def grains(rng, length, rate, lo, hi, gmin=0.0003, gmax=0.0015, env=None, amp=(0.2, 1.0)):
    """Nor de micro-pocnete (granule de zgomot) – foșnet, crănțănit, boabe, frunze."""
    m = ns(length + gmax + 0.01)
    x = np.zeros(m)
    k = rng.poisson(rate * length)
    for tt in np.sort(rng.uniform(0, length, k)):
        g = max(4, ns(rng.uniform(gmin, gmax)))
        i0 = ns(tt)
        if i0 + g >= m:
            continue
        a = rng.uniform(*amp) * (env(tt / length) if env is not None else 1.0)
        x[i0:i0 + g] += rng.standard_normal(g) * np.hanning(g) * a
    return bp(x, lo, hi, order=2)


def ping(f0, d, rise, rng=None):
    """Ton scurt al unei bule de aer în apă: frecvența urcă puțin, se stinge repede."""
    m = ns(d)
    tt = t_axis(m)
    f = f0 * (1 + rise * tt)
    ph0 = rng.uniform(0, 6.28) if rng is not None else 0.0
    return np.sin(2 * np.pi * np.cumsum(f) / SR + ph0) * np.exp(-tt / (d / 3)) * (1 - np.exp(-tt / 0.0012))


def stick_slip(rng, dur, r0, r1, res, jitter=0.15, broad=(600, 6000)):
    """Scârțâit (frecare cu „lipire și alunecare”): un tren de impulsuri cu ritm variabil,
    trecut prin rezonanțe înguste (balama de fontă, frunză crocantă)."""
    n = ns(dur)
    imp = np.zeros(n)
    wob = smooth_random(n, rng, 6.0, 0.85, 1.15)
    t = 0.0
    while t < dur:
        i = ns(t)
        if i >= n:
            break
        imp[i] += rng.uniform(0.5, 1.0)
        u = t / dur
        r = r0 * (r1 / r0) ** u * wob[i]
        t += (1.0 / r) * (1 + rng.uniform(-jitter, jitter))
    y = 0.35 * bp(imp, *broad, order=2)
    for f, g in res:
        y += g * bp(imp, f * 0.96, f * 1.04, order=2)
    return y * smooth_env(n, 0.05, 0.12)


def tv_reson(x, f, q):
    """Rezonator cu frecvență variabilă în timp (filtru „state variable”, ieșire trece-bandă)."""
    out = np.empty_like(x)
    low = band = 0.0
    F = 2 * np.sin(np.pi * np.asarray(f) / SR)
    qi = 1.0 / q
    for i in range(len(x)):
        low += F[i] * band
        high = x[i] - low - qi * band
        band += F[i] * high
        out[i] = band
    return out

def match_layer(rng):
    buf = stereo(DUR)
    sc = grains(rng, 0.2, 2200, 1500, 8500, 0.0002, 0.0012, env=lambda u: 0.4 + 0.6 * u)
    m = len(sc)
    sc += bp(rng.standard_normal(m), 2000, 6000) * smooth_env(m, 0.03, 0.03) * 0.06
    flare_n = ns(1.6)
    tt = t_axis(flare_n)
    nz = rng.standard_normal(flare_n)
    flare = bp(nz, 600, 7000) * np.minimum(1, tt / 0.02) * np.exp(-6.9 * tt / 0.9) * 0.9
    flare += lp(nz, 400) * np.minimum(1, tt / 0.01) * np.exp(-6.9 * tt / 0.35) * 0.8
    flare += hp(rng.standard_normal(flare_n), 1500) * np.exp(-6.9 * tt / 1.6) * 0.12  # flacăra mică
    x = np.zeros(m + flare_n)
    x[:m] += sc
    x[ns(0.18):ns(0.18) + flare_n] += flare
    add(buf, nrm(x), T_MATCH, gain=1.0, pan=PAN["soba"] - 0.15)
    ev("chibritul", T_MATCH, 1500, 8000, 0.5)
    return buf


# ------------------------------------------------------------------ focul
def crackle(rng, big=False):
    m = ns(0.12 if big else 0.05)
    x = np.zeros(m)
    k = int(rng.integers(3, 7) if big else rng.integers(1, 4))
    pos = 0.0
    for _ in range(k):
        g = max(4, ns(rng.uniform(0.0002, 0.0022 if big else 0.0011)))
        i0 = ns(pos)
        if i0 + g >= m:
            break
        x[i0:i0 + g] += rng.standard_normal(g) * np.hanning(g) * rng.uniform(0.4, 1.0)
        pos += rng.exponential(0.006 if big else 0.004)
    x = bp(x, 700, 9500)
    if big:
        x += lp(rng.standard_normal(m), 260) * env_exp(m, 0.035) * 0.35
    return x


def fire_layer(rng):
    n = ns(DUR)
    t = t_axis(n)
    level = np.clip((t - T_IGNITE) / 3.0, 0, 1) ** 1.5
    open_w = np.clip((T_DOOR_CLOSE + 0.05 - t) / 0.12, 0, 1)
    buf = stereo(DUR)
    # vuietul moale al flăcării
    for ch in range(2):
        nz = colored_noise(n, rng_of(3100 + ch), 2.0)
        rum = hp(lp(nz, 320), 55, order=4) * (0.6 + 0.4 * smooth_random(n, rng_of(3150 + ch), 2.0))
        hiss = hp(rng_of(3170 + ch).standard_normal(n), 2500) * smooth_random(n, rng_of(3180 + ch), 1.5) * 0.05
        buf[ch] = (0.5 * rum + hiss) * level
    # aprinderea: un „vuuf” scurt și moale, care se deschide puțin spre înalte
    m = ns(1.8)
    tt = t_axis(m)
    nz = rng.standard_normal(m)
    w = np.clip(tt / 1.2, 0, 1)
    whoosh = (lp(nz, 300) * (1 - w) + lp(hp(nz, 150), 1400) * w * 0.7) * np.minimum(1, tt / 0.25) * np.exp(-tt / 0.6)
    add(buf, whoosh * 0.35, T_IGNITE, pan=PAN["soba"])
    # trosnete: întâi surcelele (dese, mici), apoi lemnele (mai rare, cu pocnete mari)
    cr = stereo(DUR)
    tc = T_IGNITE + 0.3
    while tc < DUR - 0.2:
        if tc < T_DOOR_CLOSE:
            rate = 1.0 + 9.0 * np.clip((tc - T_IGNITE) / 3.0, 0, 1)
        else:
            rate = 3.2 + 1.5 * np.sin(2 * np.pi * tc / 23.0)
        big = tc > 9.0 and rng.uniform() < 0.07
        g_open = 0.8 if tc < T_DOOR_CLOSE else 1.0  # surcelele: dese, dar mai mici
        add(cr, crackle(rng, big), tc, gain=g_open * rng.uniform(0.25, 1.0) * (1.6 if big else 1.0),
            pan=PAN["soba"] + rng.uniform(-0.12, 0.12))
        tc += rng.exponential(1.0 / rate)
    buf += cr
    # după închiderea ușiței: totul se aude prin fontă – înfundat și mai încet.
    # Cele două părți sunt întoarse separat, ca nivelul fiecăreia să fie reglat în mix.
    closed = lp(buf, 3200, order=2) * (1 - open_w)
    # focul e prezent până la final: se compară cu fondul de dinainte de aprindere (0,5–2,9 s)
    ev("focul pornește (surcele)", 7.0, 1500, 8000, 4.0, ref=(0.5, 2.9))
    ev("lemnele trosnesc (ușița închisă)", 64.0, 700, 3000, 3.5, ref=(0.5, 2.9))
    return buf * open_w, closed

def pour_layer(rng):
    buf = stereo(DUR)
    n = ns(POUR_DUR)
    t = t_axis(n)
    u = t / POUR_DUR
    env = smooth_env(n, 0.25, 0.35)
    f_res = 380 * (1 + 1.45 * u ** 0.8)           # rezonanța „aerului” din ceaun urcă pe măsură ce se umple
    turb = rng.standard_normal(n) * (0.6 + 0.4 * np.abs(lp(rng.standard_normal(n), 40)) * 3)
    body = tv_reson(turb, f_res, 7.0)
    body = body / (np.std(body) + 1e-12) * 0.35
    splash = bp(rng.standard_normal(n), 500, 6000) * (0.5 + 0.5 * smooth_random(n, rng, 8, 0, 1)) * 0.12
    x = (body + splash) * env
    # primele clipe: apa lovește fundul gol, de fontă
    k = ns(0.7)
    metal = bp(rng.standard_normal(k), 1800, 4500) * np.linspace(1, 0, k) ** 1.5 * 0.35
    metal += modal([640, 1580, 2900], [0.4, 0.25, 0.12], [0.4, 0.25, 0.15], 0.7, rng=rng) * 0.6
    x[:k] += metal * smooth_env(k, 0.03, 0.2)
    # bule dese în jet, legate de rezonanța care urcă
    tb = 0.1
    while tb < POUR_DUR - 0.35:
        i = ns(tb)
        p = ping(f_res[i] * rng.uniform(0.9, 2.6), rng.uniform(0.008, 0.03), rng.uniform(4, 14), rng)
        j = min(len(p), n - i)
        x[i:i + j] += p[:j] * rng.uniform(0.05, 0.25) * env[i]
        tb += rng.exponential(1 / 90.0)
    add(buf, x, T_POUR, pan=PAN["ceaun"])
    add(buf, x * 0.25, T_POUR + 0.011, pan=PAN["ceaun"] - 0.5)  # puțină lățime
    # ultimele picături
    for dt in (0.3, 0.65, 1.15):
        d = ping(rng.uniform(900, 1400), 0.05, 9.0, rng) * 0.35
        add(buf, d, T_POUR + POUR_DUR + dt, pan=PAN["ceaun"])
    ev("apa turnată în ceaun", T_POUR, 300, 2500, POUR_DUR)
    return buf

def simmer_layer(rng):
    buf = stereo(DUR)
    t = T_SIMMER
    span = DUR - T_SIMMER
    while t < DUR - 0.1:
        u = (t - T_SIMMER) / span
        rate = 0.6 + 13.0 * u ** 1.4
        f0 = np.exp(rng.uniform(np.log(220), np.log(750)))
        d = rng.uniform(0.03, 0.09)
        x = ping(f0, d, rng.uniform(2, 6), rng)
        m = len(x)
        x += hp(rng.standard_normal(m), 2000) * env_exp(m, 0.004) * 0.15  # mica „pocnitură” la suprafață
        add(buf, x, t, gain=rng.uniform(0.3, 1.0) * (0.5 + 0.5 * min(1.0, 2 * u)),
            pan=PAN["ceaun"] + rng.uniform(-0.1, 0.1))
        t += rng.exponential(1.0 / rate)
    # „cântecul” discret al apei care se încălzește
    n = ns(DUR)
    tt = t_axis(n)
    ramp = np.clip((tt - T_SIMMER + 6) / (span + 6), 0, 1)
    for ch in range(2):
        s = bp(rng_of(3500 + ch).standard_normal(n), 1800, 5000) * smooth_random(n, rng_of(3550 + ch), 1.2, 0.3, 1)
        buf[ch] += s * ramp * 0.03
    # se compară cu fondul de dinainte de fierbere, fără cuțit (28,0–29,4 s: foc, ceas, cameră)
    ev("ceaunul fierbe încet", 93.0, 200, 900, 5.0, ref=(28.0, 29.4))
    return buf

def board_hit(rng):
    x = modal([185, 440, 960, 1850], [1, 0.6, 0.35, 0.18], [0.07, 0.05, 0.03, 0.02], 0.15, rng=rng,

    x += modal([4500], [0.06], [0.03], 0.15, rng=rng)  # o urmă metalică de lamă

def chop(rng, crunch_len):
    cr = grains(rng, crunch_len, 2600, 1500, 9500, 0.0002, 0.0012, env=lambda u: 1 - 0.6 * u)
    bh = board_hit(rng)
    k = ns(crunch_len * 0.85)
    x = np.zeros(k + len(bh) + 10)
    x[:len(cr)] += 0.6 * nrm(cr)
    x[k:k + len(bh)] += bh
    return x

def spoon_layer(rng):
    buf = stereo(DUR)
    n = ns(SPOON_DUR)
    tt = t_axis(n)
    stir = 0.5 + 0.5 * np.sin(2 * np.pi * 1.15 * tt - np.pi / 2)
    env = smooth_env(n, 0.3, 0.4)
    squish = lp(rng.standard_normal(n), 900) * (0.3 + 0.7 * stir) * 0.5
    squish += bp(rng.standard_normal(n), 300, 2000) * stir ** 2 * 0.25
    sticky = grains(rng, SPOON_DUR, 60, 300, 2500, 0.001, 0.004)[:n] * (0.3 + stir)  # „lipicios”
    rice = grains(rng, SPOON_DUR, 900, 2500, 9000, 0.0002, 0.0008)[:n] * stir * 0.6  # boabele de orez
    scrape = np.zeros(n)
    for c in np.arange(0.62, SPOON_DUR - 0.3, 1 / 1.15):
        i0 = ns(c)
        m = min(ns(0.25), n - i0)
        scrape[i0:i0 + m] += bp(rng.standard_normal(m), 900, 3500) * np.hanning(m) * 0.12
    x = (squish + 2.0 * sticky + rice + scrape) * env
    add(buf, nrm(x), T_SPOON, gain=1.0, pan=PAN["castron"])
    for t0, g in ((T_SPOON + SPOON_DUR + 0.4, 1.0), (T_SPOON + SPOON_DUR + 0.75, 0.8)):
        add(buf, spoon_tap(rng) * 0.9, t0, gain=g, pan=PAN["castron"])
    ev("lingura de lemn (amestecat)", T_SPOON, 300, 3000, SPOON_DUR)
    ev("lingura bate de două ori în castron", T_SPOON + SPOON_DUR + 0.4, 1000, 3000, 0.5)
    return buf


# ------------------------------------------------------------------ frunzele de varză
def leaf(rng):
    cdur = rng.uniform(0.25, 0.42)
    creak = stick_slip(rng, cdur, rng.uniform(160, 220), rng.uniform(420, 600),
                       [(1400, 0.8), (2600, 0.6), (4100, 0.35)], jitter=0.35, broad=(1200, 8000))
    snap_n = ns(0.03)
    snap = bp(rng.standard_normal(snap_n), 1000, 9000) * env_exp(snap_n, 0.006)
    snap += lp(rng.standard_normal(snap_n), 400) * env_exp(snap_n, 0.015) * 0.4
    rust = grains(rng, 0.45, 520, 2000, 10000, 0.0002, 0.0012, env=lambda u: (1 - u) ** 1.5)
    rust_soft = lp(rng.standard_normal(len(rust)), 5000) * np.exp(-np.arange(len(rust)) / ns(0.12)) * 0.08
    flop_n = ns(0.12)
    flop = lp(rng.standard_normal(flop_n), 500) * env_exp(flop_n, 0.04)
    gr = grains(rng, 0.08, 300, 2000, 8000)
    flop[:len(gr)] += gr[:flop_n] * 0.4
    total = ns(cdur) + len(rust) + ns(0.3) + flop_n
    x = np.zeros(total)
    k = 0
    x[:len(creak)] += nrm(creak) * 0.7
    k = len(creak)
    x[k:k + snap_n] += nrm(snap)
    x[k:k + len(rust)] += nrm(rust + rust_soft) * 0.6
    k2 = k + ns(0.55)
    x[k2:k2 + flop_n] += nrm(flop) * 0.45
    return x


def leaves_layer(rng):
    buf = stereo(DUR)
    for t0 in LEAF_TIMES:
        add(buf, leaf(rng), t0, gain=rng.uniform(0.85, 1.0), pan=PAN["varza"] + rng.uniform(-0.12, 0.12))
        ev("frunză de varză desfăcută", t0, 1500, 9000, 1.0)
    # între frunze: mâinile care întorc frunzele, foarte încet
    for t0 in (52.6, 55.0, 57.5, 60.0, 62.2):
        add(buf, grains(rng, 0.5, 150, 2000, 8000) * 0.12, t0, pan=0.05)
    return buf


# ------------------------------------------------------------------ oala de ceramică pe plită
def pot_layer(rng):
    buf = stereo(DUR)
    L = 0.35
    m = ns(L)
    slide = bp(rng.standard_normal(m), 900, 4500) * smooth_env(m, 0.08, 0.05) * 0.18
    slide += grains(rng, L, 900, 1500, 7000, 0.0002, 0.001)[:m] * 0.4
    add(buf, slide, T_POT, pan=PAN["oala"])
    clunk = modal([360, 905, 1690, 2620, 3850], [1, 0.7, 0.45, 0.3, 0.15], [0.22, 0.16, 0.12, 0.09, 0.06],
                  0.6, rng=rng, attack=0.0005)
    clunk += modal([150, 510], [0.6, 0.3], [0.12, 0.08], 0.6, rng=rng)  # plita de fontă
    k = len(clunk)
    clunk += lp(rng.standard_normal(k), 180) * env_exp(k, 0.05) * 0.8
    clunk += hp(rng.standard_normal(k), 2500) * env_exp(k, 0.002) * 0.3
    add(buf, nrm(clunk), T_POT + L, gain=1.0, pan=PAN["oala"])
    ev("oala de ceramică pe plită", T_POT + L, 300, 2000, 0.4)
    # capacul de ceramică: câteva clinchete care se liniștesc
    t_lid = T_POT + 1.3
    for k, (off, g) in enumerate([(0, 1.0), (0.085, 0.6), (0.15, 0.4), (0.2, 0.25), (0.235, 0.15)]):
        b = rng.uniform(0.98, 1.02)
        c = modal([2150 * b, 3350 * b, 5250 * b], [1, 0.6, 0.3], [0.18, 0.12, 0.07], 0.35, rng=rng,
                  attack=0.0003)
        add(buf, nrm(c) * g * 0.7, t_lid + off, pan=PAN["oala"])
    ev("capacul oalei", t_lid, 2000, 6000, 0.35)
    return buf


# ------------------------------------------------------------------ clopotul de duminică, departe
def church_bell(rng, hum=130.8):
    ratios = [1.0, 2.0, 2.4, 3.0, 4.0, 5.0, 5.33, 8.0]   # hum, prima, terța mică, cvinta, nominala, ...
    amps = [0.5, 0.7, 0.6, 0.25, 1.0, 0.35, 0.25, 0.15]
    t60 = [9.0, 6.0, 5.0, 3.5, 4.0, 2.5, 2.0, 1.2]
    x = modal([hum * r * rng.uniform(0.999, 1.001) for r in ratios], amps, t60, 7.0, rng=rng,
              attack=0.002, beat_hz=0.8)
    n = len(x)
    x += hp(rng.standard_normal(n), 1500) * env_exp(n, 0.01) * 0.4
    return nrm(x)


def bell_layer(rng):
    buf = stereo(DUR)
    for k, t0 in enumerate(BELL_TIMES):
        x = hp(lp(church_bell(rng), 2400, order=2), 110)
        add(buf, x, t0 + rng.uniform(-0.03, 0.03), gain=rng.uniform(0.85, 1.0), pan=PAN["clopot"])
    ev("clopotul, departe", BELL_TIMES[0], 490, 560, 1.5)
    return buf


# ------------------------------------------------------------------ cana pe masă
def mug_layer(rng):
    buf = stereo(DUR)
    touch = modal([1720, 3950], [0.5, 0.3], [0.04, 0.03], 0.1, rng=rng)
    add(buf, nrm(touch) * 0.2, T_MUG, pan=PAN["cana"])
    x = modal([135, 320, 700], [1, 0.5, 0.25], [0.09, 0.06, 0.04], 0.5, rng=rng, attack=0.0005)  # masa de lemn
    x += modal([1720, 3950, 6150], [0.5, 0.3, 0.15], [0.09, 0.05, 0.03], 0.5, rng=rng, attack=0.0003)
    n = len(x)
    tt = t_axis(n)
    slosh = lp(rng.standard_normal(n), 1400) * (0.5 + 0.5 * np.sin(2 * np.pi * 7 * tt)) * np.exp(-tt / 0.06)
    x += np.roll(slosh, ns(0.03)) * 0.25
    add(buf, nrm(x), T_MUG + 0.05, gain=1.0, pan=PAN["cana"])
    ev("cana pe masă", T_MUG + 0.05, 1500, 4500, 0.25)
    return buf

def render(seed=1907):
    rng = rng_of(seed)
    ir_room = make_ir(t60=0.45, length=0.8, predelay=0.004, seed=51, hf_ratio=0.6)
    ir_far = make_ir(t60=2.6, predelay=0.05, seed=52, hf_ratio=0.35)
    fire_open, fire_closed = fire_layer(rng)
    layers = {  # nume: (semnal, LUFS țintă al evenimentelor, IR, wet, vârf maxim peste țintă în dB, eliberare limitator ms)
        "camera": (room_layer(), -46.0, ir_room, 0.0, 12.0, 40.0),
        "ceas": (clock_layer(rng), -36.0, ir_room, 0.25, 16.0, 12.0),
        "usita": (door_layer(rng), -29.0, ir_room, 0.2, 16.0, 20.0),

        "foc_aprindere": (fire_open, -31.0, ir_room, 0.12, 16.0, 12.0),
        "foc": (fire_closed, -33.0, ir_room, 0.12, 18.0, 12.0),
        "apa": (pour_layer(rng), -27.5, ir_room, 0.15, 14.0, 30.0),
        "fierbere": (simmer_layer(rng), -31.5, ir_room, 0.12, 16.0, 20.0),
        "cutit": (knife_layer(rng), -27.0, ir_room, 0.12, 14.0, 12.0),
        "lingura": (spoon_layer(rng), -28.0, ir_room, 0.12, 16.0, 20.0),
        "varza": (leaves_layer(rng), -27.0, ir_room, 0.12, 12.0, 12.0),

        "cana": (mug_layer(rng), -28.0, ir_room, 0.15, 12.0, 12.0),
    }
    mix = stereo(DUR)
    for name, (x, target, ir, wet, crest, rel) in layers.items():
        # vârfurile foarte scurte (pocnete, lovituri) sunt îmblânzite strat cu strat, cu un limitator
        # rapid, ca limitatorul final să nu „turtească” tot peisajul
        x = match_lufs(hp(x, 40, order=4), target)  # fără infrasunete (nu se aud, dar ar ocupa vârful)

        report(x, name)
        mix += reverb(x, ir, wet=wet, dry=1.0)
    mix = peaking(mix, 3500, -1.0, 0.8)
    return mix
