def mk(ox, oy, k):
    def P(x, y):
        return (ox + x * k, oy + y * k)
    return P

def wall_path(P, pts, closed=True, tension=0.9):
    return smooth_path([P(*p) for p in pts], closed=closed, tension=tension)

def wall(P, pts, color, fill_op=0.10, w=10, closed=True, rough=True):
    d = wall_path(P, pts, closed)
    out = ""
    if fill_op:
        out += f'<path d="{d}" fill="{color}" fill-opacity="{fill_op}" stroke="none"/>\n'
    out += f'<path d="{d}" fill="none" stroke="{color}" stroke-width="{w}" stroke-linejoin="round"{" filter=\"url(#rough)\"" if rough else ""}/>\n'
    out += f'<path d="{d}" fill="none" stroke="{PARCH}" stroke-width="{w*0.28:.1f}" stroke-dasharray="7 9" opacity="0.9"/>\n'
    return out

def ghost_wall(P, pts, color=CEATA, w=5, closed=True):
    d = wall_path(P, pts, closed)
    return (f'<path d="{d}" fill="none" stroke="{color}" stroke-width="{w+6}" stroke-linejoin="round" opacity="0.18" filter="url(#blur3)"/>\n'
            f'<path d="{d}" fill="none" stroke="{color}" stroke-width="{w}" stroke-linejoin="round" stroke-dasharray="4 14" stroke-linecap="round" opacity="0.85"/>\n')

def river(P, color="#8FBFD1", w=30, dash=PARCH):
    d = smooth_path([P(*p) for p in RIVER], closed=False)
    return (f'<path d="{d}" fill="none" stroke="{color}" stroke-width="{w}" stroke-linecap="round"/>\n'
            f'<path d="{d}" fill="none" stroke="{dash}" stroke-width="2" stroke-dasharray="14 18" opacity="0.8"/>\n')

def square(P, k, box, label, size=20, lcol="#3B3326", fill="#FBF7EE", stroke="#8C7B5E", label_above=False):
    x, y, w, h = box
    X, Y = P(x, y)
    out = f'<rect x="{X:.1f}" y="{Y:.1f}" width="{w*k:.1f}" height="{h*k:.1f}" rx="4" fill="{fill}" stroke="{stroke}" stroke-width="1.5"/>\n'
    if label:
        if label_above:
            out += text(X + w * k / 2, Y - 10, label, size, lcol, "sans", 700, anchor="middle")
        else:
            out += text(X + w * k / 2, Y + h * k / 2 + size * 0.35, label, size, lcol, "sans", 700, anchor="middle")
    return out

def tower_icon(X, Y, body=OCRU, roof=TIGLA, stroke="#3B3326", sc=1.0, op=1.0):
    s = 11 * sc
    return (f'<g filter="url(#shs)" opacity="{op}"><rect x="{X-s:.1f}" y="{Y-s:.1f}" width="{2*s:.1f}" height="{2*s:.1f}" rx="3" fill="{body}" stroke="{stroke}" stroke-width="2"/>'
            f'<polygon points="{X-s*1.2:.1f},{Y-s:.1f} {X:.1f},{Y-s*2.2:.1f} {X+s*1.2:.1f},{Y-s:.1f}" fill="{roof}" stroke="{stroke}" stroke-width="1.5"/></g>\n')

def bastion_icon(X, Y, fill="#7A8A95", stroke="#2B3A44", op=1.0):
    pts = [(X, Y - 18), (X + 18, Y - 6), (X + 12, Y + 14), (X - 12, Y + 14), (X - 18, Y - 6)]
    return poly(pts, fill=fill, stroke=stroke, stroke_width=1.5, filter="url(#shs)", opacity=op)

def church(P, k, cx, cy, body="#CDBFA3", roof=TIGLA, stroke="#6B5A3E", sc=1.0):
    X, Y = P(cx, cy)
    k = k * sc
    w, h = 42 * k, 52 * k
    out = f'<rect x="{X-w/2:.1f}" y="{Y-h/2:.1f}" width="{w:.1f}" height="{h:.1f}" fill="{body}" stroke="{stroke}" stroke-width="1.5"/>\n'
    out += f'<polygon points="{X-w/2-6:.1f},{Y-h/2:.1f} {X:.1f},{Y-h/2-26*k:.1f} {X+w/2+6:.1f},{Y-h/2:.1f}" fill="{roof}" stroke="{stroke}" stroke-width="1.2"/>\n'
    tw = 12 * k
    out += f'<rect x="{X-tw/2:.1f}" y="{Y-h/2-40*k:.1f}" width="{tw:.1f}" height="{40*k:.1f}" fill="{body}" stroke="{stroke}" stroke-width="1.2"/>'
    out += f'<polygon points="{X-tw/2-2:.1f},{Y-h/2-40*k:.1f} {X:.1f},{Y-h/2-62*k:.1f} {X+tw/2+2:.1f},{Y-h/2-40*k:.1f}" fill="{roof}"/>\n'
    return out

def council_tower(P, k, cx, cy, body="#CDBFA3", roof=TIGLA, stroke="#6B5A3E"):
    X, Y = P(cx, cy)
    w, h = 18 * k, 46 * k
    return (f'<rect x="{X-w/2:.1f}" y="{Y-h/2:.1f}" width="{w:.1f}" height="{h:.1f}" fill="{body}" stroke="{stroke}" stroke-width="1.5"/>'
            f'<polygon points="{X-w/2-4:.1f},{Y-h/2:.1f} {X:.1f},{Y-h/2-20*k:.1f} {X+w/2+4:.1f},{Y-h/2:.1f}" fill="{roof}"/>\n')

def compass(X, Y, r=40, face=PARCH, stroke="#8C7B5E", north=TIGLA, south="#8C7B5E", tcol="#3B3326"):
    s = f'<circle cx="{X}" cy="{Y}" r="{r}" fill="{face}" stroke="{stroke}" stroke-width="1.5"/>\n'
    s += f'<polygon points="{X},{Y-r*0.8} {X+r*0.2},{Y} {X-r*0.2},{Y}" fill="{north}"/><polygon points="{X},{Y+r*0.8} {X+r*0.2},{Y} {X-r*0.2},{Y}" fill="{south}"/>\n'
    s += text(X, Y - r - 6, "N", 18, tcol, "sans", 700, anchor="middle")
    s += text(X, Y + r + 22, "S", 16, tcol, "sans", 600, anchor="middle")
    s += text(X - r - 14, Y + 6, "V", 16, tcol, "sans", 600, anchor="middle")
    s += text(X + r + 14, Y + 6, "E", 16, tcol, "sans", 600, anchor="middle")
    return s
