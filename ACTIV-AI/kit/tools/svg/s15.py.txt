def mini_block(x, y, w, h, fl, cols=3):
    s = f'<rect x="{x-w/2:.1f}" y="{y-h:.1f}" width="{w}" height="{h}" fill="#D9D3C7" stroke="{INK2}" stroke-width="1.2"/>'
    for i in range(fl):
        for j in range(cols):
            wx = x - w / 2 + 3 + j * (w - 6) / cols
            wy = y - h + 3 + i * (h - 6) / fl
            s += f'<rect x="{wx:.1f}" y="{wy:.1f}" width="{(w-6)/cols-1.6:.1f}" height="{(h-6)/fl-1.6:.1f}" fill="#6F8FA0"/>'
    return s

def site(x, y, n, icon, col, orient=False, r=27):
    dash = ' stroke-dasharray="6 4"' if orient else ""
    stroke = OCRU_D if orient else col
    s = f'<g filter="url(#shs)"><circle cx="{x:.1f}" cy="{y:.1f}" r="{r}" fill="#FBF7EE" stroke="{stroke}" stroke-width="3.2"{dash}/></g>\n'
    s += G(x, y + 1, (r - 4) / 30, icon, filt=False)
    bx, by = x - r * 0.78, y - r * 0.78
    s += f'<circle cx="{bx:.1f}" cy="{by:.1f}" r="11" fill="{col}" stroke="{VAR}" stroke-width="2"/>'
    s += text(bx, by + 5.5, str(n), 15, VAR, "sans", 700, anchor="middle")
    return s


def base(px, py, pw, ph, P, tint, clip):
    s = f'<clipPath id="{clip}"><rect x="{px}" y="{py}" width="{pw}" height="{ph}" rx="20"/></clipPath>\n'
    s += f'<g clip-path="url(#{clip})">\n<rect x="{px}" y="{py}" width="{pw}" height="{ph}" fill="{PARCH}" filter="url(#paper)"/>\n'
    s += f'<rect x="{px}" y="{py}" width="{pw}" height="{ph}" fill="{tint}" opacity="0.45"/>\n'
    cx, cy = P(0, 0)
    s += f'<ellipse cx="{cx:.1f}" cy="{cy+20:.1f}" rx="{pw*0.42:.1f}" ry="{ph*0.42:.1f}" fill="#E9DFC8" opacity="0.5" filter="url(#blur8)"/>\n'
    riv = [P(-0.42, -0.15), P(-0.25, -0.18), P(0.0, -0.2), P(0.3, -0.25), P(0.58, -0.3)]
    rd = smooth_path(riv, closed=False)
    a0, a1 = riv[0], riv[-1]
    gid = clip + "_rg"
    s += (f'<linearGradient id="{gid}" gradientUnits="userSpaceOnUse" x1="{a0[0]:.1f}" y1="{a0[1]:.1f}" x2="{a1[0]:.1f}" y2="{a1[1]:.1f}">'
          f'<stop offset="0" stop-color="#8FBFD1" stop-opacity="0"/><stop offset="0.18" stop-color="#8FBFD1" stop-opacity="0.9"/>'
          f'<stop offset="0.82" stop-color="#8FBFD1" stop-opacity="0.9"/><stop offset="1" stop-color="#8FBFD1" stop-opacity="0"/></linearGradient>\n')
    s += f'<path d="{rd}" fill="none" stroke="url(#{gid})" stroke-width="11" stroke-linecap="round"/>\n'
    # pădurea Dumbrava (zonă orientativă, spre sud)
    rnd = random.Random(9)
    trees = []
    for i in range(70):
        u = rnd.uniform(-0.62, -0.05); v = rnd.uniform(0.9, 1.12)
        if ((u + 0.34) / 0.3) ** 2 + ((v - 1.02) / 0.13) ** 2 < 1:
            trees.append((v, u))
    for v, u in sorted(trees):
        tx, ty = P(u, v)
        s += fir(tx, ty, rnd.uniform(18, 24), rnd.choice([GREEN2, "#2F6B59", GREEN1])) + "\n"
    # centrul istoric
    s += f'<ellipse cx="{cx:.1f}" cy="{cy:.1f}" rx="30" ry="22" fill="#EFE3CC" stroke="{STONE}" stroke-width="4"/>\n'
    s += f'<ellipse cx="{cx:.1f}" cy="{cy:.1f}" rx="30" ry="22" fill="none" stroke="#C9A27E" stroke-width="1.5" stroke-dasharray="4 4"/>\n'
    s += f'<rect x="{cx-4:.1f}" y="{cy-16:.1f}" width="8" height="20" fill="{ZID}" stroke="{INK}" stroke-width="1.2"/><polygon points="{cx-6:.1f},{cy-16:.1f} {cx:.1f},{cy-26:.1f} {cx+6:.1f},{cy-16:.1f}" fill="{TIGLA}"/>\n'
    return s


def panel(px, py, pw, ph, mode):
    """mode: 'old' / 'new'. returns svg, boxes"""
    B = Boxes()
    P = mk_proj(px + 452, py + 300, 430, 285)
    tint = SEPIA if mode == "old" else "#E3ECE6"
    s = base(px, py, pw, ph, P, tint, "clip_" + mode)
    col = TIGLA if mode == "old" else PETROL

    # repere de cartier (nume de azi), foarte discrete
    ref = []
    for nm in ["Turnișor", "Lazaret", "Broscărie", "Vasile Aaron", "Gușterița", "Valea Aurie", "Dumbrăvii", "Trei Stejari", "Piața Cluj"]:
        nx, ny, st = NB[nm]
        ref.append((nm, P(nx, ny)))

    cx0, cy0 = P(0, 0)
    # eticheta panoului (înainte de repere, ca să fie evitată)
    lab = "ALTĂDATĂ" if mode == "old" else "AZI"
    sub = "repere din sec. al XIX-lea – 1936" if mode == "old" else "ultimele date din surse: 2020–2021"
    lw = measure(lab, 22, "sans", 700, letter_spacing=3) + 40
    tag = f'<rect x="{px+20}" y="{py+18}" width="{lw:.0f}" height="40" rx="20" fill="{col}"/>\n'
    tag += text(px + 20 + lw / 2, py + 45, lab, 22, VAR, "sans", 700, anchor="middle", extra='letter-spacing="3"')
    tag += text(px + 24, py + 82, sub, 15, INK2, "sans", 400, True)
    B.add("tag", px + 20, py + 18, px + 20 + lw, py + 58)
    B.text("tagsub", px + 24, py + 82, measure(sub, 15, "sans", 400, True), 15)
    B.add("compass", px + pw - 92, py + 20, px + pw - 14, py + 112)
    if mode == "old":
        # tramvaiul 1905 / 1910 (schematic)
        tr = [P(0.57, -0.21), P(0.3, -0.07), P(0.04, 0.06), P(-0.1, 0.22), P(-0.2, 0.4)]
        tr2 = [P(-0.2, 0.4), P(-0.27, 0.62), P(-0.33, 0.86)]
        s += f'<path d="{smooth_path(tr, False)}" fill="none" stroke="#6E56CF" stroke-width="4" stroke-dasharray="9 7" stroke-linecap="round"/>\n'
        s += f'<path d="{smooth_path(tr2, False)}" fill="none" stroke="#6E56CF" stroke-width="3" stroke-dasharray="3 7" stroke-linecap="round"/>\n'
        gx, gy = P(0.57, -0.21)
        B.add("garadiam", gx - 11, gy - 11, gx + 11, gy + 11)
        s += f'<rect x="{gx-7:.1f}" y="{gy-7:.1f}" width="14" height="14" rx="3" fill="{PARCH}" stroke="#4A3A99" stroke-width="3" transform="rotate(45 {gx:.1f} {gy:.1f})"/>\n'
        s += halo(gx + 16, gy + 6, "gara", 14, "#4A3A99", 700, True, "start", hw=4)
        B.text("gara", gx + 16, gy + 6, measure("gara", 14, "sans", 700, True), 14)
        tx_, ty_ = cx0 + 46, cy0 + 30
        for i, lbl in enumerate(["tramvaiul: 1905 – până la parc,", "1910 – până la marginea pădurii"]):
            s += halo(tx_, ty_ + i * 17, lbl, 14, "#4A3A99", 600, True, "start", hw=4)
            B.text("tramlbl", tx_, ty_ + i * 17, measure(lbl, 14, "sans", 600, True), 14)

    # site-uri: (n, poziție normalizată, orientativ, icon vechi, icon nou, titlu vechi, rânduri vechi, titlu nou, rânduri nou, dx, dy, ancoră)
    SITES = [
        (1, (-0.2, 0.4), True, ic_park, ic_park, "Parcul Sub Arini", ["înființat în noiembrie 1856"],
         "Parcul Sub Arini", ["68 de specii de plante, 95 de specii de păsări", "arbori de peste 150 de ani"], 38, -8, "start"),
        (2, NB["Ștrand"][:2], False, ic_pool, ic_pool_new, "Ștrandul",
         ["anii 1920 sau 12 iulie 1936, după sursă", "bazine: olimpic și pentru copii", "Turnul lui Phleps: sărituri de la 3, 5 și 10 m"],
         "Ștrandul", ["din 2008: 5.000 de locuri", "platforma de 10 m, demontată", "redeschis la 23 iunie 2021"], -165, 50, "start"),
        (3, NB["Terezian"][:2], False, ic_market, ic_park, "Târgul Vitelor", ["Viehmarktplatz, în Terezian"],
         "Parcul Terezian", ["pe locul fostului Târg al Vitelor"], 38, -8, "start"),
        (4, NB["Hipodrom"][:2], True, ic_track, ic_blocks, "Hipodromul cavaleriei",
         ["început de sec. al XIX-lea", "iunie 1924: concursuri hipice, cu", "regele Ferdinand I și regina Maria"],
         "Cartierul Hipodrom", ["hipodromul, demolat pentru blocuri", "Hipodrom I, terminat în 1972"], 38, -8, "start"),
        (5, (-0.05, 0.92), True, ic_forest_tram, ic_forest_zoo, "Pădurea Dumbrava",
         ["1910: tramvaiul, la marginea pădurii", "1929: prima grădină zoologică din România"],
         "Pădurea Dumbrava", ["grădina zoologică, din 1929"], 38, -8, "start"),
        (6, NB["Țiglari"][:2], False, ic_factory, ic_lake, "Fabrica de țiglă (Țiglari)",
         ["a lui Gustav Binder, sec. al XIX-lea", "reper, nu loc de joacă"],
         "Lacul lui Binder (Țiglari)", ["planificate în 2018: plajă, terenuri", "de volei, loc de joacă plutitor", "aug. 2020: circa 80% din lucrări gata"], 38, -8, "start"),
    ]
    for (n, (nx, ny), orient, io, inew, t_old, l_old, t_new, l_new, dx, dy, anc) in SITES:
        X, Y = P(nx, ny)
        icon = io() if mode == "old" else inew()
        s += site(X, Y, n, icon, col, orient)
        B.add(f"s{n}:med", X - 38, Y - 38, X + 27, Y + 27)
        title = t_old if mode == "old" else t_new
        lines = l_old if mode == "old" else l_new
        tx, ty = X + dx, Y + dy
        s += halo(tx, ty, title, 18, INK, 700, False, anc, hw=5)
        B.text(f"s{n}:t", tx, ty, measure(title, 18, "sans", 700), 18, anc)
        yy = ty + 19
        for ln in lines:
            s += halo(tx, yy, ln, 14, INK2, 400, True, anc, hw=4)
            B.text(f"s{n}:l{yy}", tx, yy, measure(ln, 14, "sans", 400, True), 14, anc)
            yy += 17

    # repere discrete – doar unde nu se suprapun
    for nm, (X, Y) in ref:
        w = measure(nm, 13, "sans", 600, True)
        bx0, by0, bx1, by1 = X - w / 2 - 4, Y - 14, X + w / 2 + 4, Y + 8
        clash = any(bx0 < it[3] and it[1] < bx1 and by0 < it[4] and it[2] < by1 for it in B.items)
        inside = px + 10 < bx0 and bx1 < px + pw - 10 and py + 70 < by0 and by1 < py + ph - 40
        if not clash and inside:
            s += f'<circle cx="{X:.1f}" cy="{Y-16:.1f}" r="3" fill="#B5A385"/>'
            s += halo(X, Y, nm, 13, "#9C8B6E", 600, True, "middle", hw=4, hcol="#EFE6D3")
            B.add(nm + ":ref", bx0, by0, bx1, by1)
    cx, cy = P(0, 0)
    s += halo(cx, cy + 38, "centru", 14, "#7A5636", 700, True, "middle", hw=4)
    B.text("centru", cx, cy + 38, measure("centru", 14, "sans", 700, True), 14, "middle")

    s += tag
    # busolă mică
    s += SM.compass(px + pw - 52, py + 64, r=22)
    s += text(px + pw - 20, py + ph - 16, "schematic, poziții aproximative", 14, INK2, "sans", 400, True, "end")
    B.text("note", px + pw - 20, py + ph - 16, measure("schematic, poziții aproximative", 14, "sans", 400, True), 14, "end")
    s += "</g>\n"
    s += f'<rect x="{px}" y="{py}" width="{pw}" height="{ph}" rx="20" fill="none" stroke="#2A3845" stroke-width="2"/>\n'
    return s, B
