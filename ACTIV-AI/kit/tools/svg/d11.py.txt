Y0, H = 222, 744          # cardurile coloanelor

TOPD = 16                 # adâncimea fețelor de sus (pseudo-3D)


def lighten(hexc, k):
    hexc = hexc.lstrip("#")
    r, g, b = int(hexc[0:2], 16), int(hexc[2:4], 16), int(hexc[4:6], 16)
    r = int(r + (255 - r) * k); g = int(g + (255 - g) * k); b = int(b + (255 - b) * k)
    return f"#{r:02X}{g:02X}{b:02X}"


def step(x, rank, col=None, numcol=VAR):
    """Treaptă de podium; col=None -> treaptă goală (punctată)."""
    h = HSTEP[rank]
    top = BASE - h
    o = ""
    if col:
        o += poly([(x, top), (x + 10, top - TOPD), (x + SW + 10, top - TOPD), (x + SW, top)], fill=lighten(col, 0.35))
        o += poly([(x + SW, top), (x + SW + 10, top - TOPD), (x + SW + 10, BASE - TOPD), (x + SW, BASE)], fill=col, opacity="0.55")
        o += f'<rect x="{x}" y="{top}" width="{SW}" height="{h}" fill="{col}"/>\n'
        o += text(x + SW / 2, top + min(h - 14, 64), str(rank), 50 if rank == 1 else 42, numcol, "serif", 700, anchor="middle")
    else:
        o += (f'<rect x="{x}" y="{top}" width="{SW}" height="{h}" fill="none" stroke="{DASH}" '
              f'stroke-width="2.5" stroke-dasharray="9 7"/>\n')
        o += text(x + SW / 2, top + min(h - 14, 60), "?", 40 if rank != 3 else 34, DASH, "serif", 700, anchor="middle")
        o += text(x + SW / 2, top - 12, f"locul {rank}", 15, MUTED, "sans", 500, True, anchor="middle")
    return o


def podium_x(cx):
    # ordinea clasică: 2 (stânga) – 1 (mijloc) – 3 (dreapta)
    return {2: cx - SW * 1.5, 1: cx - SW / 2, 3: cx + SW / 2}


def column_head(x, color, label, labcol, criterion, srcnote):
    o = card(x, Y0, CW, H)
    o += f'<rect x="{x}" y="{Y0}" width="{CW}" height="7" rx="3" fill="{color}"/>\n'
    pw = measure(label, 19, "sans", 700, letter_spacing=2) + 36
    o += f'<rect x="{x+28}" y="{Y0+24}" width="{pw:.1f}" height="38" rx="19" fill="{color}"/>\n'
    o += text(x + 28 + pw / 2, Y0 + 50, label, 19, labcol, "sans", 700, anchor="middle", extra='letter-spacing="2"')
    o += text(x + 28, Y0 + 90, criterion, 18, VAR, "sans", 500, True)
    o += text(x + 28, Y0 + 114, srcnote, 15, MUTED, "sans", 400, True)
    return o

def scaled(svg, x, y, k):
    return f'<g transform="translate({x:.1f},{y:.1f}) scale({k})">\n{svg}</g>\n'


# ------------------------------------------------------------------ ilustrații
def ic_center(cx, by):
    """Centrul istoric: zid cu creneluri, turn de biserică înalt, turn de poartă."""
    o = ""
    o += f'<rect x="{cx-62}" y="{by-34}" width="124" height="34" fill="#C9BFAE"/>\n'
    for i in range(7):
        o += f'<rect x="{cx-62+i*18}" y="{by-42}" width="10" height="9" fill="#C9BFAE"/>\n'
    # turn de biserică
    o += f'<rect x="{cx-6}" y="{by-84}" width="26" height="84" fill="{VAR}"/>\n'
    o += poly([(cx - 9, by - 84), (cx + 7, by - 118), (cx + 23, by - 84)], fill=TIGLA)
    o += f'<rect x="{cx+3}" y="{by-72}" width="8" height="12" rx="4" fill="{BG}"/>\n'
    # turn de poartă
    o += f'<rect x="{cx-50}" y="{by-64}" width="26" height="64" fill="{ZID}"/>\n'
    o += poly([(cx - 54, by - 64), (cx - 37, by - 86), (cx - 20, by - 64)], fill=TIGLA)
    o += f'<path d="M{cx-44},{by} v-14 a7,7 0 0 1 14,0 v14 Z" fill="{BG}"/>\n'
    # case în spate
    o += f'<rect x="{cx+30}" y="{by-50}" width="26" height="50" fill="#E3D9C6"/>\n'
    o += poly([(cx + 27, by - 50), (cx + 43, by - 66), (cx + 59, by - 50)], fill=TIGLA)
    return o


def ic_suburb(cx, by):
    """Suburbie: case mici și o biserică."""
    o = ""
    for dx, w, h in [(-62, 30, 26), (-28, 30, 30), (32, 30, 24)]:
        o += f'<rect x="{cx+dx}" y="{by-h}" width="{w}" height="{h}" fill="#E3D9C6"/>\n'
        o += poly([(cx + dx - 3, by - h), (cx + dx + w / 2, by - h - 16), (cx + dx + w + 3, by - h)], fill=TIGLA)
        o += f'<rect x="{cx+dx+10}" y="{by-h+8}" width="9" height="9" fill="{BG}" opacity="0.7"/>\n'
    # biserică cu turn
    o += f'<rect x="{cx+6}" y="{by-58}" width="20" height="58" fill="{VAR}"/>\n'
    o += poly([(cx + 4, by - 58), (cx + 16, by - 82), (cx + 28, by - 58)], fill=TIGLA)
    o += f'<rect x="{cx+12}" y="{by-46}" width="8" height="11" rx="4" fill="{BG}"/>\n'
    return o


def blocks(x, by, n_floors, w=24, col="#C9D3DC"):
    h = 6 + n_floors * 7
    o = f'<rect x="{x}" y="{by-h}" width="{w}" height="{h}" fill="{col}"/>\n'
    for f in range(n_floors):
        for c in range(3):
            o += f'<rect x="{x+3+c*7:.1f}" y="{by-h+4+f*7}" width="4" height="4" fill="{BG}" opacity="0.65"/>\n'
    return o


def tree(cx, by, r=13, col="#3E8E5A"):
    return (f'<rect x="{cx-2}" y="{by-14}" width="4" height="14" fill="#6B4E2E"/>'
            f'<circle cx="{cx}" cy="{by-14-r+3}" r="{r}" fill="{col}"/>\n')


def ic_newest(cx, by):
    """Valea Aurie: blocuri lângă pădure (Dumbrava)."""
    o = ""
    o += blocks(cx - 64, by, 8, 26)
    o += blocks(cx - 32, by, 10, 26, "#DCE3E9")
    o += blocks(cx, by, 8, 26)
    o += tree(cx + 40, by, 14, "#2F7A4B") + tree(cx + 58, by, 12, "#3E8E5A") + tree(cx + 50, by - 2, 9, "#4FA36A")
    return o


def ic_fortchurch(cx, by, sc=1.0):
    """Biserică fortificată de sat (schematic)."""
    o = f'<g transform="translate({cx},{by}) scale({sc})">'
    o += '<rect x="-34" y="-16" width="68" height="16" fill="#8C8270"/>'
    o += '<rect x="-20" y="-30" width="34" height="18" fill="#E3D9C6"/>'
    o += f'<polygon points="-23,-30 -3,-42 17,-30" fill="{TIGLA}"/>'
    o += f'<rect x="10" y="-52" width="14" height="40" fill="{VAR}"/>'
    o += f'<polygon points="8,-52 17,-66 26,-52" fill="{TIGLA}"/>'
    o += '</g>\n'
    return o
