def sheep(x, y, s=1.0, face=1):
    """oaie stilizată; (x,y) = centrul corpului"""
    f = face
    o = f'<g transform="translate({x:.1f},{y:.1f}) scale({s*f:.2f},{s:.2f})">'
    o += f'<path d="M-6,5 v8 M-1,6 v8 M5,6 v8 M9,5 v7" stroke="{INK}" stroke-width="2.2" stroke-linecap="round"/>'
    for (cx, cy, r) in [(-8, -1, 6), (-2, -4, 6.5), (5, -3, 6), (9, 1, 5), (1, 3, 7), (-7, 4, 5)]:
        o += f'<circle cx="{cx}" cy="{cy}" r="{r}" fill="{WOOL}" stroke="{INK2}" stroke-width="1"/>'
    for (cx, cy, r) in [(-8, -1, 5), (-2, -4, 5.5), (5, -3, 5), (9, 1, 4), (1, 3, 6), (-7, 4, 4)]:
        o += f'<circle cx="{cx}" cy="{cy}" r="{r}" fill="{WOOL}"/>'
    o += f'<ellipse cx="14" cy="-3" rx="4.6" ry="3.6" fill="#3B3326" transform="rotate(20 14 -3)"/>'
    o += f'<ellipse cx="12" cy="-6" rx="2.6" ry="1.3" fill="#3B3326" transform="rotate(-30 12 -6)"/>'
    o += f'<circle cx="15.2" cy="-4" r="0.9" fill="#FBF7EE"/>'
    return o + "</g>"

def flag(x, y, col=TIGLA):
    return (f'<path d="M{x:.1f},{y:.1f} V{y-30:.1f}" stroke="{INK}" stroke-width="2.2"/>'
            f'<path d="M{x:.1f},{y-30:.1f} L{x+20:.1f},{y-24:.1f} L{x:.1f},{y-17:.1f} Z" fill="{col}" stroke="{INK}" stroke-width="1.2"/>'
            f'<ellipse cx="{x:.1f}" cy="{y+1:.1f}" rx="6" ry="2" fill="#000" opacity="0.2"/>')


def medallion(x, y, n, icon, r=42):
    s = (f'<g filter="url(#sh)"><circle cx="{x:.1f}" cy="{y:.1f}" r="{r}" fill="#FBF7EE" stroke="{PETROL}" '
         f'stroke-width="4"/></g>')
    s += f'<clipPath id="mc{n}"><circle cx="{x:.1f}" cy="{y:.1f}" r="{r-3}"/></clipPath>'
    s += f'<g clip-path="url(#mc{n})">' + G(x, y + 2, (r - 6) / 31, icon, filt=False) + "</g>"
    bx, by = x - r * 0.74, y - r * 0.74
    s += (f'<circle cx="{bx:.1f}" cy="{by:.1f}" r="15" fill="{OCRU}" stroke="{VAR}" stroke-width="2.4"/>'
          + text(bx, by + 6.5, str(n), 18, BG, "sans", 700, anchor="middle"))
    return s + "\n"

    def mount_top(x):
        for (ax, ay), (bx2, by2) in zip(mpoly, mpoly[1:]):
            if min(ax, bx2) <= x <= max(ax, bx2) and ay < 990 and by2 < 990:
                t = (x - ax) / ((bx2 - ax) or 1)
                return ay + (by2 - ay) * t
        return None
