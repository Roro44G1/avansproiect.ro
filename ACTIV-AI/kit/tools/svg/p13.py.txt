def oak_leaf(L, fill, vein=None, n=4.5, base=0.45, p=0.25, fwd=0.05, W=0.27, N=300):
    """Frunză de stejar stilizată (contur parametric: lobi rotunjiți, sinusuri înguste), tulpina în (0,0), vârful în sus."""
    def env(t):
        return W * L * (math.sin(math.pi * min(1.0, t ** 1.25)) ** 0.8)
    left = []
    for i in range(N + 1):
        t = 0.06 + 0.94 * i / N
        lob = abs(math.sin(math.pi * n * (t - 0.06) / 0.94)) ** p
        h = env(t) * (base + (1 - base) * lob)
        left.append((-h, -t * L - fwd * h * lob))
    right = [(-x, y) for x, y in reversed(left)]
    pts = [(0, -0.02 * L)] + left + right
    d = "M" + " L".join(f"{x:.1f},{y:.1f}" for x, y in pts) + " Z"
    o = f'<path d="{d}" fill="{fill}" stroke="{fill}" stroke-width="{L*0.012:.1f}" stroke-linejoin="round"/>'
    o += f'<line x1="0" y1="{0.05*L:.1f}" x2="0" y2="{-0.03*L:.1f}" stroke="{fill}" stroke-width="{L*0.045:.1f}" stroke-linecap="round"/>'
    if vein:
        o += f'<line x1="0" y1="{-0.02*L:.1f}" x2="0" y2="{-0.88*L:.1f}" stroke="{vein}" stroke-width="{L*0.02:.1f}" stroke-linecap="round"/>'
    return o

def flag_trei_stejari(x, y, w, h, callouts=False):
    o = f'<g filter="url(#sh)"><rect x="{x}" y="{y}" width="{w}" height="{h}" fill="{GREEN}"/></g>\n'
    cx, cy = x + w / 2, y + h / 2
    b = h * 0.12
    o += f'<rect x="{x}" y="{cy-b/2}" width="{w}" height="{b}" fill="{VAR}"/>\n'
    o += f'<rect x="{cx-b/2}" y="{y}" width="{b}" height="{h}" fill="{VAR}"/>\n'
    R = h * 0.27
    o += f'<circle cx="{cx}" cy="{cy}" r="{R+b*0.32}" fill="{VAR}"/>\n'
    o += f'<circle cx="{cx}" cy="{cy}" r="{R}" fill="{OCRU}"/>\n'
    L = R * 0.90
    for ang in (0, 120, 240):   # trei frunze în treflă = cei trei stejari
        o += f'<g transform="translate({cx:.1f},{cy:.1f}) rotate({ang})">{oak_leaf(L, GREEN_D, OCRU)}</g>\n'
    o += f'<circle cx="{cx:.1f}" cy="{cy:.1f}" r="{R*0.07:.1f}" fill="{GREEN_D}"/>\n'
    o += f'<rect x="{x}" y="{y}" width="{w}" height="{h}" fill="url(#cloth)"/>\n'
    if callouts:
        def co(n, px, py):
            return (f'<circle cx="{px}" cy="{py}" r="19" fill="{BG}" stroke="{ORANGE}" stroke-width="3"/>'
                    + text(px, py + 7, str(n), 20, VAR, "sans", 700, anchor="middle"))
        o += co(1, x + w * 0.86, cy)
        o += co(2, cx + R * 0.58, cy - R * 0.56)
        o += co(3, cx, cy + R * 0.70)
    return o

def flag_tiglari(x, y, w, h):
    o = f'<g filter="url(#sh)"><rect x="{x}" y="{y}" width="{w}" height="{h}" fill="{LAKE}"/></g>\n'
    top = h * 0.40
    n = 5; r = w / n / 2
    # câmp de țiglă cu marginea inferioară în „solzi” (profil de țiglă)
    arcs = ""
    for k in range(n, 0, -1):
        xa = x + k * 2 * r; xb = xa - 2 * r
        arcs += f"A{r:.1f},{r:.1f} 0 0 1 {xb:.1f},{y+top:.1f} "
    o += f'<path d="M{x},{y} H{x+w} V{y+top} {arcs}Z" fill="{TIGLA}"/>\n'
    o += f'<path d="M{x+w},{y+top} {arcs}" fill="none" stroke="{VAR}" stroke-width="{h*0.035:.1f}" stroke-linejoin="round"/>\n'
    # val de var pe lac
    wy = y + h * 0.80; amp = h * 0.05; seg = w / 4
    d = f"M{x},{wy:.1f} "
    for k in range(4):
        d += f"q{seg/4:.1f},{-amp*2:.1f} {seg/2:.1f},0 t{seg/2:.1f},0 "
    o += f'<path d="{d}" fill="none" stroke="{VAR}" stroke-width="{h*0.05:.1f}" stroke-linecap="round"/>\n'
    o += f'<rect x="{x}" y="{y}" width="{w}" height="{h}" fill="url(#cloth)"/>\n'
    return o

def flag_strand(x, y, w, h):
    o = f'<g filter="url(#sh)"><rect x="{x}" y="{y}" width="{w}" height="{h}" fill="{PETROL}"/></g>\n'
    # două valuri de var
    amp = h * 0.05; seg = w / 4
    for wy in (y + h * 0.66, y + h * 0.84):
        d = f"M{x},{wy:.1f} "
        for k in range(4):
            d += f"q{seg/4:.1f},{-amp*2:.1f} {seg/2:.1f},0 t{seg/2:.1f},0 "
        o += f'<path d="{d}" fill="none" stroke="{VAR}" stroke-width="{h*0.06:.1f}" stroke-linecap="round"/>\n'
    # turnul trambulinei: corp plin, în ocru, cu zăbrele decupate; o platformă sus; arcul săriturii punctat
    px = x + w * 0.19; pb = y + h * 0.62; pt = y + h * 0.20
    hb, ht = w * 0.075, w * 0.035          # semi-lățimea la bază / sus
    o += f'<polygon points="{px-hb:.1f},{pb:.1f} {px-ht:.1f},{pt:.1f} {px+ht:.1f},{pt:.1f} {px+hb:.1f},{pb:.1f}" fill="{OCRU}"/>\n'
    def half_at(yy):
        f = (pb - yy) / (pb - pt)
        return hb + (ht - hb) * f
    for (y1, y2) in ((pb - (pb - pt) * 0.08, pb - (pb - pt) * 0.50), (pb - (pb - pt) * 0.50, pb - (pb - pt) * 0.86)):
        h1, h2 = half_at(y1) * 0.62, half_at(y2) * 0.62
        o += (f'<g stroke="{PETROL}" stroke-width="{h*0.022:.1f}" stroke-linecap="round">'
              f'<line x1="{px-h1:.1f}" y1="{y1:.1f}" x2="{px+h2:.1f}" y2="{y2:.1f}"/>'
              f'<line x1="{px+h1:.1f}" y1="{y1:.1f}" x2="{px-h2:.1f}" y2="{y2:.1f}"/></g>\n')
    sl = h * 0.05
    o += f'<rect x="{px-ht-w*0.02:.1f}" y="{pt-sl:.1f}" width="{w*0.30:.1f}" height="{sl:.1f}" rx="{sl/2:.1f}" fill="{OCRU}"/>\n'
    tx0, ty0 = px - ht - w * 0.02 + w * 0.30, pt - sl / 2
    o += (f'<path d="M{tx0:.1f},{ty0:.1f} q{w*0.14:.1f},{-h*0.14:.1f} {w*0.25:.1f},{h*0.40:.1f}" fill="none" stroke="{VAR}" '
          f'stroke-width="{h*0.028:.1f}" stroke-dasharray="{h*0.04:.1f} {h*0.05:.1f}" stroke-linecap="round"/>\n')
    o += f'<rect x="{x}" y="{y}" width="{w}" height="{h}" fill="url(#cloth)"/>\n'
    return o
