def pill(x, y, lab, col, txt=BG, size=15):
    w = measure(lab, size, "sans", 700, letter_spacing=1.5) + 28
    o = f'<rect x="{x:.1f}" y="{y}" width="{w:.1f}" height="30" rx="15" fill="{col}"/>\n'
    o += text(x + w / 2, y + 20, lab, size, txt, "sans", 700, anchor="middle", extra='letter-spacing="1.5"')
    return o, w


def head(x, y, w, lab, col, sub=None, txt=BG):
    o, pw = pill(x + 22, y + 20, lab, col, txt)
    if sub:
        assert x + 22 + pw + 12 + measure(sub, 14, "sans", 400, True) <= x + w - 16, sub
        o += text(x + 22 + pw + 12, y + 40, sub, 14, MUTED, "sans", 400, True)
    return o


def block(x, y, body, maxw, size=14, fill=CEATA, lh=17, weight=400, anchor="middle", italic=False):
    lines = wrap(body, maxw, size, "sans", weight, italic)
    o = ""
    for k, ln in enumerate(lines):
        o += text(x, y + k * lh, ln, size, fill, "sans", weight, italic, anchor)
    return o, y + (len(lines) - 1) * lh, len(lines)

def granules(x0, x1, y0, y1, n, rmin=1.6, rmax=3.6, op=1.0, pepper=True):
    o = ""
    for _ in range(n):
        gx = random.uniform(x0, x1); gy = random.uniform(y0, y1)
        rx = random.uniform(rmin, rmax); ry = rx * random.uniform(0.6, 1.0)
        o += (f'<ellipse cx="{gx:.1f}" cy="{gy:.1f}" rx="{rx:.1f}" ry="{ry:.1f}" fill="{FAT}" opacity="{op}" '
              f'transform="rotate({random.randint(0, 180)} {gx:.1f} {gy:.1f})"/>')
    if pepper:
        for _ in range(n // 6):
            gx = random.uniform(x0, x1); gy = random.uniform(y0, y1)
            o += f'<circle cx="{gx:.1f}" cy="{gy:.1f}" r="{random.uniform(0.9, 1.6):.1f}" fill="#1E1412" opacity="{op}"/>'
    return o + "\n"

def mini_salami(cx, cy, length, r, mold_frac=0.0, vertical=False):
    """Salam mic, opțional parțial acoperit cu floare albă."""
    if vertical:
        o = f'<rect x="{cx-r}" y="{cy-length/2}" width="{2*r}" height="{length}" rx="{r}" fill="{MEAT_D}"/>'
        if mold_frac > 0:
            o += (f'<rect x="{cx-r}" y="{cy-length/2}" width="{2*r}" height="{length*mold_frac:.1f}" rx="{r}" '
                  f'fill="{MOLD}"/>')
        return o
    o = f'<rect x="{cx-length/2}" y="{cy-r}" width="{length}" height="{2*r}" rx="{r}" fill="{MEAT_D}"/>'
    if mold_frac > 0:
        o += (f'<rect x="{cx-length/2 + length*(1-mold_frac):.1f}" y="{cy-r}" width="{length*mold_frac:.1f}" '
              f'height="{2*r}" rx="{r}" fill="{MOLD}"/>')
    return o
