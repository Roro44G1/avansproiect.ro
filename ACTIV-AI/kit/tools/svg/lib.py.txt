_cache = {}

def _font(fam, size, weight, italic):
    key = (fam, weight, italic, size)
    if key not in _cache:
        path = FONT_FILES.get((fam, weight, italic)) or FONT_FILES[(fam, 400, False)]
        _cache[key] = ImageFont.truetype(path, size)
    return _cache[key]

def measure(text, size, fam="sans", weight=400, italic=False, letter_spacing=0):
    f = _font(fam, size, weight, italic)
    w = f.getlength(text)
    return w + letter_spacing * max(len(text) - 1, 0)

def esc(s):
    return html.escape(s, quote=False)

def wrap(text, maxw, size, fam="sans", weight=400, italic=False):
    words = text.split()
    lines, cur = [], ""
    for w in words:
        t = (cur + " " + w).strip()
        if measure(t, size, fam, weight, italic) <= maxw or not cur:
            cur = t
        else:
            lines.append(cur); cur = w
    if cur:
        lines.append(cur)
    return lines

def text(x, y, s, size, fill=VAR, fam="sans", weight=400, italic=False, anchor="start", extra=""):
    ff = SANS if fam == "sans" else SERIF
    st = ' font-style="italic"' if italic else ""
    return (f'<text x="{x:.1f}" y="{y:.1f}" font-family="{ff}" font-size="{size}" font-weight="{weight}"'
            f' fill="{fill}" text-anchor="{anchor}"{st}{(" " + extra) if extra else ""}>{esc(s)}</text>\n')

def para(x, y, s, size, maxw, fill=VAR, fam="sans", weight=400, italic=False, lh=None, anchor="start"):
    """Wrapped paragraph; returns (svg, next_y)."""
    lh = lh or round(size * 1.38)
    out = ""
    for ln in wrap(s, maxw, size, fam, weight, italic):
        out += text(x, y, ln, size, fill, fam, weight, italic, anchor)
        y += lh
    return out, y

def frame_head(chip_label, chip_color, title, subtitle=None, title_size=60):
    cw = measure(chip_label, 20, "sans", 700, letter_spacing=2) + 48
    s = f'<rect width="1920" height="1080" fill="{BG}"/>\n'
    s += '<g transform="translate(80,70)">\n'
    s += f'<rect x="0" y="0" rx="22" ry="22" width="{cw:.0f}" height="44" fill="{chip_color}"/>\n'
    s += (f'<text x="{cw/2:.1f}" y="30" text-anchor="middle" font-family="{SANS}" font-weight="700" '
          f'font-size="20" letter-spacing="2" fill="{BG}">{esc(chip_label)}</text>\n')
    s += (f'<text x="0" y="120" font-family="{SERIF}" font-weight="700" font-size="{title_size}" '
          f'fill="{VAR}">{esc(title)}</text>\n</g>\n')
    if subtitle:
        s += text(1840, 100, subtitle, 24, CEATA, "sans", 400, True, "end")
    return s

def svg_open():
    return ('<svg xmlns="http://www.w3.org/2000/svg" width="1920" height="1080" viewBox="0 0 1920 1080">\n' + DEFS)

def svg_close():
    return "</svg>\n"

def card(x, y, w, h, fill=CARD, stroke="#2A3845", rx=22, extra=""):
    return f'<rect x="{x}" y="{y}" width="{w}" height="{h}" rx="{rx}" fill="{fill}" stroke="{stroke}" stroke-width="1.5" {extra}/>\n'

def pin(x, y, n, color=TIGLA, r=19, textcol=VAR):
    s = f'<g filter="url(#shs)"><circle cx="{x}" cy="{y}" r="{r}" fill="{color}" stroke="{VAR}" stroke-width="2.5"/>'
    s += text(x, y + 7, str(n), 20, textcol, "sans", 700, anchor="middle")
    return s + "</g>\n"

def poly(points, **kw):
    attrs = " ".join(f'{k.replace("_","-")}="{v}"' for k, v in kw.items())
    return f'<polygon points="{" ".join(f"{px:.1f},{py:.1f}" for px, py in points)}" {attrs}/>\n'

def smooth_path(points, closed=True, tension=0.5):
    """Catmull-Rom to cubic Bezier."""
    n = len(points)
    if closed:
        pts = points
    else:
        pts = points
    d = f"M{pts[0][0]:.1f},{pts[0][1]:.1f}"
    rng = range(n) if closed else range(n - 1)
    for i in rng:
        p0 = pts[(i - 1) % n]; p1 = pts[i]; p2 = pts[(i + 1) % n]; p3 = pts[(i + 2) % n]
        if not closed:
            p0 = pts[max(i - 1, 0)]; p3 = pts[min(i + 2, n - 1)]
        c1 = (p1[0] + (p2[0] - p0[0]) * tension / 3, p1[1] + (p2[1] - p0[1]) * tension / 3)
        c2 = (p2[0] - (p3[0] - p1[0]) * tension / 3, p2[1] - (p3[1] - p1[1]) * tension / 3)
        d += f" C{c1[0]:.1f},{c1[1]:.1f} {c2[0]:.1f},{c2[1]:.1f} {p2[0]:.1f},{p2[1]:.1f}"
    if closed:
        d += " Z"
    return d

def write(path, body):
    with open(path, "w", encoding="utf-8") as f:
        f.write(svg_open() + body + svg_close())
