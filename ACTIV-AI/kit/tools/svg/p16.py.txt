def mirror(pts):
    """Puncte pentru jumătatea stângă (x<=0), de sus în jos -> contur închis simetric."""
    right = [(-x, y) for x, y in reversed(pts)]
    return pts + right


def P(pts):
    return " ".join(f"{x:.1f},{y:.1f}" for x, y in pts)

    o += f'<circle cx="{cx-24}" cy="{cy-24}" r="6" fill="#B08860"/><circle cx="{cx+24}" cy="{cy-24}" r="6" fill="#B08860"/>'
    o += f'<ellipse cx="{cx}" cy="{cy}" rx="34" ry="31" fill="#7A5230"/>'
    o += f'<ellipse cx="{cx}" cy="{cy+12}" rx="16" ry="12" fill="#C2A07A"/>'
    o += f'<ellipse cx="{cx}" cy="{cy+6}" rx="6" ry="4.5" fill="{DARK}"/>'
    o += f'<circle cx="{cx-13}" cy="{cy-6}" r="3.4" fill="{DARK}"/><circle cx="{cx+13}" cy="{cy-6}" r="3.4" fill="{DARK}"/>'
    o += f'<path d="M{cx},{cy+11} v5 M{cx},{cy+16} q-5,4 -9,1 M{cx},{cy+16} q5,4 9,1" fill="none" stroke="{DARK}" stroke-width="1.8" stroke-linecap="round"/>'
    return o


def head_chamois(cx, cy):
    o = ""
    for sg in (-1, 1):  # coarnele scurte, întoarse înapoi la vârf
        o += f'<path d="M{cx+sg*8},{cy-22} C{cx+sg*9},{cy-44} {cx+sg*12},{cy-54} {cx+sg*22},{cy-50}" fill="none" stroke="{DARK}" stroke-width="5" stroke-linecap="round"/>'
        o += f'<ellipse cx="{cx+sg*26}" cy="{cy-18}" rx="12" ry="6" transform="rotate({sg*-25} {cx+sg*26} {cy-18})" fill="#8C6A48"/>'
    o += f'<path d="M{cx-20},{cy-24} q20,-10 40,0 q4,26 -8,46 q-12,10 -24,0 q-12,-20 -8,-46 Z" fill="#E9DFCF"/>'
    for sg in (-1, 1):  # dungile închise ale feței (desen stilizat)
        o += f'<path d="M{cx+sg*13},{cy-20} q{sg*2},22 {sg*-3},38" fill="none" stroke="#5A4330" stroke-width="7" stroke-linecap="round"/>'
        o += f'<circle cx="{cx+sg*12}" cy="{cy-10}" r="3" fill="{DARK}"/>'
    o += f'<ellipse cx="{cx}" cy="{cy+20}" rx="6" ry="4" fill="{DARK}"/>'
    return o


def head_wolf(cx, cy):
    o = ""
    for sg in (-1, 1):
        o += f'<polygon points="{cx+sg*8},{cy-20} {cx+sg*26},{cy-48} {cx+sg*34},{cy-12}" fill="#6F777F"/>'
        o += f'<polygon points="{cx+sg*15},{cy-20} {cx+sg*26},{cy-38} {cx+sg*29},{cy-16}" fill="#C9CED3"/>'
    o += f'<path d="M{cx-34},{cy-14} q34,-20 68,0 q4,18 -10,30 l-12,16 q-12,8 -24,0 l-12,-16 q-14,-12 -10,-30 Z" fill="#8E959C"/>'
    o += f'<path d="M{cx-30},{cy+2} q12,8 18,22 l12,10 l12,-10 q6,-14 18,-22 q-6,20 -18,30 q-12,8 -24,0 q-12,-10 -18,-30 Z" fill="#DCDFE2"/>'
    o += f'<path d="M{cx-9},{cy-14} h18 l-3,26 h-12 Z" fill="#B7BCC1"/>'
    o += f'<ellipse cx="{cx}" cy="{cy+14}" rx="7" ry="5" fill="{DARK}"/>'
    for sg in (-1, 1):
        o += f'<path d="M{cx+sg*20},{cy-6} q{sg*-6},-6 {sg*-12},0 q{sg*6},4 {sg*12},0 Z" fill="#E2A62B" stroke="{DARK}" stroke-width="1.5"/>'
    return o
