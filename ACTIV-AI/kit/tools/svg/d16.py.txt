def pill(x, y, lab, col, txt=BG, size=15):
    w = measure(lab, size, "sans", 700, letter_spacing=1.5) + 28
    o = f'<rect x="{x:.1f}" y="{y}" width="{w:.1f}" height="30" rx="15" fill="{col}"/>\n'
    o += text(x + w / 2, y + 20, lab, size, txt, "sans", 700, anchor="middle", extra='letter-spacing="1.5"')
    return o, w


def head(x, y, w, lab, col, sub=None, txt=BG):
    o, pw = pill(x + 22, y + 20, lab, col, txt)
    if sub:

    return o


def block(cx, y, body, maxw, size=14, fill=CEATA, lh=17, weight=400, anchor="middle"):
    """Text centrat (sau aliniat) pe mai multe rânduri; întoarce (svg, y_ultim_rând, lățime_max)."""
    lines = wrap(body, maxw, size, "sans", weight)
    o = ""
    wmax = 0
    for k, ln in enumerate(lines):
        o += text(cx, y + k * lh, ln, size, fill, "sans", weight, anchor=anchor)
        wmax = max(wmax, measure(ln, size, "sans", weight))
    return o, y + (len(lines) - 1) * lh, wmax

def shield(x, y, w, h, named, num=None):
    d = (f"M{x},{y} h{w} v{h*0.5:.1f} q0,{h*0.35:.1f} -{w/2:.1f},{h*0.5:.1f} "
         f"q-{w/2:.1f},-{h*0.15:.1f} -{w/2:.1f},-{h*0.5:.1f} Z")
    if named:
        o = f'<path d="{d}" fill="{OCRU}" stroke="#F2C766" stroke-width="1.5"/>'
        o += text(x + w / 2, y + h * 0.58, str(num), 17, BG, "sans", 700, anchor="middle")
    else:
        o = f'<path d="{d}" fill="none" stroke="{MUTED}" stroke-width="1.8" stroke-dasharray="4 3"/>'
        o += text(x + w / 2, y + h * 0.58, "?", 16, MUTED, "sans", 700, anchor="middle")
    return o + "\n"
