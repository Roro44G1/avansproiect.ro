def waves(x0, x1, y, col=WATER_L, amp=3, step=14, sw=2):
    d = f"M{x0:.1f},{y}"
    x = x0; up = True
    while x < x1:
        d += f" q{step/2:.1f},{-amp if up else amp} {step:.1f},0"
        x += step; up = not up
    return f'<path d="{d}" fill="none" stroke="{col}" stroke-width="{sw}" stroke-linecap="round"/>\n'


def tree_round(cx, by, r, col=GREEN):
    return (f'<rect x="{cx-3}" y="{by-r-6}" width="6" height="{r+6}" fill="{WOOD_D}"/>'
            f'<circle cx="{cx}" cy="{by-r-10}" r="{r}" fill="{col}"/>'
            f'<circle cx="{cx-r*0.35:.1f}" cy="{by-r-14}" r="{r*0.55:.1f}" fill="{GREEN_L}" opacity="0.55"/>\n')


def lake(cx, cy, rx, ry):
    o = f'<ellipse cx="{cx}" cy="{cy}" rx="{rx}" ry="{ry}" fill="{WATER}"/>\n'
    o += f'<ellipse cx="{cx}" cy="{cy-ry*0.25:.1f}" rx="{rx*0.92:.1f}" ry="{ry*0.55:.1f}" fill="#2AAAB6" opacity="0.35"/>\n'
    o += waves(cx - rx * 0.6, cx - rx * 0.1, cy - 3, WATER_L, 2, 10, 1.6)
    o += waves(cx + rx * 0.05, cx + rx * 0.55, cy + ry * 0.35, WATER_L, 2, 10, 1.6)

    """Moară de apă: căsuță de lemn cu roată."""
    o = f'<rect x="{cx-30}" y="{by-56}" width="62" height="56" fill="{WOOD}"/>\n'
    for k in range(5):
        o += f'<line x1="{cx-30}" y1="{by-48+k*11}" x2="{cx+32}" y2="{by-48+k*11}" stroke="{WOOD_D}" stroke-width="1.5"/>\n'
    o += poly([(cx - 40, by - 54), (cx + 1, by - 92), (cx + 42, by - 54)], fill=ROOF)
    o += f'<rect x="{cx-6}" y="{by-30}" width="16" height="30" fill="{WOOD_D}"/>\n'
    # roata
    wx, wy, wr = cx - 50, by - 20, 26
    o += f'<circle cx="{wx}" cy="{wy}" r="{wr}" fill="none" stroke="{WOOD_D}" stroke-width="5"/>\n'
    for k in range(8):
        a = k * math.pi / 4
        o += (f'<line x1="{wx}" y1="{wy}" x2="{wx + wr*math.cos(a):.1f}" y2="{wy + wr*math.sin(a):.1f}" '
              f'stroke="{WOOD}" stroke-width="3"/>\n')
        o += (f'<rect x="{wx + (wr+2)*math.cos(a) - 4:.1f}" y="{wy + (wr+2)*math.sin(a) - 4:.1f}" width="8" height="8" '
              f'fill="{WOOD}" transform="rotate({math.degrees(a):.0f} {wx + (wr+2)*math.cos(a):.1f} {wy + (wr+2)*math.sin(a):.1f})"/>\n')
    o += f'<circle cx="{wx}" cy="{wy}" r="5" fill="{ROOF}"/>\n'
    # jgheabul
    o += f'<path d="M{wx-4},{wy-wr-14} L{wx+34},{wy-wr-24}" stroke="{WOOD}" stroke-width="6"/>\n'
    o += f'<path d="M{wx-6},{wy-wr-10} q-4,10 0,20" stroke="{WATER_L}" stroke-width="3" fill="none"/>\n'
    return o
