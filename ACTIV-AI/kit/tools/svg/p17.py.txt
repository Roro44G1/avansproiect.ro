def P(pts):
    return " ".join(f"{x:.1f},{y:.1f}" for x, y in pts)


def shield_d(W, H):
    """Scut „în formă de fier de călcat” (heater), centrat pe x=0, sus la y=0."""
    w = W / 2
    return (f"M{-w},0 L{w},0 L{w},{0.5*H:.1f} C{w},{0.80*H:.1f} {0.20*W:.1f},{0.93*H:.1f} 0,{H} "
            f"C{-0.20*W:.1f},{0.93*H:.1f} {-w},{0.80*H:.1f} {-w},{0.5*H:.1f} Z")


def sun(cx, cy, r, n=16):
    o = ""
    for i in range(n):
        a = 2 * math.pi * i / n - math.pi / 2
        L = r * (1.85 if i % 2 == 0 else 1.5)
        hw = math.radians(7.5 if i % 2 == 0 else 6)
        p1 = (cx + r * 0.92 * math.cos(a - hw * 1.6), cy + r * 0.92 * math.sin(a - hw * 1.6))
        p2 = (cx + L * math.cos(a), cy + L * math.sin(a))
        p3 = (cx + r * 0.92 * math.cos(a + hw * 1.6), cy + r * 0.92 * math.sin(a + hw * 1.6))
        o += f'<polygon points="{P([p1, p2, p3])}" fill="url(#au17)" stroke="{OUTL}" stroke-width="2.2" stroke-linejoin="round"/>'
    o += f'<circle cx="{cx}" cy="{cy}" r="{r}" fill="url(#au17)" stroke="{OUTL}" stroke-width="2.6"/>'
    o += f'<circle cx="{cx}" cy="{cy}" r="{r*0.68:.1f}" fill="none" stroke="{AU_D}" stroke-width="2"/>'
    return o + "\n"


def bolt(cx, cy, h, flip=False):
    sg = -1 if flip else 1
    pts = [(0.10, -0.5), (-0.26, 0.04), (-0.02, 0.04), (-0.12, 0.5), (0.26, -0.08), (0.02, -0.08), (0.14, -0.5)]
    pts = [(cx + sg * x * h, cy + y * h) for x, y in pts]
    return f'<polygon points="{P(pts)}" fill="url(#au17)" stroke="{OUTL}" stroke-width="2.4" stroke-linejoin="round"/>\n'


def wrench(cx, cy, L):
    """Cheie fixă verticală, de argint: cap deschis sus, inel jos. (cx,cy) = centrul."""
    hw = L * 0.075
    top = cy - L / 2; bot = cy + L / 2
    o = f'<g stroke="{OUTL}" stroke-width="2.6" stroke-linejoin="round">'
    # mânerul
    o += f'<rect x="{cx-hw:.1f}" y="{top+L*0.16:.1f}" width="{2*hw:.1f}" height="{L*0.66:.1f}" rx="{hw*0.6:.1f}" fill="url(#ar17)"/>'
    # capul deschis (sus) – cerc cu o fantă
    R = L * 0.17
    hy = top + R
    gap = R * 0.5
    a0 = math.asin(gap / R)
    x1 = cx - gap; x2 = cx + gap
    yj = hy - R * math.cos(a0)
    d = (f"M{x1:.1f},{yj:.1f} L{x1:.1f},{hy+R*0.12:.1f} L{x2:.1f},{hy+R*0.12:.1f} L{x2:.1f},{yj:.1f} "
         f"A{R:.1f},{R:.1f} 0 1 1 {x1:.1f},{yj:.1f} Z")
    o += f'<path d="{d}" fill="url(#ar17)"/>'
    # inelul (jos) cu gaură hexagonală
    R2 = L * 0.14
    ry = bot - R2
    o += f'<circle cx="{cx:.1f}" cy="{ry:.1f}" r="{R2:.1f}" fill="url(#ar17)"/>'
    hexr = R2 * 0.52
    hexp = [(cx + hexr * math.cos(math.pi / 6 + k * math.pi / 3), ry + hexr * math.sin(math.pi / 6 + k * math.pi / 3)) for k in range(6)]
    o += f'<polygon points="{P(hexp)}" fill="{AZ_D}"/>'
    o += '</g>'
    # o linie de lumină pe mâner
    o += f'<line x1="{cx-hw*0.35:.1f}" y1="{top+L*0.26:.1f}" x2="{cx-hw*0.35:.1f}" y2="{top+L*0.72:.1f}" stroke="#FFFFFF" stroke-width="2" opacity="0.7"/>'
    return o + "\n"


def chevron_panel(W, H, apex_y, foot_y, d, clip_id):
    """Căprior de argint, împărțit în celule albastre (panou solar pe acoperiș)."""
    o = ""
    w = W / 2 + 30
    for sg in (-1, 1):
        ex, ey = sg * w, foot_y          # capătul brațului
        L = math.hypot(ex, ey - apex_y) + 60
        ang = math.degrees(math.atan2(ey - apex_y, ex))
        cid = f"{clip_id}{'L' if sg < 0 else 'R'}"
        half = f"M0,-10 L{sg*(W/2+60)},-10 L{sg*(W/2+60)},{H+10} L0,{H+10} Z"
        o += f'<clipPath id="{cid}"><path d="{half}"/></clipPath>'
        o += f'<g clip-path="url(#{cid})"><g transform="translate(0,{apex_y}) rotate({ang:.3f})">'
        # banda (grosimea măsurată perpendicular pe braț)
        t = d * math.cos(math.radians(abs(ang) if abs(ang) < 90 else 180 - abs(ang)))
        x0 = -40
        yb = 0 if sg > 0 else -t          # banda stă sub linia care trece prin vârf, pe ambele brațe
        o += f'<rect x="{x0}" y="{yb:.1f}" width="{L+40:.1f}" height="{t:.1f}" fill="url(#ar17)"/>'
        # celulele
        rows, cw = 2, t / 2 * 1.25
        gap = 3.2
        ncols = int((L + 40) / cw) + 2
        for r_ in range(rows):
            for c in range(ncols):
                cx0 = x0 + c * cw
                o += (f'<rect x="{cx0+gap:.1f}" y="{yb+r_*t/2+gap:.1f}" width="{cw-2*gap:.1f}" height="{t/2-2*gap:.1f}" '
                      f'fill="{AZ_D}" stroke="{AZ_L}" stroke-width="0.8"/>')
                o += (f'<line x1="{cx0+gap+3:.1f}" y1="{yb+r_*t/2+gap+3:.1f}" x2="{cx0+cw*0.55:.1f}" y2="{yb+r_*t/2+gap+3:.1f}" '
                      f'stroke="#9CC0EE" stroke-width="1.4" opacity="0.65"/>')
        o += f'<line x1="{x0}" y1="{yb:.1f}" x2="{L:.1f}" y2="{yb:.1f}" stroke="{OUTL}" stroke-width="3"/>'
        o += f'<line x1="{x0}" y1="{yb+t:.1f}" x2="{L:.1f}" y2="{yb+t:.1f}" stroke="{OUTL}" stroke-width="3"/>'
        o += '</g></g>\n'
    return o


def blazon(W, H, uid, pins=False):
    """Blazonul complet, cu originea în mijlocul marginii de sus a scutului."""
    o = f'<clipPath id="sc{uid}"><path d="{shield_d(W, H)}"/></clipPath>\n'
    o += f'<g clip-path="url(#sc{uid})">'
    o += f'<rect x="{-W/2-5}" y="-5" width="{W+10}" height="{H+10}" fill="url(#az17)"/>\n'
    apex_y = 0.36 * H; foot_y = 0.84 * H; d = 0.15 * H
    o += chevron_panel(W, H, apex_y, foot_y, d, f"cv{uid}")
    o += sun(0, 0.175 * H, 0.085 * W)
    o += bolt(-0.33 * W, 0.24 * H, 0.21 * H)
    o += bolt(0.33 * W, 0.24 * H, 0.21 * H, flip=True)
    o += wrench(0, 0.77 * H, 0.31 * H)
    o += f'<rect x="{-W/2-5}" y="-5" width="{W+10}" height="{H+10}" fill="url(#gloss17)"/>'
    o += '</g>\n'
    o += f'<path d="{shield_d(W, H)}" fill="none" stroke="{OUTL}" stroke-width="{max(2.5, W*0.014):.1f}"/>\n'
    return o
