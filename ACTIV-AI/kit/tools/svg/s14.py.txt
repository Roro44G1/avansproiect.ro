def island(x, y, rx, ry, seed, land="#BFD49B", orient=False):
    rnd = random.Random(seed)
    pts = []
    n = 14
    for i in range(n):
        a = 2 * math.pi * i / n + rnd.uniform(-0.12, 0.12)
        k = rnd.uniform(0.86, 1.12)
        pts.append((x + rx * k * math.cos(a), y + ry * k * math.sin(a)))
    d = smooth_path(pts, closed=True, tension=0.8)
    inner = [(x + (px - x) * 0.84, y + (py - y) * 0.80 - 2) for px, py in pts]
    di = smooth_path(inner, closed=True, tension=0.8)
    s = f'<path d="{d}" fill="#2A8C8F" opacity="0.35" stroke="#2A8C8F" stroke-width="22" stroke-linejoin="round" filter="url(#blur3)"/>\n'
    s += f'<path d="{d}" fill="none" stroke="#CFEDEB" stroke-width="2" stroke-dasharray="3 7" opacity="0.6" transform="translate(0,0)"/>\n'
    s += f'<path d="{d}" fill="{SAND}" stroke="{SAND_D}" stroke-width="2" filter="url(#sh)"/>\n'
    s += f'<path d="{di}" fill="{land}"/>\n'
    if orient:
        big = [(x + (px - x) * 1.13, y + (py - y) * 1.16) for px, py in pts]
        s += f'<path d="{smooth_path(big, True, 0.8)}" fill="none" stroke="{OCRU}" stroke-width="2.6" stroke-dasharray="9 7"/>\n'
    return s


def house(x, y, w=16, h=11, roof=TIGLA):
    return (f'<rect x="{x-w/2:.1f}" y="{y-h:.1f}" width="{w}" height="{h}" fill="#F1E6D2" stroke="{INK2}" stroke-width="1"/>'
            f'<polygon points="{x-w/2-2:.1f},{y-h:.1f} {x:.1f},{y-h-8:.1f} {x+w/2+2:.1f},{y-h:.1f}" fill="{roof}" stroke="{INK2}" stroke-width="1"/>\n')


def block(x, y, w=26, h=36, floors=6, cols=3):
    s = f'<rect x="{x-w/2:.1f}" y="{y-h:.1f}" width="{w}" height="{h}" fill="#D9D3C7" stroke="{INK2}" stroke-width="1.3"/>'
    for i in range(floors):
        for j in range(cols):
            wx = x - w / 2 + 4 + j * (w - 8) / cols
            wy = y - h + 4 + i * (h - 8) / floors
            s += f'<rect x="{wx:.1f}" y="{wy:.1f}" width="{(w-8)/cols-2:.1f}" height="{(h-8)/floors-2:.1f}" fill="#6F8FA0"/>'
    return s + "\n"

def rose(x, y, r=46):
    s = f'<circle cx="{x}" cy="{y}" r="{r}" fill="none" stroke="#CFEDEB" stroke-width="1.5" opacity="0.6"/>\n'
    s += f'<circle cx="{x}" cy="{y}" r="{r*0.72:.1f}" fill="none" stroke="#CFEDEB" stroke-width="1" stroke-dasharray="2 4" opacity="0.6"/>\n'
    for a in range(0, 360, 45):
        L = r * (0.95 if a % 90 == 0 else 0.55)
        ca, sa = math.cos(math.radians(a - 90)), math.sin(math.radians(a - 90))
        cl, sl = math.cos(math.radians(a - 90 + 90)), math.sin(math.radians(a - 90 + 90))
        tip = (x + L * ca, y + L * sa); b1 = (x + 7 * cl, y + 7 * sl); b2 = (x - 7 * cl, y - 7 * sl)
        col = TIGLA if a == 0 else (VAR if a % 90 == 0 else "#7FB8B8")
        s += f'<polygon points="{tip[0]:.1f},{tip[1]:.1f} {b1[0]:.1f},{b1[1]:.1f} {x},{y} {b2[0]:.1f},{b2[1]:.1f}" fill="{col}" stroke="{SEA_HALO}" stroke-width="1"/>\n'
    s += f'<circle cx="{x}" cy="{y}" r="4" fill="{OCRU}"/>\n'
    for lab, dx, dy in (("N", 0, -r - 10), ("S", 0, r + 24), ("V", -r - 18, 6), ("E", r + 18, 6)):
        s += text(x + dx, y + dy, lab, 20 if lab == "N" else 17, VAR, "sans", 700, anchor="middle")
    return s

    for nm, (nx, ny, st) in NB.items():
        poetic, spec, deco, rx, ry, land, dx, dy, anc = ISL[nm]
        X, Y = P(nx, ny)
        isl_svg += island(X, Y, rx, ry, seed=sum(map(ord, nm)) + 7, land=land, orient=(st == "o"))

            name_parts.append(("  · poziție orientativă", OCRU, 600, True, 14, "sans"))
        lab_svg += rich(tx, ty, [(poetic, OCRU, 500, True)], 16, anc, SEA_HALO, 5)

        B.text(nm + ":name", tx, ty + 27, rich_width(name_parts, 24), 24, anc)
        B.text(nm + ":spec", tx, ty + 48, rich_width([(spec, VAR, 400, False)], 15), 15, anc)

    lg_body += p
    ky = yy + 6
    lg_body += f'<path d="M{LGX+18},{ky-5} q10,-9 20,0 t20,0" fill="none" stroke="{OCRU}" stroke-width="2.6" stroke-dasharray="8 6"/>'
    lg_body += text(LGX + 70, ky, "contur punctat = poziție orientativă", 15, VAR, "sans", 400, True)
    LGH = ky + 14 - LGY
    lg = f'<rect x="{LGX}" y="{LGY}" width="{LGW}" height="{LGH}" rx="14" fill="{BG}" fill-opacity="0.82" stroke="{OCRU}" stroke-width="1.5"/>\n' + lg_body
    B.add("legend", LGX, LGY, LGX + LGW, LGY + LGH)
    print("legend bottom", LGY + LGH)
    note = "Hartă schematică, poziții aproximative – direcțiile urmează harta reală"
    key = text(1010, 986, note, 15, "#CFE0E0", "sans", 400, True, anchor="middle")
    B.text("note", 1010, 986, measure(note, 15, "sans", 400, True), 15, "middle")

    # roza vânturilor (jos-dreapta) + corabie
    RX, RY = 1752, 926
    B.add("rose", RX - 62, RY - 62, RX + 62, RY + 66)
    SHX, SHY = 1215, 300
    B.add("ship", SHX - 42, SHY - 40, SHX + 42, SHY + 26)

    # valuri decorative (evită insulele și etichetele)
    rnd = random.Random(21)
    waves = ""
    placed = 0
    tries = 0
    while placed < 70 and tries < 4000:
        tries += 1
        wx = rnd.uniform(MX + 20, MX + MW - 40); wy = rnd.uniform(MY + 20, MY + MH - 20)
        ok = True
        for it in B.items:
            if it[1] - 12 < wx + 26 and wx - 4 < it[3] + 12 and it[2] - 10 < wy + 4 and wy - 6 < it[4] + 10:
                ok = False; break
        if ok:
            waves += f'<path d="M{wx:.1f},{wy:.1f} q6,-5 12,0 t12,0" fill="none" stroke="#CFEDEB" stroke-width="1.4" opacity="0.35"/>\n'
            placed += 1
    s += waves
    s += isl_svg
    s += ship(SHX, SHY, 1.0)
    s += rose(RX, RY, 36)
    s += lab_svg
    s += lg + key
    s += "</g>\n"
    s += f'<rect x="{MX}" y="{MY}" width="{MW}" height="{MH}" rx="22" fill="none" stroke="#2A3845" stroke-width="2"/>\n'
    s += frame_foot()
    write(OUT, s)
    print(OUT)
    bad = B.report(bounds=(MX + 4, MY + 4, MX + MW - 4, MY + MH - 4))
    print("overlaps:", len(bad))
