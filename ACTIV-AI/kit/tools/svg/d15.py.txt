def pill(x, y, lab, col, txt=BG):
    w = measure(lab, 15, "sans", 700, letter_spacing=1.5) + 28
    o = f'<rect x="{x}" y="{y}" width="{w:.1f}" height="30" rx="15" fill="{col}"/>\n'
    o += text(x + w / 2, y + 20, lab, 15, txt, "sans", 700, anchor="middle", extra='letter-spacing="1.5"')

def head(x, y, w, lab, col, sub=None, txt=BG):

def sheep(cx, cy, sc=1.0, face=FACE, wool=VAR):
    o = f'<g transform="translate({cx:.1f},{cy:.1f}) scale({sc})">'
    o += f'<rect x="-9" y="6" width="3.5" height="9" fill="{face}"/><rect x="5" y="6" width="3.5" height="9" fill="{face}"/>'
    for dx, dy, r in [(-8, 0, 7), (0, -4, 8), (8, 0, 7), (0, 4, 7), (-4, 2, 6), (5, 3, 6)]:
        o += f'<circle cx="{dx}" cy="{dy}" r="{r}" fill="{wool}"/>'
    o += f'<ellipse cx="14" cy="-3" rx="5" ry="6.5" fill="{face}" transform="rotate(-15 14 -3)"/>'
    o += f'<ellipse cx="11" cy="-7" rx="3" ry="1.6" fill="{face}" transform="rotate(-30 11 -7)"/>'
    return o + "</g>\n"


def arrow_v(x, y0, y1, col=OCRU):
    up = y1 < y0
    o = f'<line x1="{x}" y1="{y0}" x2="{x}" y2="{y1 + (6 if up else -6)}" stroke="{col}" stroke-width="3"/>'
    o += poly([(x - 5, y1 + (7 if up else -7)), (x, y1), (x + 5, y1 + (7 if up else -7))], fill=col)
    return o

# casete-reper (callout)
def callout(v, bx, by, bw, title, body, col=OCRU, attach="bottom"):
    vx, vy = P(*C[v])
    lines = wrap(body, bw - 24, 14) if body else []
    bh = 34 + 18 * len(lines) + (6 if lines else 0)
    if attach == "bottom":
        ax, ay = min(max(vx, bx + 16), bx + bw - 16), by + bh
    elif attach == "top":
        ax, ay = min(max(vx, bx + 16), bx + bw - 16), by
    elif attach == "left":
        ax, ay = bx, by + bh / 2
    else:
        ax, ay = bx + bw, by + bh / 2
    o = f'<line x1="{vx:.1f}" y1="{vy:.1f}" x2="{ax:.1f}" y2="{ay:.1f}" stroke="{col}" stroke-width="1.6" opacity="0.8"/>\n'
    o += f'<rect x="{bx}" y="{by}" width="{bw}" height="{bh}" rx="10" fill="{BG}" stroke="{col}" stroke-width="1.5" opacity="0.96"/>\n'
    o += text(bx + 12, by + 24, title, 16, col, "sans", 700)
    assert measure(title, 16, "sans", 700) <= bw - 20, title
    yy = by + 44
    for ln in lines:
        o += text(bx + 12, yy, ln, 14, CEATA)
        yy += 18
    o += f'<circle cx="{vx:.1f}" cy="{vy:.1f}" r="7" fill="{col}" stroke="{BG}" stroke-width="2.5"/>\n'
    return o, (bx, by, bw, bh)

def overlap(a, b):
    return not (a[0] + a[2] <= b[0] or b[0] + b[2] <= a[0] or a[1] + a[3] <= b[1] or b[1] + b[3] <= a[1])
