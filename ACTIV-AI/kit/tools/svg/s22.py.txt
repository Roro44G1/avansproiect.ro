def medallion(x, y, kind, r=27, num=None):
    s = (f'<g filter="url(#shs)"><circle cx="{x:.1f}" cy="{y:.1f}" r="{r}" fill="#FBF7EE" stroke="{VIOLET}" '
         f'stroke-width="3" stroke-dasharray="7 4"/></g>')
    s += G(x, y, (r - 4) / 29, ICONS[kind](), filt=False)
    if num is not None:
        bx, by = x + r * 0.74, y - r * 0.74
        s += (f'<circle cx="{bx:.1f}" cy="{by:.1f}" r="10" fill="{VIO_D}" stroke="{VAR}" stroke-width="1.8"/>'
              + text(bx, by + 4.6, str(num), 13, VAR, "sans", 700, anchor="middle"))
    return s + "\n"


def fpin(x, y, n, col=VIOLET, r=15):
    """ac de hartă cu număr; vârful în (x, y). Returnează (svg, bbox)."""
    cy = y - 31
    d = (f"M{x:.1f},{y:.1f} C{x-5:.1f},{y-9:.1f} {x-r:.1f},{cy+11:.1f} {x-r:.1f},{cy:.1f} "
         f"A{r},{r} 0 1 1 {x+r:.1f},{cy:.1f} C{x+r:.1f},{cy+11:.1f} {x+5:.1f},{y-9:.1f} {x:.1f},{y:.1f} Z")
    s = f'<ellipse cx="{x:.1f}" cy="{y+1:.1f}" rx="7" ry="2.4" fill="#000" opacity="0.25"/>'
    s += f'<g filter="url(#shs)"><path d="{d}" fill="{col}" stroke="{VAR}" stroke-width="2.5"/></g>'
    s += text(x, cy + 6.5, str(n), 18, VAR, "sans", 700, anchor="middle")
    return s + "\n", (x - r - 2, cy - r - 2, x + r + 2, y + 3)

def pill(x, y, label, fill, tcol, size=11):
    w = measure(label, size, "sans", 700, letter_spacing=1.2) + 14
    s = f'<rect x="{x:.1f}" y="{y-size-3:.1f}" width="{w:.1f}" height="{size+8}" rx="{(size+8)/2:.1f}" fill="{fill}"/>'
    s += text(x + w / 2, y + 0.5, label, size, tcol, "sans", 700, anchor="middle", extra='letter-spacing="1.2"')
    return s, w


def para_after(x, y, first_offset, txt, size, maxw, fill, lh, italic=False, weight=400):
    """paragraf al cărui prim rând începe după o etichetă (first_offset); returnează (svg, nr_rânduri)"""
    words = txt.split()
    lines, cur, first = [], "", True
    for wd in words:
        t = (cur + " " + wd).strip()
        lim = maxw - (first_offset if first else 0)
        if measure(t, size, "sans", weight, italic) <= lim or not cur:
            cur = t
        else:
            lines.append(cur); cur = wd; first = False
    if cur:
        lines.append(cur)
    out = ""
    for i, ln in enumerate(lines):
        out += text(x + (first_offset if i == 0 else 0), y + i * lh, ln, size, fill, "sans", weight, italic)
    return out, len(lines)
