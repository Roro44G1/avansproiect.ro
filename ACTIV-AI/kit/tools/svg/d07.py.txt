def ic_doc(cx, cy, col):
    o = f'<rect x="{cx-30}" y="{cy-38}" width="60" height="72" rx="4" fill="#F3EFE6" stroke="{col}" stroke-width="2"/>\n'
    for i in range(4):
        o += f'<line x1="{cx-18}" y1="{cy-22+i*12}" x2="{cx+18}" y2="{cy-22+i*12}" stroke="#8C7B5E" stroke-width="2"/>\n'
    o += f'<circle cx="{cx+18}" cy="{cy+22}" r="11" fill="{TIGLA}"/><circle cx="{cx+18}" cy="{cy+22}" r="5" fill="none" stroke="#F3EFE6" stroke-width="1.5"/>\n'
    return o

def ic_rings(cx, cy, n, col, hl=None):
    o = ""
    for i in range(n):
        rx = 24 + i * 14; ry = 16 + i * 10
        c = col if (hl is None or i == hl) else "#56657A"
        w = 5 if (hl is None or i == hl) else 3
        o += f'<ellipse cx="{cx}" cy="{cy}" rx="{rx}" ry="{ry}" fill="none" stroke="{c}" stroke-width="{w}"/>\n'
    o += f'<rect x="{cx-5}" y="{cy-9}" width="10" height="12" fill="#F3EFE6"/><polygon points="{cx-7},{cy-9} {cx},{cy-16} {cx+7},{cy-9}" fill="{TIGLA}"/>\n'
    return o

def ic_tower(cx, cy, col, h=64, w=26, roof=TIGLA, oct=False, extra=""):
    o = f'<rect x="{cx-w/2}" y="{cy+34-h}" width="{w}" height="{h}" fill="{col}"/>\n'
    if oct:
        o += f'<rect x="{cx-w/6}" y="{cy+34-h}" width="{w/3}" height="{h}" fill="#000" opacity="0.15"/>\n'
    o += f'<polygon points="{cx-w/2-5},{cy+34-h} {cx},{cy+34-h-26} {cx+w/2+5},{cy+34-h}" fill="{roof}"/>\n'
    o += extra
    return o

def ic_gate(cx, cy, col, roof=TIGLA, tall=True):
    o = f'<rect x="{cx-22}" y="{cy-16}" width="44" height="50" fill="{col}"/>\n'
    o += f'<path d="M{cx-10},{cy+34} v-22 a10,10 0 0 1 20,0 v22 Z" fill="{BG}"/>\n'
    if tall:
        o += f'<polygon points="{cx-26},{cy-16} {cx},{cy-56} {cx+26},{cy-16}" fill="{roof}"/>\n'
    else:
        o += f'<polygon points="{cx-26},{cy-16} {cx},{cy-28} {cx+26},{cy-16}" fill="{roof}"/>\n'
    return o

def ic_bastion(cx, cy, col):
    pts = [(cx, cy - 34), (cx + 40, cy - 8), (cx + 26, cy + 34), (cx - 26, cy + 34), (cx - 40, cy - 8)]
    o = poly(pts, fill=col, stroke="#F3EFE6", stroke_width=1.5)
    o += poly([(cx, cy - 18), (cx + 24, cy - 2), (cx + 15, cy + 22), (cx - 15, cy + 22), (cx - 24, cy - 2)], fill="#000", opacity=0.18)
    return o

def ic_demol(cx, cy, col):
    o = ""
    for r, (x0, n) in enumerate([(-44, 4), (-32, 3), (-44, 3)]):
        for i in range(n):
            if (r, i) in [(0, 3), (1, 2)]:
                continue
            o += f'<rect x="{cx+x0+i*24}" y="{cy+16-r*18}" width="22" height="16" fill="{col}" opacity="{0.9 if r<2 else 0.6}"/>\n'
    o += f'<line x1="{cx+10}" y1="{cy+30}" x2="{cx+44}" y2="{cy-30}" stroke="#F3EFE6" stroke-width="5" stroke-linecap="round"/>\n'
    o += f'<path d="M{cx+30},{cy-38} q18,6 24,20" fill="none" stroke="{TIGLA}" stroke-width="8" stroke-linecap="round"/>\n'
    return o

def ic_ghostwall(cx, cy, col):
    o = ""
    for i in range(4):
        o += f'<rect x="{cx-46+i*24}" y="{cy}" width="22" height="16" fill="none" stroke="{col}" stroke-width="2" stroke-dasharray="4 4"/>\n'

        o += f'<rect x="{cx-34+i*24}" y="{cy-18}" width="22" height="16" fill="none" stroke="{col}" stroke-width="2" stroke-dasharray="4 4"/>\n'
    o += f'<rect x="{cx-46}" y="{cy+16}" width="94" height="14" fill="{col}" opacity="0.5"/>\n'
    return o

def ic_roof(cx, cy, col):
    o = f'<rect x="{cx-22}" y="{cy-6}" width="44" height="40" fill="{col}"/>\n'
    o += f'<path d="M{cx-10},{cy+34} v-18 a10,10 0 0 1 20,0 v18 Z" fill="{BG}"/>\n'

    return o

def ic_scaffold(cx, cy, col):
    o = ic_tower(cx, cy, col, h=64, w=26, oct=True)
    for i in range(4):
        y = cy + 30 - i * 20
        o += f'<line x1="{cx-40}" y1="{y}" x2="{cx+40}" y2="{y}" stroke="{PETROL}" stroke-width="2"/>\n'
    for x in (cx - 40, cx + 40):
        o += f'<line x1="{x}" y1="{cy+34}" x2="{x}" y2="{cy-36}" stroke="{PETROL}" stroke-width="3"/>\n'
    return o

def ic_towers3(cx, cy, col):
    o = ic_tower(cx - 34, cy, col, h=40, w=22, oct=True)
    o += ic_tower(cx, cy, col, h=46, w=22)
    o += ic_tower(cx + 34, cy, col, h=42, w=22, oct=True)
    return o

    o += text(cx, cy + 6, "1668", 13, "#3B3326", "serif", 700, anchor="middle")
    return o
