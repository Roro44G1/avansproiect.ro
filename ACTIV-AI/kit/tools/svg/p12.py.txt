def arc_chars(cx, cy, r, txt, size, top=True, fill=INK, ls=2.5, weight=700):
    widths = [measure(ch, size, "sans", weight) for ch in txt]
    total = sum(widths) + ls * (len(txt) - 1)
    o = ""; pos = -total / 2
    for ch, w in zip(txt, widths):
        deg = math.degrees((pos + w / 2) / r)
        if top:
            o += (f'<g transform="translate({cx},{cy}) rotate({deg:.2f})"><text x="0" y="{-r:.1f}" font-family="{SANS}" '
                  f'font-weight="{weight}" font-size="{size}" fill="{fill}" text-anchor="middle">{esc(ch)}</text></g>\n')
        else:
            o += (f'<g transform="translate({cx},{cy}) rotate({-deg:.2f})"><text x="0" y="{r:.1f}" font-family="{SANS}" '
                  f'font-weight="{weight}" font-size="{size}" fill="{fill}" text-anchor="middle">{esc(ch)}</text></g>\n')
        pos += w + ls
    return o

         (1662, RED, "orașul se extinde spre vest; anul exact nu e dat de surse")]
NWD = 236
manual = {1134: ["landlerii: protestanți strămutați", "din Austria de azi, din ordinul",
                  "împăraților Carol al VI-lea", "și Maria Tereza"]}
for x, c, t in notes:
    if x in manual:
        for i, ln in enumerate(manual[x]):
            assert measure(ln, 16, "serif", 400, True) <= 262, ln
            s += text(x, 696 + i * 22, ln, 16, INK, "serif", 400, True, anchor="middle")
    else:
        p, _ = para(x, 696, t, 16, NWD, INK, "serif", 400, True, lh=22, anchor="middle")
        s += p

# ---- „viza ta” (de completat)
vx, vy, vr = 1086, 900, 72

s += text(vx, vy - 18, "VIZA TA", 16, PLUM, "sans", 700, anchor="middle", extra='letter-spacing="2"')

# ---- caseta de onestitate

            15, HW - 40, INK, "sans", 400, lh=21)
s += p
s += washi(HX + HW - 70, HY - 14, 84, 22, 8, VIOLET, 0.4)
s += text(PX1 - 30, PY1 - 14, "2", 14, INK2, "serif", 400, True, anchor="end")

# semn de carte (panglică) pe cotor
