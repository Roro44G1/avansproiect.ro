def icon_svg(typ):
    fill, fg, det = TYPES[typ]
    if typ == "muzeu":
        return ICON[typ](fg, fill)
    return ICON[typ](fg, det)


def star_pts(cx, cy, R, r, n=5):
    pts = []
    for i in range(2 * n):
        a = -math.pi / 2 + i * math.pi / n
        rr = R if i % 2 == 0 else r
        pts.append((cx + rr * math.cos(a), cy + rr * math.sin(a)))
    return pts


def star(cx, cy, R, fill=OCRU, stroke=DARK, sw=1.2):
    return poly(star_pts(cx, cy, R, R * 0.45), fill=fill, stroke=stroke, stroke_width=sw, stroke_linejoin="round")


def marker(x, y, typ, r=17, second=None, ring=True):
    """disc colorat cu pictograma tipului; contur punctat = poziție orientativă; insignă mică = a doua funcție"""
    fill = TYPES[typ][0]
    s = ""
    if ring:
        s += (f'<circle cx="{x:.1f}" cy="{y:.1f}" r="{r+5}" fill="#FBF7EE" fill-opacity="0.55" stroke="{fill}" '
              f'stroke-width="2.2" stroke-dasharray="4 3"/>')
    s += f'<g filter="url(#shs)"><circle cx="{x:.1f}" cy="{y:.1f}" r="{r}" fill="{fill}" stroke="{VAR}" stroke-width="2.5"/></g>'
    s += G(x, y + 1, (r - 4) / 28, icon_svg(typ), filt=False)
    if second:
        bx, by, r2 = x + r * 0.9, y - r * 0.9, max(9, r * 0.55)
        f2 = TYPES[second][0]
        s += f'<circle cx="{bx:.1f}" cy="{by:.1f}" r="{r2:.1f}" fill="{f2}" stroke="{VAR}" stroke-width="2"/>'
        s += G(bx, by + 0.5, (r2 - 2.5) / 28, icon_svg(second), filt=False)
    return s + "\n"


def marker_box(x, y, r=17, second=None, ring=True):
    e = r + (6 if ring else 1)
    x0, y0, x1, y1 = x - e, y - e, x + e, y + e
    if second:
        r2 = max(9, r * 0.55)
        x1 = max(x1, x + r * 0.9 + r2 + 1); y0 = min(y0, y - r * 0.9 - r2 - 1)
    return (x0, y0, x1, y1)


def leader(B, name, mx, my, rr, box, col=INK2):
    """linie de legătură de la marginea pictogramei la cel mai apropiat punct al etichetei"""
    bx0, by0, bx1, by1 = box
    nx = min(max(mx, bx0), bx1); ny = min(max(my, by0), by1)
    d = math.hypot(nx - mx, ny - my)
    if d <= rr + 12:
        return ""
    ux, uy = (nx - mx) / d, (ny - my) / d
    sx, sy = mx + ux * (rr + 1), my + uy * (rr + 1)
    ex, ey = nx - ux * 4, ny - uy * 4
    pts = [(sx + (ex - sx) * t / 20, sy + (ey - sy) * t / 20) for t in range(21)]
    add_curve(B, name, pts[3:], step=6, r=2.5)
    return (f'<path d="M{sx:.1f},{sy:.1f} L{ex:.1f},{ey:.1f}" stroke="#FBF7EE" stroke-width="4" stroke-linecap="round"/>'
            f'<path d="M{sx:.1f},{sy:.1f} L{ex:.1f},{ey:.1f}" stroke="{col}" stroke-width="1.6" stroke-linecap="round"/>'
            f'<circle cx="{ex:.1f}" cy="{ey:.1f}" r="2.4" fill="{col}"/>\n')


def blob(pts, fill=PARK, op=0.75, stroke=PARK_D):
    d = smooth_path(pts, closed=True, tension=0.8)
    return (f'<path d="{d}" fill="{fill}" fill-opacity="{op}" stroke="{stroke}" stroke-width="1.4" '
            f'stroke-dasharray="5 4"/>\n')
