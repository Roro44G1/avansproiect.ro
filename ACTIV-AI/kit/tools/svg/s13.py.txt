def mk_proj(cx, cy, sx, sy):
    def P(nx, ny):
        return (cx + nx * sx, cy + ny * sy)
    return P


# ---------------- text ----------------
def halo(x, y, s, size, fill=INK, weight=700, italic=False, anchor="start", hw=6, hcol=PARCH, fam="sans"):
    return text(x, y, s, size, fill, fam, weight, italic, anchor,
                extra=f'stroke="{hcol}" stroke-width="{hw}" stroke-linejoin="round" paint-order="stroke"')


def rich(x, y, parts, size, anchor="start", hcol=None, hw=5, fam="sans"):
    """parts: list of (text, fill, weight, italic[, size])"""
    ff = SANS if fam == "sans" else SERIF
    h = (f' stroke="{hcol}" stroke-width="{hw}" stroke-linejoin="round" paint-order="stroke"' if hcol else "")
    out = f'<text x="{x:.1f}" y="{y:.1f}" font-family="{ff}" font-size="{size}" text-anchor="{anchor}" xml:space="preserve"{h}>'
    for p in parts:
        t, fill, w, it = p[:4]
        sz = p[4] if len(p) > 4 else size
        pf = p[5] if len(p) > 5 else fam
        pff = SANS if pf == "sans" else SERIF
        out += (f'<tspan fill="{fill}" font-weight="{w}" font-size="{sz}" font-family="{pff}"'
                f'{" font-style=" + chr(34) + "italic" + chr(34) if it else ""}>{esc(t)}</tspan>')
    return out + "</text>\n"


def rich_width(parts, size, fam="sans"):
    w = 0
    for p in parts:
        t, fill, wt, it = p[:4]
        sz = p[4] if len(p) > 4 else size
        pf = p[5] if len(p) > 5 else fam
        w += measure(t, sz, pf, wt, it)
    return w

class Boxes:
    def __init__(self):
        self.items = []

    def add(self, name, x0, y0, x1, y1):
        self.items.append((name, min(x0, x1), min(y0, y1), max(x0, x1), max(y0, y1)))

    def text(self, name, x, y, w, size, anchor="start"):
        x0 = x if anchor == "start" else (x - w / 2 if anchor == "middle" else x - w)
        self.add(name, x0, y - size * 0.78, x0 + w, y + size * 0.24)

    def report(self, bounds=None, ignore=()):
        bad = []
        it = self.items
        for i in range(len(it)):
            for j in range(i + 1, len(it)):
                a, b = it[i], it[j]
                if a[0].split(":")[0] == b[0].split(":")[0]:
                    continue
                if (a[0], b[0]) in ignore or (b[0], a[0]) in ignore:
                    continue
                if a[1] < b[3] and b[1] < a[3] and a[2] < b[4] and b[2] < a[4]:
                    bad.append((a[0], b[0]))
        if bounds:
            X0, Y0, X1, Y1 = bounds
            for a in it:
                if a[1] < X0 or a[2] < Y0 or a[3] > X1 or a[4] > Y1:
                    bad.append((a[0], "OUT"))
        for b in bad:
            print("  overlap:", b)
        return bad
