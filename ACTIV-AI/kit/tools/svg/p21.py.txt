def total_duration():
    t = sum(int(d.split(":")[0]) * 60 + int(d.split(":")[1]) for _, _, d, _, _ in TRACKS)
    return f"{t//60}:{t%60:02d}"


def note(x, y, sz, fill, double=False):
    """Notă muzicală simplă (cap eliptic + coadă + steguleț)."""
    o = f'<ellipse cx="{x:.1f}" cy="{y:.1f}" rx="{sz*0.36:.1f}" ry="{sz*0.26:.1f}" transform="rotate(-20 {x:.1f} {y:.1f})" fill="{fill}"/>'
    sx = x + sz * 0.31
    o += f'<line x1="{sx:.1f}" y1="{y-sz*0.05:.1f}" x2="{sx:.1f}" y2="{y-sz*1.1:.1f}" stroke="{fill}" stroke-width="{sz*0.09:.1f}" stroke-linecap="round"/>'
    if double:
        x2 = x + sz * 0.95
        o += f'<ellipse cx="{x2:.1f}" cy="{y-sz*0.18:.1f}" rx="{sz*0.36:.1f}" ry="{sz*0.26:.1f}" transform="rotate(-20 {x2:.1f} {y-sz*0.18:.1f})" fill="{fill}"/>'
        o += f'<line x1="{x2+sz*0.31:.1f}" y1="{y-sz*0.23:.1f}" x2="{x2+sz*0.31:.1f}" y2="{y-sz*1.28:.1f}" stroke="{fill}" stroke-width="{sz*0.09:.1f}" stroke-linecap="round"/>'
        o += f'<polygon points="{P([(sx-1, y-sz*1.1), (x2+sz*0.31+1, y-sz*1.28), (x2+sz*0.31+1, y-sz*1.06), (sx-1, y-sz*0.88)])}" fill="{fill}"/>'
    else:
        o += f'<path d="M{sx:.1f},{y-sz*1.1:.1f} q{sz*0.32:.1f},{sz*0.18:.1f} {sz*0.30:.1f},{sz*0.55:.1f} q{-sz*0.04:.1f},{-sz*0.24:.1f} {-sz*0.30:.1f},{-sz*0.32:.1f} Z" fill="{fill}"/>'
    return o


def cover(X, Y, S):
    """Coperta albumului (desen original), pătrat S × S."""
    o = ('<defs>'
         '<linearGradient id="dusk21" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="#2A1F5C"/>'
         '<stop offset="0.45" stop-color="#6E56CF"/><stop offset="0.78" stop-color="#C8708A"/><stop offset="1" stop-color="#E2A62B"/></linearGradient>'
         '<linearGradient id="roof21" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="#C9663F"/><stop offset="1" stop-color="#8E3F2A"/></linearGradient>'
         '<linearGradient id="band21" x1="0" y1="0" x2="1" y2="0"><stop offset="0" stop-color="#2B3440"/><stop offset="0.5" stop-color="#3B4654"/><stop offset="1" stop-color="#2B3440"/></linearGradient>'
         '<radialGradient id="cup21" cx="0.4" cy="0.35" r="0.7"><stop offset="0" stop-color="#4A5666"/><stop offset="1" stop-color="#1B2733"/></radialGradient>'
         '<radialGradient id="vinyl21" cx="0.5" cy="0.5" r="0.5"><stop offset="0" stop-color="#20262E"/><stop offset="0.6" stop-color="#11161B"/><stop offset="1" stop-color="#0A0D10"/></radialGradient>'
         f'<clipPath id="cv21"><rect x="{X}" y="{Y}" width="{S}" height="{S}" rx="6"/></clipPath>'
         '</defs>\n')
    cx = X + S / 2
    # ---- discul de vinil, ieșit din copertă
    R = 190
    dcx, dcy = X + S - 32, Y + S / 2
    o += f'<g filter="url(#sh)"><circle cx="{dcx}" cy="{dcy}" r="{R:.1f}" fill="url(#vinyl21)"/></g>'
    for k in range(14):
        rr = R * (0.40 + 0.042 * k)
        o += f'<circle cx="{dcx}" cy="{dcy}" r="{rr:.1f}" fill="none" stroke="#3A444F" stroke-width="0.8" opacity="{0.35 + 0.03*k:.2f}"/>'
    o += f'<path d="M{dcx+R*0.35:.1f},{dcy-R*0.75:.1f} A{R*0.83:.1f},{R*0.83:.1f} 0 0 1 {dcx+R*0.80:.1f},{dcy-R*0.18:.1f}" fill="none" stroke="#FFFFFF" stroke-width="3" opacity="0.10"/>'
    o += f'<circle cx="{dcx}" cy="{dcy}" r="{R*0.32:.1f}" fill="#E2A62B"/>'
    o += text(X + S + 15, dcy + 9, "A", 24, "#2A1F5C", "serif", 700, anchor="middle")
    # ---- coperta
    o += f'<g filter="url(#sh)"><rect x="{X}" y="{Y}" width="{S}" height="{S}" rx="6" fill="#2A1F5C"/></g>\n'
    o += f'<g clip-path="url(#cv21)">'
    o += f'<rect x="{X}" y="{Y}" width="{S}" height="{S}" fill="url(#dusk21)"/>'
    rnd = random.Random(21)
    for _ in range(34):
        sx = rnd.uniform(X + 10, X + S - 10); sy = rnd.uniform(Y + 130, Y + S * 0.42)
        o += f'<circle cx="{sx:.1f}" cy="{sy:.1f}" r="{rnd.uniform(0.8, 2.0):.1f}" fill="#F3EFE6" opacity="{rnd.uniform(0.3, 0.8):.2f}"/>'
    # orizontul: siluete de acoperișuri, cu „ochi” aprinși
    HZ = Y + S * 0.70
    SIL = "#3B2C78"; SL = "#2A1F5C"
    o += council_tower(X + 92, HZ, 30, 112, SIL, SIL, SL, dark="#F2C46B", sw=1.2, detail=False)
    for hx, hw, wh, rh, gb in ((X - 10, 70, 40, 64, False), (X + 120, 64, 36, 70, True), (X + S - 186, 66, 40, 58, True),
                               (X + S - 128, 76, 44, 78, False), (X + S - 50, 70, 38, 60, False)):
        o += house(hx, HZ, hw, wh, rh, SIL, SIL, SL, eyes=1, eye_open="#F2C46B", win="#F2C46B", sw=1.2, tiles=False, gable=gb, wins=1)

    o += f'<rect x="{tx-13}" y="{HZ-120}" width="26" height="120" fill="{SIL}" stroke="{SL}" stroke-width="1.2"/>'

        yy = top + (Y + S + 4 - top) * t

        n = int((xr - xl) / 26)
        dx = (xr - xl) / n
        d = f"M{xl:.1f},{yy:.1f}"
        for j in range(n):
            d += f" q{dx/2:.1f},{9:.1f} {dx:.1f},0"
        o += f'<path d="{d}" fill="none" stroke="#6E2A1A" stroke-width="1.6" opacity="0.55"/>'
    # ochii: lucarne cu albul ochiului și pupile care privesc în sus

        o += f'<circle cx="{ex+14:.1f}" cy="{ey-19:.1f}" r="3.6" fill="#FFFFFF"/>'

          f'stroke="#55627A" stroke-width="5" stroke-linecap="round" opacity="0.8"/>')
    for sg, xx in ((-1, L), (1, Rr)):
        o += f'<rect x="{xx-26:.1f}" y="{ycup-56:.1f}" width="52" height="96" rx="24" fill="url(#cup21)" stroke="#0F1720" stroke-width="2.5"/>'
        o += f'<rect x="{xx-sg*20-9:.1f}" y="{ycup-46:.1f}" width="18" height="76" rx="9" fill="#E2A62B" opacity="0.9"/>'
        # unde sonore
        for k in range(3):
            rr = 34 + k * 16
            a0, a1 = (150, 210) if sg < 0 else (-30, 30)
            p1 = (xx + rr * math.cos(math.radians(a0)), ycup - 8 + rr * math.sin(math.radians(a0)))
            p2 = (xx + rr * math.cos(math.radians(a1)), ycup - 8 + rr * math.sin(math.radians(a1)))
            sweep = 1 if sg > 0 else 0
            o += (f'<path d="M{p1[0]:.1f},{p1[1]:.1f} A{rr},{rr} 0 0 {sweep} {p2[0]:.1f},{p2[1]:.1f}" fill="none" stroke="#F3EFE6" '
                  f'stroke-width="3" stroke-linecap="round" opacity="{0.75 - 0.2*k:.2f}"/>')
    # note muzicale
    o += note(X + 70, Y + S * 0.36, 30, "#F3EFE6", True)
    o += note(X + S - 96, Y + S * 0.33, 34, "#E2A62B")
    o += note(X + S - 52, Y + S * 0.47, 22, "#F3EFE6")
    # ---- titlul albumului
    ts = fit("HARTĂ PENTRU CĂȘTI", S - 90, 44, "serif", 700)
    o += text(cx, Y + 72, "HARTĂ PENTRU CĂȘTI", ts, "#F3EFE6", "serif", 700, anchor="middle")
    o += text(cx, Y + 104, "playlistul Dariei M. · zece locuri din Sibiu", 17, "#E9E1FF", "sans", 500, True, anchor="middle")
    # ---- banda obligatorie

    o += f'<rect x="{X}" y="{BY}" width="{S}" height="40" fill="#E2A62B"/>'
    o += text(cx, BY + 27, "PIESE IMAGINARE – NU EXISTĂ ÎNREGISTRĂRI", 16.5, "#0F1720", "sans", 800, anchor="middle", extra='letter-spacing="1.6"')
    o += '</g>\n'
    o += f'<rect x="{X}" y="{Y}" width="{S}" height="{S}" rx="6" fill="none" stroke="#000" stroke-opacity="0.35" stroke-width="1.5"/>\n'
    return o

            f'<circle cx="{x}" cy="{y-1.5}" r="2.4" fill="{BG}"/>')
