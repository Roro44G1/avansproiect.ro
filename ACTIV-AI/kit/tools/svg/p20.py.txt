def open_book(cx, top, w, h, C, sw):
    """Carte deschisă văzută din față (pagini curbate), cu rânduri de text schițate."""
    o = ""
    # coperta
    o += (f'<path d="M{cx-w/2-5:.1f},{top+8:.1f} L{cx-w/2-5:.1f},{top+h+6:.1f} Q{cx-w/4:.1f},{top+h-2:.1f} {cx:.1f},{top+h+8:.1f} '
          f'Q{cx+w/4:.1f},{top+h-2:.1f} {cx+w/2+5:.1f},{top+h+6:.1f} L{cx+w/2+5:.1f},{top+8:.1f} Z" fill="{C["cover"]}" stroke="{C["ink"]}" stroke-width="{sw:.1f}" stroke-linejoin="round"/>')
    for sg in (-1, 1):
        d = (f"M{cx:.1f},{top+6:.1f} Q{cx+sg*w*0.22:.1f},{top-6:.1f} {cx+sg*w/2:.1f},{top+2:.1f} L{cx+sg*w/2:.1f},{top+h:.1f} "
             f"Q{cx+sg*w*0.22:.1f},{top+h-8:.1f} {cx:.1f},{top+h+4:.1f} Z")
        o += f'<path d="{d}" fill="{C["page"]}" stroke="{C["ink"]}" stroke-width="{sw:.1f}" stroke-linejoin="round"/>'
        # rânduri
        for j in range(1, 4):
            t = j / 4.2
            y1 = top + 6 + (h - 4) * t
            o += (f'<path d="M{cx+sg*w*0.08:.1f},{y1+1:.1f} Q{cx+sg*w*0.25:.1f},{y1-7:.1f} {cx+sg*w*0.42:.1f},{y1-3:.1f}" fill="none" '
                  f'stroke="{C["ink"]}" stroke-width="{sw*0.7:.1f}" stroke-linecap="round" opacity="0.8"/>')
    o += f'<line x1="{cx:.1f}" y1="{top+6:.1f}" x2="{cx:.1f}" y2="{top+h+4:.1f}" stroke="{C["ink"]}" stroke-width="{sw:.1f}"/>'
    return o


def diamond(x, y, r, fill):
    return f'<polygon points="{P([(x, y-r), (x+r, y), (x, y+r), (x-r, y)])}" fill="{fill}"/>'


def exlibris(uid, C, mono=False):
    """Ex-librisul în coordonate locale 300 × 400 (origine sus-stânga)."""
    sw = 1.9 if mono else 1.5
    INK = C["ink"]
    o = ""
    # hârtia și chenarul dublu
    o += f'<rect x="0" y="0" width="300" height="400" fill="{C["paper"]}"/>'
    o += f'<rect x="8" y="8" width="284" height="384" fill="none" stroke="{C["frame"]}" stroke-width="{4 if mono else 3.2}"/>'
    o += f'<rect x="14" y="14" width="272" height="372" fill="none" stroke="{C["frame"]}" stroke-width="{1.4 if mono else 1.1}"/>'
    # colțuri: mici „ochi”
    for (x_, y_) in ((26, 27), (274, 27)):
        o += eye_dormer(x_, y_ + 4, 18, 6, C["roof"], INK, C["eye"], 1.0)
    # EX LIBRIS
    o += (f'<text x="152.5" y="55" font-family="{SERIF}" font-weight="700" font-size="27" letter-spacing="5" '
          f'fill="{C["ex"]}" text-anchor="middle">EX LIBRIS</text>')
    o += diamond(47, 46, 4, C["ex"]) + diamond(253, 46, 4, C["ex"])
    # fereastra în arc
    AX0, AX1, AT, AB = 30, 270, 72, 292
    R = (AX1 - AX0) / 2
    arch = f"M{AX0},{AB} V{AT+R} A{R},{R} 0 0 1 {AX1},{AT+R} V{AB} Z"
    o += f'<clipPath id="win{uid}"><path d="{arch}"/></clipPath>'
    o += f'<g clip-path="url(#win{uid})">'
    o += f'<rect x="{AX0}" y="{AT}" width="{AX1-AX0}" height="{AB-AT}" fill="{C["sky"]}"/>'
    # soarele (sau razele, la varianta alb-negru)
    scx, scy = 150, 150
    for kk in range(16):
        a = 2 * math.pi * kk / 16
        r1, r2 = 62, (112 if kk % 2 == 0 else 92)
        o += (f'<line x1="{scx+r1*math.cos(a):.1f}" y1="{scy+r1*math.sin(a):.1f}" x2="{scx+r2*math.cos(a):.1f}" y2="{scy+r2*math.sin(a):.1f}" '
              f'stroke="{INK if mono else C["sun"]}" stroke-width="{1.4 if mono else 2.2}" stroke-linecap="round" opacity="{1 if mono else 0.75}"/>')
    if C["sun"]:
        o += f'<circle cx="{scx}" cy="{scy}" r="54" fill="{C["sun"]}" opacity="0.9"/>'
    else:
        o += f'<circle cx="{scx}" cy="{scy}" r="54" fill="none" stroke="{INK}" stroke-width="1.6"/>'
    # casele, cu „ochi” mari (desenați separat, ca să se vadă și la 6 cm)
    E_ = dict(lid=C["roof"], line=INK, opening=C["eye"], sw=sw)
    o += house(AX0 - 6, 244, 60, 42, 70, C["wall"], C["roof"], INK, eyes=0, win=C["win"], sw=sw, tiles=not mono)
    o += house(AX1 - 54, 244, 60, 42, 74, C["wall"], C["roof"], INK, eyes=0, win=C["win"], sw=sw, tiles=not mono)
    o += house(AX0 + 56, 246, 46, 30, 50, C["wall"], C["roof"], INK, eyes=0, win=C["win"], sw=sw, gable=True, tiles=not mono, wins=1)
    o += house(AX1 - 102, 246, 46, 30, 52, C["wall"], C["roof"], INK, eyes=0, win=C["win"], sw=sw, gable=True, tiles=not mono, wins=1)
    for hx, top_, rh in ((AX0 - 6, 202, 70), (AX1 - 54, 202, 74)):
        o += eye_dormer(hx + 17, top_ - rh * 0.30, 24, 8.5, **E_)
        o += eye_dormer(hx + 43, top_ - rh * 0.30, 24, 8.5, **E_)
        o += eye_dormer(hx + 30, top_ - rh * 0.68, 18, 6.5, **E_)
    for hx, top_, rh in ((AX0 + 56, 216, 50), (AX1 - 102, 216, 52)):
        o += eye_dormer(hx + 23, top_ - rh * 0.34, 20, 7, **E_)
    # Turnul Sfatului
    o += council_tower(150, 246, 36, 96, C["stone"], C["troof"], INK, dark=C["tdark"], sw=sw)
    # Podul Minciunilor, peste cotorul cărții
    o += lies_bridge(72, 228, 228, 22, C["iron"], INK, 12, sw=sw, stone=C["abut"], stone_h=26, detail=True)
    # cartea deschisă, în prim-plan: orașul „crește” din ea
    o += open_book(150, 250, 226, 34, C, sw)
    o += '</g>'
    o += f'<path d="{arch}" fill="none" stroke="{INK}" stroke-width="{sw*1.6:.1f}"/>'
    # panglica cu deviza
    ms = fit(MOTTO, 200, 19, "serif", 700, True)
    mw = measure(MOTTO, ms, "serif", 700, True) + 36
    bx0, bx1 = 150 - mw / 2, 150 + mw / 2
    MY = 280
    E = 22
    for sg, xb in ((-1, bx0), (1, bx1)):
        o += f'<polygon points="{P([(xb - sg*E, MY+6), (xb + sg*2, MY+6), (xb + sg*2, MY+34), (xb - sg*E, MY+34), (xb - sg*E + sg*10, MY+20)])}" fill="{C["ribbon_d"] if not mono else C["ribbon"]}" stroke="{INK}" stroke-width="{sw:.1f}" stroke-linejoin="round"/>'
    o += f'<rect x="{bx0+6:.1f}" y="{MY}" width="{mw-12:.1f}" height="30" fill="{C["ribbon"]}" stroke="{INK}" stroke-width="{sw:.1f}"/>'
    o += (f'<text x="150" y="{MY+21}" font-family="{SERIF}" font-weight="700" font-style="italic" font-size="{ms}" '
          f'fill="{C["motto"]}" text-anchor="middle">{esc(MOTTO)}</text>')
    # numele
    ns = fit(OWNER, 230, 38, "serif", 700)
    o += (f'<text x="150" y="358" font-family="{SERIF}" font-weight="700" font-size="{ns}" letter-spacing="3" '
          f'fill="{C["name"]}" text-anchor="middle">{esc(OWNER)}</text>')
    o += f'<line x1="70" y1="372" x2="130" y2="372" stroke="{C["frame"]}" stroke-width="1.2"/><line x1="170" y1="372" x2="230" y2="372" stroke="{C["frame"]}" stroke-width="1.2"/>'
    o += diamond(150, 372, 3.5, C["frame"])
    return o
