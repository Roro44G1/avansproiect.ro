assert len(EV) == 30
by_year = {}
for e in EV:
    by_year.setdefault(e[0], []).append(e)

ROMAN = {13: "XIII", 14: "XIV", 15: "XV", 16: "XVI", 17: "XVII", 18: "XVIII", 19: "XIX", 20: "XX", 21: "XXI"}

def century(y):
    return (int(y) - 1) // 100 + 1

# rândurile (fiecare citit de la stânga la dreapta)
ROWS = [[e for e in EV if int(e[0]) < 1800],
        [e for e in EV if 1800 <= int(e[0]) < 1900],
        [e for e in EV if int(e[0]) >= 1900]]
ROW_Y = [330, 572, 814]
X0, X1 = 100, 1820
MARK_W = 0.9   # lățimea (în sloturi) a marcajului de secol
GAP_W = 1.0    # slot pentru un secol fără premiere în listă

s = frame_head("DATE · D08", BLUE, "30 de premiere sibiene pe o linie a timpului",
               "linie a timpului neliniară · 1292 → 2019 · segmente pe secole, intervalele nu sunt la scară")
s += ('<defs><marker id="arr" viewBox="0 0 10 10" refX="6" refY="5" markerWidth="4" markerHeight="4" orient="auto">'
      '<path d="M0,0 L10,5 L0,10 Z" fill="#56657A"/></marker></defs>\n')

def node(x, y, col):
    return (f'<circle cx="{x:.1f}" cy="{y}" r="15" fill="{BG}" stroke="{col}" stroke-width="4" filter="url(#glow)"/>'
            f'<circle cx="{x:.1f}" cy="{y}" r="6" fill="{col}"/>\n')

def cpill(x, y, label, muted=False):
    w = measure(label, 13, "sans", 700, letter_spacing=1) + 22
    col = "#56657A" if muted else CEATA
    o = f'<rect x="{x-w/2:.1f}" y="{y-13}" width="{w:.1f}" height="26" rx="13" fill="{CARD}" stroke="{col}" stroke-width="1.5"/>\n'
    o += text(x, y + 5, label, 13, col, "sans", 700, anchor="middle", extra='letter-spacing="1"')
    return o

LH = 18
for r, (row, Y) in enumerate(zip(ROWS, ROW_Y)):
    # construim secvența de elemente: marcaj secol / eveniment / pauză
    items = []
    cents = sorted({century(e[0]) for e in row})
    first_c, last_c = cents[0], cents[-1]
    for c in range(first_c, last_c + 1):
        evs = [e for e in row if century(e[0]) == c]
        if evs:
            items.append(("mark", c, MARK_W))
            for e in evs:
                items.append(("ev", e, 1.0))
        else:
            items.append(("gap", c, GAP_W))
    total = sum(it[2] for it in items)
    unit = (X1 - X0) / total
    # linia
    s += f'<line x1="{X0-20}" y1="{Y}" x2="{X1+10}" y2="{Y}" stroke="#2A3845" stroke-width="8" stroke-linecap="round"/>\n'
    s += f'<line x1="{X1+10}" y1="{Y}" x2="{X1+22}" y2="{Y}" stroke="#56657A" stroke-width="8" stroke-linecap="round" marker-end="url(#arr)"/>\n'
    pos = X0
    ev_i = 0
    slot = unit
    for kind, val, w in items:
        cx = pos + w * unit / 2
        if kind == "mark":
            s += cpill(cx, Y, "sec. " + ROMAN[val])
        elif kind == "gap":
            # pauză: secol fără premieră în lista oficială
            s += f'<line x1="{pos+12:.1f}" y1="{Y}" x2="{pos+w*unit-12:.1f}" y2="{Y}" stroke="{BG}" stroke-width="10"/>\n'
            s += f'<line x1="{pos+12:.1f}" y1="{Y}" x2="{pos+w*unit-12:.1f}" y2="{Y}" stroke="#56657A" stroke-width="3" stroke-dasharray="6 7"/>\n'
            s += cpill(cx, Y - 44, "sec. " + ROMAN[val], muted=True)
            s += text(cx, Y + 36, "nicio premieră", 13, "#56657A", "sans", 400, True, anchor="middle")
            s += text(cx, Y + 54, "în lista oficială", 13, "#56657A", "sans", 400, True, anchor="middle")
        else:
            yr, label, d = val
            col = DOM[d][1]
            s += node(cx, Y, col)
            above = (ev_i % 2 == 0)
            maxw = min(2 * slot - 14, 2 * (1840 - cx), 2 * (cx - 80))
            lines = wrap(label, maxw, 15)
            if len(lines) > 3:
                lines = wrap(label, maxw, 14)
            fs = 15 if len(wrap(label, maxw, 15)) <= 3 else 14
            assert len(lines) <= 4, (yr, lines)
            if above:
                ly = Y - 34 - LH * (len(lines) - 1)
                s += text(cx, ly - 28, yr, 26, col, "serif", 700, anchor="middle")
                for ln in lines:
                    s += text(cx, ly, ln, fs, CEATA, anchor="middle"); ly += LH
                # tijă
                s += f'<line x1="{cx:.1f}" y1="{Y-19}" x2="{cx:.1f}" y2="{Y-26}" stroke="{col}" stroke-width="2"/>\n'
            else:
                s += text(cx, Y + 54, yr, 26, col, "serif", 700, anchor="middle")
                ly = Y + 78
                for ln in lines:
                    s += text(cx, ly, ln, fs, CEATA, anchor="middle"); ly += LH
                s += f'<line x1="{cx:.1f}" y1="{Y+19}" x2="{cx:.1f}" y2="{Y+26}" stroke="{col}" stroke-width="2"/>\n'
            ev_i += 1
        pos += w * unit
