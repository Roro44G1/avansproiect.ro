def halo(x, y, s, size, fill=INK, weight=700, italic=False, anchor="start", hw=6, hcol=PARCH):
    return text(x, y, s, size, fill, "sans", weight, italic, anchor,
                extra=f'stroke="{hcol}" stroke-width="{hw}" stroke-linejoin="round" paint-order="stroke"')

def tree(x, y, r=11, c=GREEN1):
    return (f'<rect x="{x-2:.1f}" y="{y:.1f}" width="4" height="{r*0.9:.1f}" fill="#6B4A2E"/>'
            f'<circle cx="{x:.1f}" cy="{y-r*0.5:.1f}" r="{r:.1f}" fill="{c}"/>'
            f'<circle cx="{x-r*0.55:.1f}" cy="{y-r*0.15:.1f}" r="{r*0.7:.1f}" fill="{c}"/>'
            f'<circle cx="{x+r*0.55:.1f}" cy="{y-r*0.15:.1f}" r="{r*0.7:.1f}" fill="{c}"/>\n')

def fir(x, y, h=30, c=GREEN2):
    w = h * 0.55
    return (f'<rect x="{x-2:.1f}" y="{y-4:.1f}" width="4" height="8" fill="#6B4A2E"/>'
            f'<polygon points="{x:.1f},{y-h:.1f} {x+w/2:.1f},{y-h*0.45:.1f} {x-w/2:.1f},{y-h*0.45:.1f}" fill="{c}"/>'
            f'<polygon points="{x:.1f},{y-h*0.75:.1f} {x+w*0.62:.1f},{y-h*0.15:.1f} {x-w*0.62:.1f},{y-h*0.15:.1f}" fill="{c}"/>'
            f'<polygon points="{x:.1f},{y-h*0.5:.1f} {x+w*0.75:.1f},{y-4:.1f} {x-w*0.75:.1f},{y-4:.1f}" fill="{c}"/>\n')

def house(x, y, w=20, h=12, roof=TIGLA):
    return (f'<rect x="{x-w/2:.1f}" y="{y-h:.1f}" width="{w}" height="{h}" fill="#F1E6D2" stroke="{INK2}" stroke-width="1"/>'
            f'<polygon points="{x-w/2-2:.1f},{y-h:.1f} {x:.1f},{y-h-8:.1f} {x+w/2+2:.1f},{y-h:.1f}" fill="{roof}" stroke="{INK2}" stroke-width="1"/>\n')

def station(x, y):
    s = f'<g filter="url(#shs)"><rect x="{x-36:.1f}" y="{y-24:.1f}" width="72" height="28" fill="#F1E6D2" stroke="{INK}" stroke-width="2"/>'

    s += f'<circle cx="{x:.1f}" cy="{y-33:.1f}" r="5" fill="{PARCH}" stroke="{INK}" stroke-width="1.5"/>'
    for dx in (-24, -8, 8, 24):
        s += f'<rect x="{x+dx-3:.1f}" y="{y-16:.1f}" width="6" height="12" fill="{INK}"/>'
    return s + "</g>\n"

def hospital(x, y):
    s = f'<g filter="url(#shs)"><rect x="{x-30:.1f}" y="{y-26:.1f}" width="60" height="30" fill="#F1E6D2" stroke="{INK}" stroke-width="2"/>'

def tram(x, y, sc=1.0, body="#E8C877", band=PETROL, ink="#2B241A"):
    """Stylised early-1900s electric tramcar, side view; (x,y) = centre of the rail line."""
    g = f'<g transform="translate({x:.1f},{y:.1f}) scale({sc:.3f})" filter="url(#shs)">\n'
    # rail
    g += f'<line x1="-120" y1="12" x2="120" y2="12" stroke="{ink}" stroke-width="3"/>\n'
    # wheels
    for wx in (-62, -38, 38, 62):
        g += f'<circle cx="{wx}" cy="4" r="9" fill="#3A3A3A" stroke="{ink}" stroke-width="2"/><circle cx="{wx}" cy="4" r="3" fill="{PARCH}"/>\n'
    # undercarriage
    g += f'<rect x="-84" y="-8" width="168" height="10" rx="2" fill="#5A5A5A" stroke="{ink}" stroke-width="1.5"/>\n'
    # open platforms (ends)
    g += f'<rect x="-104" y="-56" width="22" height="50" rx="4" fill="{band}" stroke="{ink}" stroke-width="2"/>\n'
    g += f'<rect x="82" y="-56" width="22" height="50" rx="4" fill="{band}" stroke="{ink}" stroke-width="2"/>\n'
    g += f'<rect x="-100" y="-50" width="14" height="30" rx="2" fill="#CFE3E6" stroke="{ink}" stroke-width="1.2"/>\n'
    g += f'<rect x="86" y="-50" width="14" height="30" rx="2" fill="#CFE3E6" stroke="{ink}" stroke-width="1.2"/>\n'
    # body
    g += f'<rect x="-84" y="-66" width="168" height="60" rx="6" fill="{body}" stroke="{ink}" stroke-width="2.5"/>\n'
    g += f'<rect x="-84" y="-26" width="168" height="14" fill="{band}" stroke="{ink}" stroke-width="1.5"/>\n'
    # windows
    for i, wx in enumerate((-72, -44, -16, 12, 40)):
        g += f'<rect x="{wx}" y="-58" width="22" height="26" rx="3" fill="#CFE3E6" stroke="{ink}" stroke-width="1.5"/>\n'
        g += f'<line x1="{wx+3}" y1="-55" x2="{wx+9}" y2="-55" stroke="#FFFFFF" stroke-width="2" opacity="0.9"/>\n'
    g += f'<rect x="68" y="-58" width="12" height="26" rx="2" fill="#CFE3E6" stroke="{ink}" stroke-width="1.5"/>\n'
    # roof + clerestory
    g += f'<rect x="-90" y="-74" width="180" height="10" rx="5" fill="#6B4A2E" stroke="{ink}" stroke-width="2"/>\n'
    g += f'<rect x="-60" y="-86" width="120" height="13" rx="4" fill="#8A6A3E" stroke="{ink}" stroke-width="2"/>\n'
    for wx in range(-52, 50, 16):
        g += f'<rect x="{wx}" y="-83" width="10" height="7" fill="#CFE3E6"/>\n'
    # trolley pole + wire
    g += f'<line x1="0" y1="-86" x2="48" y2="-124" stroke="{ink}" stroke-width="3" stroke-linecap="round"/>\n'
    g += f'<circle cx="48" cy="-124" r="4" fill="{ink}"/>\n'
    g += f'<line x1="-140" y1="-126" x2="140" y2="-126" stroke="{ink}" stroke-width="2" stroke-dasharray="1 0" opacity="0.8"/>\n'
    # headlamp + number plate
    g += f'<circle cx="100" cy="-24" r="5" fill="{OCRU}" stroke="{ink}" stroke-width="1.5"/>\n'
    g += f'<rect x="-14" y="-24" width="28" height="10" rx="2" fill="{PARCH}" stroke="{ink}" stroke-width="1"/>\n'
    g += f'<text x="0" y="-16" font-family="{SANS}" font-size="9" font-weight="700" fill="{ink}" text-anchor="middle">1905</text>\n'
    return g + "</g>\n"

def route(pts, w=9, color=RAIL, dash=None, op=1.0):
    d = smooth_path(pts, closed=False, tension=0.5)
    out = f'<path d="{d}" fill="none" stroke="{color}" stroke-width="{w}" stroke-linecap="round" opacity="{op}"' + (f' stroke-dasharray="{dash}"' if dash else "") + '/>\n'
    if not dash:
        out += f'<path d="{d}" fill="none" stroke="{OCRU}" stroke-width="{w*0.3:.1f}" stroke-dasharray="12 14" stroke-linecap="round"/>\n'
    return out
