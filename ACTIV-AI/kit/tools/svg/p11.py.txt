def ribbon(cx, top, w=150, h=120):
    """Panglică în culorile proiectului, deasupra medaliei."""
    o = ""
    stripes = [PETROL, VIOLET, OCRU, VIOLET, PETROL]
    sw = w / len(stripes)
    for i, c in enumerate(stripes):
        x = cx - w / 2 + i * sw
        o += f'<rect x="{x:.1f}" y="{top}" width="{sw+0.6:.1f}" height="{h}" fill="{c}"/>\n'
    # capete tăiate în „V” inversat la bază, prin mască simplă: triunghi de fundal
    o += f'<polygon points="{cx-w/2},{top+h} {cx},{top+h-26} {cx+w/2},{top+h}" fill="{BG}"/>\n'
    # bară de prindere
    o += f'<rect x="{cx-w/2-8}" y="{top-6}" width="{w+16}" height="16" rx="4" fill="url(#gold)" stroke="#8A5F12" stroke-width="1"/>\n'
    # inel
    o += f'<circle cx="{cx}" cy="{top+h+6}" r="16" fill="none" stroke="url(#gold)" stroke-width="8"/>\n'
    return o

def medal_base(cx, cy, R, field):
    o = f'<g filter="url(#sh)"><circle cx="{cx}" cy="{cy}" r="{R}" fill="url(#gold)"/></g>\n'
    o += f'<circle cx="{cx}" cy="{cy}" r="{R}" fill="url(#gold)" stroke="#8A5F12" stroke-width="2"/>\n'
    # zimți pe margine
    n = 72
    for i in range(n):
        a = 2 * math.pi * i / n
        x1 = cx + (R - 4) * math.cos(a); y1 = cy + (R - 4) * math.sin(a)
        x2 = cx + (R - 14) * math.cos(a); y2 = cy + (R - 14) * math.sin(a)
        o += f'<line x1="{x1:.1f}" y1="{y1:.1f}" x2="{x2:.1f}" y2="{y2:.1f}" stroke="#8A5F12" stroke-width="1.2" opacity="0.6"/>\n'
    o += f'<circle cx="{cx}" cy="{cy}" r="{R-18}" fill="none" stroke="#FBE9B0" stroke-width="2"/>\n'
    o += f'<circle cx="{cx}" cy="{cy}" r="{R-22}" fill="none" stroke="#8A5F12" stroke-width="1.5"/>\n'
    # câmp interior
    o += f'<circle cx="{cx}" cy="{cy}" r="{R-80}" fill="url(#{field})" stroke="#FBE9B0" stroke-width="3"/>\n'
    o += f'<circle cx="{cx}" cy="{cy}" r="{R-86}" fill="none" stroke="#8A5F12" stroke-width="1" opacity="0.6"/>\n'
    return o

def arc_chars(cx, cy, r, txt, size, top=True, fill="#3A2A08", ls=3):
    """Text pe arc, caracter cu caracter (librsvg nu redă textPath)."""
    widths = [measure(ch, size, "serif", 700) for ch in txt]
    total = sum(widths) + ls * (len(txt) - 1)
    o = ""
    pos = -total / 2
    for ch, w in zip(txt, widths):
        mid = pos + w / 2
        ang = mid / r  # radiani
        deg = math.degrees(ang)
        if top:
            o += (f'<g transform="translate({cx},{cy}) rotate({deg:.2f})">'
                  f'<text x="0" y="{-r:.1f}" font-family="{SERIF}" font-weight="700" font-size="{size}" fill="{fill}" text-anchor="middle">{esc(ch)}</text></g>\n')
        else:
            o += (f'<g transform="translate({cx},{cy}) rotate({-deg:.2f})">'
                  f'<text x="0" y="{r:.1f}" font-family="{SERIF}" font-weight="700" font-size="{size}" fill="{fill}" text-anchor="middle">{esc(ch)}</text></g>\n')
        pos += w + ls
    return o

def ring_text(cx, cy, top_txt, bot_txt, size=22):
    o = arc_chars(cx, cy, 200, top_txt, size, True)
    o += arc_chars(cx, cy, 214, bot_txt, size - 4, False)
    return o
