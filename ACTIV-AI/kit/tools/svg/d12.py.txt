def node(x, y, col, r=19):
    return (f'<circle cx="{x:.1f}" cy="{y}" r="{r}" fill="{BG}" stroke="{col}" stroke-width="5" filter="url(#glow)"/>'
            f'<circle cx="{x:.1f}" cy="{y}" r="7" fill="{col}"/>\n')


def period(x0, x1, y, label, col):
    o = text(x0, y, label, 15, col, "sans", 700, extra='letter-spacing="1.5"')
    lw = measure(label, 15, "sans", 700, letter_spacing=1.5)
    o += f'<line x1="{x0+lw+14:.1f}" y1="{y-5}" x2="{x1}" y2="{y-5}" stroke="{col}" stroke-width="2" opacity="0.55"/>\n'
    return o


def box(x, y, w, h):
    return f'<rect x="{x:.1f}" y="{y}" width="{w:.1f}" height="{h}" rx="14" fill="{CARD}" stroke="#2A3845" stroke-width="1.5"/>\n'


def clip_open(cid, x, y, w, h):
    return (f'<clipPath id="{cid}"><rect x="{x:.1f}" y="{y}" width="{w:.1f}" height="{h}" rx="14"/></clipPath>'
            f'<g clip-path="url(#{cid})">\n')


def event_text(x, w, r, year, title, body, col, ysize=30):
    R = ROW[r]
    o = text(x + w / 2, R["year"], year, ysize, col, "serif", 700, anchor="middle")
    o += text(x + w / 2, R["title"], title, 18, VAR, "sans", 700, anchor="middle")
    yy = R["body"]
    for ln in wrap(body, w - 24, 16):          # „~” = spațiu care nu se rupe (ex. „10~m”)
        o += text(x + w / 2, yy, ln.replace("~", "\u00a0"), 16, CEATA, anchor="middle")
        yy += 21
    assert yy - 21 <= R["bottom"], (year, yy)
    return o

def waves(x0, x1, y, col=WATER_L, amp=4, step=16, sw=2.5):
    d = f"M{x0:.1f},{y}"
    x = x0
    up = True
    while x < x1:
        d += f" q{step/2:.1f},{-amp if up else amp} {step:.1f},0"
        x += step
        up = not up
    return f'<path d="{d}" fill="none" stroke="{col}" stroke-width="{sw}" stroke-linecap="round"/>\n'

    o += f'<rect x="{px-8}" y="{py-8}" width="{pw+16}" height="{ph+16}" rx="4" fill="{DECK}"/>\n'
    o += f'<rect x="{px}" y="{py}" width="{pw}" height="{ph}" fill="{WATER}"/>\n'
    for i in range(1, 6):

        o += f'<line x1="{px+6}" y1="{yy:.1f}" x2="{px+pw-6}" y2="{yy:.1f}" stroke="{VAR}" stroke-width="1.5" stroke-dasharray="6 5" opacity="0.8"/>\n'

    o += f'<rect x="{tx+30}" y="{by-98}" width="10" height="78" fill="{CEATA}"/>\n'
