def medal(x, y, col, icon, orient=False, r=27):
    if orient:
        ring = (f'<circle cx="{x:.1f}" cy="{y:.1f}" r="{r}" fill="#FBF7EE" stroke="{OCRU_D}" stroke-width="3.2" '
                f'stroke-dasharray="7 5"/>')
        inner = f'<circle cx="{x:.1f}" cy="{y:.1f}" r="{r-5}" fill="none" stroke="{col}" stroke-width="2"/>'
    else:
        ring = f'<circle cx="{x:.1f}" cy="{y:.1f}" r="{r}" fill="#FBF7EE" stroke="{col}" stroke-width="4"/>'
        inner = ""
    return f'<g filter="url(#shs)">{ring}</g>{inner}\n' + G(x, y + 1, (r - 6) / 26, icon, filt=False)


def blob(x, y, rx, ry, seed, fill, op=0.85, dashed=False):
    rnd = random.Random(seed)
    pts = []
    n = 12
    for i in range(n):
        a = 2 * math.pi * i / n + rnd.uniform(-0.1, 0.1)
        k = rnd.uniform(0.85, 1.1)
        pts.append((x + rx * k * math.cos(a), y + ry * k * math.sin(a)))
    d = smooth_path(pts, True, 0.8)
    st = f' stroke="{OCRU_D}" stroke-width="2.2" stroke-dasharray="7 5"' if dashed else f' stroke="#7FAF7A" stroke-width="1.4"'
    return f'<path d="{d}" fill="{fill}" opacity="{op}"{st}/>\n'
