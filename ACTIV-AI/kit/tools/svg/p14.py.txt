def oak_leaf(L, fill, vein=None, n=4.5, base=0.45, p=0.25, fwd=0.05, W=0.27, N=300):
    """Frunză de stejar stilizată (copiată din p13.py): tulpina în (0,0), vârful în sus."""
    def env(t):
        return W * L * (math.sin(math.pi * min(1.0, t ** 1.25)) ** 0.8)
    left = []
    for i in range(N + 1):
        t = 0.06 + 0.94 * i / N
        lob = abs(math.sin(math.pi * n * (t - 0.06) / 0.94)) ** p
        h = env(t) * (base + (1 - base) * lob)
        left.append((-h, -t * L - fwd * h * lob))
    right = [(-x, y) for x, y in reversed(left)]
    pts = [(0, -0.02 * L)] + left + right
    d = "M" + " L".join(f"{x:.1f},{y:.1f}" for x, y in pts) + " Z"
    o = f'<path d="{d}" fill="{fill}" stroke="{fill}" stroke-width="{L*0.012:.1f}" stroke-linejoin="round"/>'
    o += f'<line x1="0" y1="{0.05*L:.1f}" x2="0" y2="{-0.03*L:.1f}" stroke="{fill}" stroke-width="{L*0.045:.1f}" stroke-linecap="round"/>'
    if vein:
        o += f'<line x1="0" y1="{-0.02*L:.1f}" x2="0" y2="{-0.88*L:.1f}" stroke="{vein}" stroke-width="{L*0.02:.1f}" stroke-linecap="round"/>'
        for k in range(1, 5):
            t = 0.16 + k * 0.15
            for sg in (-1, 1):
                o += (f'<line x1="0" y1="{-t*L:.1f}" x2="{sg*0.16*L:.1f}" y2="{-(t+0.10)*L:.1f}" stroke="{vein}" '
                      f'stroke-width="{L*0.012:.1f}" stroke-linecap="round" opacity="0.8"/>')
    return o

def acorn_stalk(x, y, length, n=2, sc=1.0, sway=6):
    """Codiță lungă (peduncul) cu 2–3 ghinde la capăt – de aici „pedunculat”."""
    ex, ey = x + sway, y + length
    o = f'<path d="M{x:.1f},{y:.1f} q{sway*1.6:.1f},{length*0.5:.1f} {sway:.1f},{length:.1f}" fill="none" stroke="{BARK}" stroke-width="{2.2*sc:.1f}" stroke-linecap="round"/>'
    offs = [(-9, -18), (10, -14), (0, 0)][:n] if n == 3 else [(-8, -6), (9, -2)]
    for dx, da in offs:
        o += f'<line x1="{ex:.1f}" y1="{ey:.1f}" x2="{ex+dx*sc:.1f}" y2="{ey+5*sc:.1f}" stroke="{BARK}" stroke-width="{1.6*sc:.1f}"/>'
        o += acorn(ex + dx * sc, ey + 5 * sc, sc, da)
    return o


def limb(x1, y1, x2, y2, w1, w2, col="url(#bark14)"):
    dx, dy = x2 - x1, y2 - y1
    L = math.hypot(dx, dy); nx, ny = -dy / L, dx / L
    pts = [(x1 + nx * w1 / 2, y1 + ny * w1 / 2), (x2 + nx * w2 / 2, y2 + ny * w2 / 2),
           (x2 - nx * w2 / 2, y2 - ny * w2 / 2), (x1 - nx * w1 / 2, y1 - ny * w1 / 2)]
    o = f'<polygon points="{" ".join(f"{a:.1f},{b:.1f}" for a, b in pts)}" fill="{col}"/>'
    o += f'<circle cx="{x2:.1f}" cy="{y2:.1f}" r="{w2/2:.1f}" fill="{col}"/>'
    return o + "\n"


def cluster(cx, cy, r, seed, tone=0):
    """Un nor de frunziș (cerculețe suprapuse, cu umbră și lumină)."""
    rnd = random.Random(seed)
    circ = []
    for _ in range(9):
        a = rnd.uniform(0, 2 * math.pi); d = rnd.uniform(0, r * 0.55)
        circ.append((cx + d * math.cos(a), cy + d * math.sin(a) * 0.8, r * rnd.uniform(0.38, 0.56)))
    cols = [(GD, G1, G2, G3), (G1, G2, G3, G4)][tone]
    o = "".join(f'<circle cx="{x+r*0.10:.1f}" cy="{y+r*0.14:.1f}" r="{rr:.1f}" fill="{cols[0]}"/>' for x, y, rr in circ)
    o += "".join(f'<circle cx="{x:.1f}" cy="{y:.1f}" r="{rr:.1f}" fill="{cols[1]}"/>' for x, y, rr in circ)
    o += "".join(f'<circle cx="{x-rr*0.18:.1f}" cy="{y-rr*0.22:.1f}" r="{rr*0.72:.1f}" fill="{cols[2]}"/>' for x, y, rr in circ)
    o += "".join(f'<circle cx="{x-rr*0.34:.1f}" cy="{y-rr*0.40:.1f}" r="{rr*0.36:.1f}" fill="{cols[3]}" opacity="0.75"/>'
                 for x, y, rr in circ[:5])
    return o + "\n"
