"""D20 – Cât citește clasa mea – infografic AI cu date fictive, declarate.

assert sum(n for _, n in BOOKS) == N

TOTAL = sum(b * n for b, n in BOOKS)

s += ('<pattern id="warn" width="28" height="28" patternUnits="userSpaceOnUse" patternTransform="rotate(45)">'
      f'<rect width="28" height="28" fill="{OCRU}"/><rect width="13" height="28" fill="#1A1A14"/></pattern>\n')


def pill(x, y, lab, col, txt=BG, size=15):
    w = measure(lab, size, "sans", 700, letter_spacing=1.5) + 28
    o = f'<rect x="{x:.1f}" y="{y}" width="{w:.1f}" height="30" rx="15" fill="{col}"/>\n'
    o += text(x + w / 2, y + 20, lab, size, txt, "sans", 700, anchor="middle", extra='letter-spacing="1.5"')
    return o, w


def head(x, y, w, lab, col, sub="date fictive", txt=BG):
    o, pw = pill(x + 22, y + 20, lab, col, txt)
    if sub:
        assert x + 22 + pw + 12 + measure(sub, 14, "sans", 600, True) <= x + w - 16, sub
        o += text(x + 22 + pw + 12, y + 40, sub, 14, TIGLA_L, "sans", 600, True)
    return o


# ------------------------------------------------------------------ banda de avertizare
BY0, BH0 = 218, 46

msg = "DATE FICTIVE – exemplu de lucru, nu un sondaj real"

    s += f'<line x1="{CX0-10}" y1="{y}" x2="{CX0 + 6*(BW+BG_) - BG_ + 10}" y2="{y}" stroke="{LINE}" stroke-width="{2 if v == 0 else 1}"/>\n'
    s += text(CX0 - 18, y + 5, str(v), 13, MUTED, anchor="end")
s += text(CX0 - 40, BASE - 9 * UNIT - 14, "elevi", 13, MUTED, "sans", 400, True)
for k, (b, n) in enumerate(BOOKS):
    x = CX0 + k * (BW + BG_)
    h = n * UNIT
    col = MUTED if b == 0 else OCRU
    s += f'<rect x="{x}" y="{BASE-h}" width="{BW}" height="{h}" rx="4" fill="{col}"/>\n'
    s += text(x + BW / 2, BASE - h - 8, str(n), 17, VAR, "sans", 700, anchor="middle")
    s += text(x + BW / 2, BASE + 22, str(b), 16, VAR, "sans", 700, anchor="middle")
s += text(CX0 + (6 * (BW + BG_) - BG_) / 2, BASE + 42, "cărți citite într-o lună", 13, MUTED, "sans", 400, True,
          anchor="middle")
assert BASE - 9 * UNIT - 30 > R1 + 92
# indicatori
KX = CX0 + 6 * (BW + BG_) + 36
KW = AX_ + AW_ - 22 - KX
assert KW >= 220, KW
KPI = [("43", "de cărți, în total"), ("≈ 1,7", "cărți pe elev, în medie"), ("1 carte", "cel mai des răspuns"),
       ("5 elevi", "n-au citit nicio carte")]
ky = R1 + 104

    s += f'<rect x="{KX}" y="{ky-28}" width="{KW}" height="52" rx="10" fill="{DISC}"/>\n'
    s += text(KX + 14, ky + 6, big, 24, OCRU, "serif", 700)
    s += text(KX + 14 + measure(big, 24, "serif", 700) + 10, ky + 3, small, 14, VAR)
    assert KX + 14 + measure(big, 24, "serif", 700) + 10 + measure(small, 14) < KX + KW - 8, small
    ky += 58

# ------------------------------------------------------------------ B. genul preferat
BX_, BW_ = AX_ + AW_ + 16, 480

    assert measure(lab, 16, "sans", 600) < LBW - 8
    gy += 33
assert BX_ + 22 + LBW + 7 * GUNIT + 30 < BX_ + BW_ - 10

# ------------------------------------------------------------------ C. hârtie sau ecran (1 pătrat = 1 elev)
CX_ = BX_ + BW_ + 16
