def legend_layout():
    """Întoarce [(caracter, unghi_în_grade)] – sens orar, începând de sus, după steaua de început."""
    cap = LS * 0.714
    r_mid = R_BASE + cap / 2
    items = [("STAR", 24)]
    for ch in LEGEND:
        if ch == "*":
            items.append(("DIAM", 12))
        else:
            items.append((ch, measure(ch, LS, "serif", 700)))
    circ = 2 * math.pi * r_mid
    gap = (circ - sum(w for _, w in items)) / len(items)
    out, pos = [], 0.0
    for i, (ch, w) in enumerate(items):
        centre = pos + w / 2 - items[0][1] / 2      # steaua e centrată sus (0°)
        out.append((ch, math.degrees(centre / r_mid)))
        pos += w + gap
    return out, gap


LAYOUT, GAP = legend_layout()


def star(r_out, r_in, n=6):
    pts = []
    for k in range(2 * n):
        rr = r_out if k % 2 == 0 else r_in
        a = math.pi * k / n - math.pi / 2
        pts.append((rr * math.cos(a), rr * math.sin(a)))
    return " ".join(f"{x:.1f},{y:.1f}" for x, y in pts)


def design(c_line, c_text, c_key, sw=1.0):
    """Sigiliul, centrat în (0,0). Culori: linii, text, cheie. sw = factor pentru grosimea liniilor."""
    o = ""
    # cercul dublu exterior
    o += f'<circle r="{R}" fill="none" stroke="{c_line}" stroke-width="{4*sw:.1f}"/>'
    o += f'<circle r="{R-7}" fill="none" stroke="{c_line}" stroke-width="{1.8*sw:.1f}"/>'
    # cercul interior al legendei + șirul de perle
    o += f'<circle r="{R-37}" fill="none" stroke="{c_line}" stroke-width="{1.8*sw:.1f}"/>'
    nb = 60
    for k in range(nb):
        a = 2 * math.pi * k / nb
        o += f'<circle cx="{(R-42)*math.sin(a):.1f}" cy="{-(R-42)*math.cos(a):.1f}" r="1.9" fill="{c_line}"/>'
    # legenda, literă cu literă
    for ch, deg in LAYOUT:
        o += f'<g transform="rotate({deg:.2f})">'
        if ch == "STAR":
            o += f'<polygon transform="translate(0,{-(R_BASE+LS*0.36):.1f})" points="{star(9.5, 4.2)}" fill="{c_text}"/>'
        elif ch == "DIAM":
            yc = -(R_BASE + LS * 0.36)
            o += f'<polygon points="0,{yc-5:.1f} 4,{yc:.1f} 0,{yc+5:.1f} -4,{yc:.1f}" fill="{c_text}"/>'
        elif ch != " ":
            o += (f'<text x="0" y="{-R_BASE:.1f}" font-family="{SERIF}" font-weight="700" font-size="{LS}" '
                  f'fill="{c_text}" text-anchor="middle">{esc(ch)}</text>')
        o += '</g>'
    # câmpul: cheia (asimetrică – barba spre dreapta), inițialele, anul
    k = F / 88.0
    o += f'<g transform="scale({k:.3f})">'
    o += f'<circle cx="0" cy="-46" r="17" fill="none" stroke="{c_key}" stroke-width="{6*sw:.1f}"/>'
    o += f'<polygon points="0,-55 7,-46 0,-37 -7,-46" fill="{c_key}"/>'
    o += f'<line x1="0" y1="-29" x2="0" y2="46" stroke="{c_key}" stroke-width="{7*sw:.1f}" stroke-linecap="round"/>'
    o += f'<line x1="-9" y1="-20" x2="9" y2="-20" stroke="{c_key}" stroke-width="{5*sw:.1f}" stroke-linecap="round"/>'
    o += f'<path d="M2,20 H30 V27 H22 V33 H30 V44 H2 Z" fill="{c_key}"/>'
    for t_, x_ in (("I", -46), ("V", 46)):
        o += (f'<text x="{x_}" y="12" font-family="{SERIF}" font-weight="700" font-size="34" fill="{c_text}" '
              f'text-anchor="middle">{t_}</text>')
    o += (f'<text x="0" y="74" font-family="{SERIF}" font-weight="700" font-size="15" letter-spacing="2" '
          f'fill="{c_text}" text-anchor="middle">{YEAR}</text>')
    for x_ in (-46, 46):
        o += f'<polygon transform="translate({x_},-40)" points="{star(6, 2.6)}" fill="{c_text}"/>'
    o += '</g>'
    return o


def mono(col, sw=1.0):
    return design(col, col, col, sw)
