# ---------------- geometrie: eșantionarea curbelor Catmull-Rom (aceeași formulă ca lib.smooth_path) ----------------
def curve_pts(points, tension=0.5, n=28):
    out = [points[0]]
    m = len(points)
    for i in range(m - 1):
        p0 = points[max(i - 1, 0)]; p1 = points[i]; p2 = points[i + 1]; p3 = points[min(i + 2, m - 1)]
        c1 = (p1[0] + (p2[0] - p0[0]) * tension / 3, p1[1] + (p2[1] - p0[1]) * tension / 3)
        c2 = (p2[0] - (p3[0] - p1[0]) * tension / 3, p2[1] - (p3[1] - p1[1]) * tension / 3)
        for k in range(1, n + 1):
            t = k / n; u = 1 - t
            x = u ** 3 * p1[0] + 3 * u * u * t * c1[0] + 3 * u * t * t * c2[0] + t ** 3 * p2[0]
            y = u ** 3 * p1[1] + 3 * u * u * t * c1[1] + 3 * u * t * t * c2[1] + t ** 3 * p2[1]
            out.append((x, y))
    return out


def along(poly, frac):
    """punct și unghi la fracțiunea frac din lungimea polilinei"""
    L = [0.0]
    for a, b in zip(poly, poly[1:]):
        L.append(L[-1] + math.hypot(b[0] - a[0], b[1] - a[1]))
    tgt = frac * L[-1]
    for i in range(1, len(L)):
        if L[i] >= tgt:
            a, b = poly[i - 1], poly[i]
            seg = L[i] - L[i - 1] or 1
            t = (tgt - L[i - 1]) / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t), math.atan2(b[1] - a[1], b[0] - a[0])
    return poly[-1], 0.0


def poly_len(poly):
    return sum(math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(poly, poly[1:]))

# ---------------- plasare automată a etichetelor ----------------
def hits(B, box, own=None):
    x0, y0, x1, y1 = box
    for it in B.items:
        if own and it[0].split(":")[0] == own:
            continue
        if x0 < it[3] and it[1] < x1 and y0 < it[4] and it[2] < y1:
            return it[0]
    return None


def add_curve(B, prefix, pts, step=7, r=3.5):
    last = None
    for (x, y) in pts:
        if last is None or math.hypot(x - last[0], y - last[1]) >= step:
            B.add(f"{prefix}:pt", x - r, y - r, x + r, y + r)
            last = (x, y)


def layout_block(x, y, lines, anchor):
    """lines: (txt, size, fill, weight, italic, fam). Returnează (baseline-uri, lățimi, bbox)."""
    ys, ws = [], []
    cy = y
    for i, ln in enumerate(lines):
        t, sz, _, w, it, fam = ln
        if i > 0:
            cy += round(sz * 1.18) + 2
        ys.append(cy)
        ws.append(measure(t, sz, fam, w, it))
    W = max(ws)
    x0 = x if anchor == "start" else (x - W / 2 if anchor == "middle" else x - W)
    top = y - lines[0][1] * 0.80
    bot = ys[-1] + lines[-1][1] * 0.26
    return ys, ws, (x0, top, x0 + W, bot)


def place(B, name, x, y, lines, cands, bounds, hcol=PARCH, hw=5, pad=3, report=True):
    """Încearcă pe rând pozițiile candidate (dx, dy, ancoră); alege prima fără coliziuni și în interiorul cadrului."""
    chosen = None
    for (dx, dy, anc) in cands:
        ys, ws, (bx0, by0, bx1, by1) = layout_block(x + dx, y + dy, lines, anc)
        box = (bx0 - pad, by0 - pad, bx1 + pad, by1 + pad)
        inside = bounds[0] <= box[0] and bounds[1] <= box[1] and box[2] <= bounds[2] and box[3] <= bounds[3]
        if inside and not hits(B, box, name):
            chosen = (dx, dy, anc, ys, ws)
            break
    if chosen is None:
        dx, dy, anc = cands[0]
        ys, ws, _ = layout_block(x + dx, y + dy, lines, anc)
        chosen = (dx, dy, anc, ys, ws)
        if report:
            print("  ! fără loc liber pentru", name)
    dx, dy, anc, ys, ws = chosen
    out = ""
    for i, (ln, yy, w) in enumerate(zip(lines, ys, ws)):
        t, sz, fill, wt, it, fam = ln
        out += halo(x + dx, yy, t, sz, fill, wt, it, anc, hw=hw, hcol=hcol, fam=fam)
        B.text(f"{name}:l{i}", x + dx, yy, w, sz, anc)
    return out, (x + dx, y + dy, anc)

def hits_ign(B, box, own, ignore=()):
    x0, y0, x1, y1 = box
    for it in B.items:
        pf = it[0].split(":")[0]
        if pf == own or pf in ignore:
            continue
        if x0 < it[3] and it[1] < x1 and y0 < it[4] and it[2] < y1:
            return it[0]
    return None


def check(B, bounds):
    """Raport filtrat: semnalează doar etichete care ating orice altceva și pictograme suprapuse;
    liniile (râu, trasee) au voie să treacă prin pictogramele propriilor repere."""
    def kind(n):
        if n.endswith(":pt") or n in ("riv:lake", "riv:src"):
            return "curve"
        if ":l" in n or n in ("munti", "note", "legend", "compass", "inset", "hdr"):
            return "label"
        return "icon"
    bad = []
    it = B.items
    for i in range(len(it)):
        for j in range(i + 1, len(it)):
            a, b = it[i], it[j]
            if a[0].split(":")[0] == b[0].split(":")[0]:
                continue
            if not (a[1] < b[3] and b[1] < a[3] and a[2] < b[4] and b[2] < a[4]):
                continue
            ka, kb = kind(a[0]), kind(b[0])
            if "label" in (ka, kb) or (ka == kb == "icon"):
                bad.append((a[0], b[0]))
    X0, Y0, X1, Y1 = bounds
    for a in it:
        if kind(a[0]) != "curve" and a[0] != "hdr" and (a[1] < X0 or a[2] < Y0 or a[3] > X1 or a[4] > Y1):
            bad.append((a[0], "OUT"))
    for b in sorted(set(bad)):
        print("  overlap:", b)
    return bad


def place_marks(B, prefix, curve, kind, n, size, lo=0.1, hi=0.92, gap=0.16, ignore=()):
    out = ""
    chosen = []
    cands = [lo + i * (hi - lo) / 60 for i in range(61)]
    targets = [lo + (k + 0.5) * (hi - lo) / n for k in range(n)]
    h = size / 2 + 3
    for t in targets:
        for c in sorted(cands, key=lambda c: abs(c - t)):
            if any(abs(c - o) < gap for o in chosen):
                continue
            (x, y), _ = along(curve, c)
            if hits_ign(B, (x - h, y - h, x + h, y + h), prefix, ignore):
                continue
            chosen.append(c)
            out += mark_symbol(kind, x, y, size)
            B.add(f"{prefix}:m{c:.3f}", x - h + 3, y - h + 3, x + h - 3, y + h - 3)
            break
    return out


def place_badge(B, prefix, curve, n, col, r=12, fr=(0.5, 0.4, 0.6, 0.3, 0.7, 0.2, 0.8), off=22, ignore=()):
    for f in fr:
        (x, y), ang = along(curve, f)
        for side in (1, -1):
            bx = x + side * off * math.cos(ang + math.pi / 2); by = y + side * off * math.sin(ang + math.pi / 2)
            box = (bx - r - 2, by - r - 2, bx + r + 2, by + r + 2)
            if not hits_ign(B, box, prefix, ignore):
                B.add(f"{prefix}:badge", *box)
                return num_badge(bx, by, n, col, r)
    print("  ! fără loc pentru insigna", n)
    (x, y), _ = along(curve, fr[0])
    return num_badge(x, y - off, n, col, r)

def inset_paltinis(IX, IY, IW, IH, HDR, Z0, Z1, T1, T2):
    B = Boxes()
    pad = 14
    kx = (IW - 2 * pad) / (Z1[0] - Z0[0]); ky = (IH - HDR - 2 * pad) / (Z1[1] - Z0[1])
    BND = (IX + 6, IY + HDR + 4, IX + IW - 6, IY + IH - 6)
    HC = "#EEF0DE"

    def Q(nx, ny):
        return (IX + pad + (nx - Z0[0]) * kx, IY + HDR + pad + (ny - Z0[1]) * ky)

    s = f'<g filter="url(#sh)"><rect x="{IX}" y="{IY}" width="{IW}" height="{IH}" rx="16" fill="#E9EDD8" stroke="{OCRU_D}" stroke-width="2.4"/></g>\n'
    s += f'<clipPath id="iclip"><rect x="{IX}" y="{IY}" width="{IW}" height="{IH}" rx="16"/></clipPath>\n<g clip-path="url(#iclip)">\n'
    s += f'<rect x="{IX}" y="{IY+HDR}" width="{IW}" height="{IH-HDR}" fill="#B9CFA6"/>\n'
    mx, my = Q(-0.66, 0.735)
    s += f'<ellipse cx="{mx:.1f}" cy="{my:.1f}" rx="78" ry="56" fill="#DCDDB8" opacity="0.9" filter="url(#blur8)"/>\n'
    s += f'<rect x="{IX}" y="{IY}" width="{IW}" height="{HDR}" fill="#1B2733"/>\n'
    s += text(IX + 16, IY + 27, "Detaliu: zona Păltiniș", 19, VAR, "serif", 700)
    sub = "fondată în 1894 · prima stațiune montană din România"
    s += text(IX + 16, IY + 50, sub, 13, CEATA, "sans", 400, True)
    assert IX + 16 + measure(sub, 13, "sans", 400, True) <= IX + IW - 10
    B.add("hdr", IX, IY, IX + IW, IY + HDR)

    t1 = [Q(*q) for q in T1]
    t2 = [Q(*q) for q in T2]
    c1 = curve_pts(t1); c2 = curve_pts(t2)
    add_curve(B, "t1", c1, 6, 4)
    add_curve(B, "t2", c2, 6, 4)

    pX, pY = Q(-0.5, 0.62)
    oX, oY = Q(-0.58, 0.68)
    bX, bY = Q(-0.66, 0.72)
    plX, plY = Q(*T2[3])
    stX, stY = Q(*T2[5])
    dcX, dcY = Q(*T1[2])
    gjX, gjY = Q(*T1[4])

    icons = ""
    icons += resort(pX, pY - 4, 1.0)
    B.add("Pal:icon", pX - 28, pY - 32, pX + 28, pY + 10)
    icons += f'<g filter="url(#shs)">' + peak(oX, oY, 46, 32, False, "#8F8A78", "#746F60") + "</g>" + summit_mark(oX, oY - 4, 6)
    B.add("Onc:icon", oX - 23, oY - 10, oX + 23, oY + 32)
    icons += saddle(bX, bY + 6, 46)
    B.add("Bat:icon", bX - 23, bY - 8, bX + 23, bY + 15)
    icons += meadow(plX, plY, 7) + ring_orient(stX, stY, 7) + spring(dcX, dcY, 8) + meadow(gjX, gjY, 8)
    # repere orientative: inel punctat în jurul pictogramei
    for (qx, qy) in ((plX, plY), (dcX, dcY), (gjX, gjY)):
        icons += f'<circle cx="{qx:.1f}" cy="{qy:.1f}" r="13" fill="none" stroke="{OCRU_D}" stroke-width="2" stroke-dasharray="4 3"/>'
    for nm, (qx, qy) in (("PL", (plX, plY)), ("St", (stX, stY)), ("DC", (dcX, dcY)), ("GJ", (gjX, gjY))):
        B.add(nm + ":icon", qx - 14, qy - 14, qx + 14, qy + 14)
    B.add("t1:end", bX - 8, bY - 10, bX + 8, bY + 6)

    labels = ""

    def put(name, x, y, lines, cands, **kw):
        nonlocal labels
        svg, _ = place(B, name, x, y, lines, cands, BND, hcol=HC, **kw)
        labels += svg

    put("Pal", pX, pY, [L("Păltiniș", 19, INK, 700, False, "serif"), L("1.440 / 1.442 m", 13, OCRU_D, 700),
                        L("start: telescaun", 13, INK2, 400, True)],
        [(30, 2, "start"), (30, 14, "start"), (24, 22, "start"), (-30, 2, "end")])
    put("St", stX, stY, [L("Șanta", 14)], [(0, -16, "middle"), (16, 5, "start"), (-16, 5, "end"), (0, 28, "middle")])
    put("PL", plX, plY, [L("Poiana", 14), L("Lupilor", 14)], [(-18, -2, "end"), (-18, 6, "end"), (0, 30, "middle"), (18, -2, "start")])
    put("Bat", bX, bY, [L("Șaua Bătrâna", 15), L("1.863 m", 13, OCRU_D, 700)],
        [(-8, -30, "middle"), (-14, -32, "middle"), (0, -26, "middle"), (28, 4, "start"), (0, 40, "middle")])
    put("Onc", oX, oY, [L("Vf. Oncești", 15)], [(0, 52, "middle"), (-6, 52, "middle"), (26, 22, "start"), (-26, 22, "end"), (0, -16, "middle")])
    put("DC", dcX, dcY, [L("izvorul", 14), L("Dorul Clujului", 14)], [(18, -2, "start"), (18, 8, "start"), (-18, -2, "end"), (0, 30, "middle")])
    put("GJ", gjX, gjY, [L("Poiana Găujoara", 14)], [(0, 32, "middle"), (-20, 30, "middle"), (18, 26, "start"), (0, -20, "middle")])
    put("orient", IX + IW - 14, IY + IH - 14, [L("cerc punctat = loc orientativ", 12, OCRU_D, 600, True)],
        [(0, 0, "end"), (-200, -10, "start")], hw=4)

    # traseele, simbolurile și capetele (după etichete, ca să le ocolească)
    tr = trail_path(t2, BLUE, 5, 9) + trail_path(t1, RED, 5, 9)
    tr += endpoint(pX, pY, INK, 7) + endpoint(bX, bY - 2, RED, 6)
    tr += place_marks(B, "t2", c2, "triunghi_albastru", 3, 20, 0.1, 0.92, 0.18, ignore=("t1",))
    tr += place_marks(B, "t1", c1, "cruce_rosie", 2, 20, 0.12, 0.9, 0.25, ignore=("t2",))
    tr += place_badge(B, "t1", c1, 1, RED, 12, fr=(0.55, 0.45, 0.65, 0.35, 0.75), off=20)
    tr += place_badge(B, "t2", c2, 2, BLUE, 12, fr=(0.3, 0.4, 0.2, 0.6, 0.7), off=20)

    # molizi în detaliu
    trees = ""
    rnd = random.Random(23)
    cand = [(rnd.uniform(IY + HDR + 16, IY + IH + 10), rnd.uniform(IX + 6, IX + IW - 6)) for _ in range(300)]
    for y, x in sorted(cand):
        h = rnd.uniform(14, 20)
        if hits(B, (x - h * 0.45, y - h, x + h * 0.45, y + 4)):
            continue
        if math.hypot((x - mx) / 1.3, y - my) < 52:
            continue
        trees += fir(x, y, h, rnd.choice([FIR1, FIR2, FIR3])) + "\n"

    s += trees + tr + icons + labels + "</g>\n"
    s += f'<rect x="{IX}" y="{IY}" width="{IW}" height="{IH}" rx="16" fill="none" stroke="{OCRU_D}" stroke-width="2.4"/>\n'
    return s, B
