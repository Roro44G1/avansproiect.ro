def P(pts):
    return " ".join(f"{x:.1f},{y:.1f}" for x, y in pts)

def eye_dormer(cx, cy, w, h, lid, line, opening="#1A1410", sw=1.4, pupil=None, lash=False):
    """Lucarnă „ochi” pe acoperiș: pleoapa (arcul de țiglă ridicat) și fanta de ventilare dedesubt.
    (cx, cy) = mijlocul bazei; w = lățimea, h = înălțimea pleoapei."""
    o = (f'<path d="M{cx-w/2:.1f},{cy:.1f} C{cx-w*0.30:.1f},{cy-h*1.32:.1f} {cx+w*0.30:.1f},{cy-h*1.32:.1f} {cx+w/2:.1f},{cy:.1f} Z" '
         f'fill="{lid}" stroke="{line}" stroke-width="{sw:.1f}" stroke-linejoin="round"/>')
    o += (f'<path d="M{cx-w*0.31:.1f},{cy:.1f} C{cx-w*0.17:.1f},{cy-h*0.78:.1f} {cx+w*0.17:.1f},{cy-h*0.78:.1f} {cx+w*0.31:.1f},{cy:.1f} Z" '
          f'fill="{opening}" stroke="{line}" stroke-width="{sw*0.7:.1f}"/>')
    if pupil:
        o += f'<circle cx="{cx:.1f}" cy="{cy-h*0.2:.1f}" r="{h*0.26:.1f}" fill="{pupil}"/>'
    if lash:
        for k in (-1, 0, 1):
            x_ = cx + k * w * 0.16
            o += (f'<line x1="{x_:.1f}" y1="{cy-h*0.98:.1f}" x2="{x_+k*w*0.05:.1f}" y2="{cy-h*1.28:.1f}" '
                  f'stroke="{line}" stroke-width="{sw:.1f}" stroke-linecap="round"/>')
    return o + "\n"


def house(x, base, w, wall_h, roof_h, wall, roof, line, eyes=1, eye_open="#1A1410", win="#3A3128",
          sw=1.4, tiles=True, gable=False, wins=2, lit=None):
    """Casă simplificată, cu acoperiș înalt și lucarne „ochi”. Returnează SVG."""
    o = ""
    top = base - wall_h
    o += f'<rect x="{x:.1f}" y="{top:.1f}" width="{w:.1f}" height="{wall_h:.1f}" fill="{wall}" stroke="{line}" stroke-width="{sw:.1f}"/>'
    # ferestre
    if wins:
        ww = w * 0.16; wh = wall_h * 0.34
        for i in range(wins):
            wx = x + w * (i + 1) / (wins + 1) - ww / 2
            for row in range(1 if wall_h < 60 else 2):
                wy = top + wall_h * (0.16 + row * 0.44)
                fc = lit if (lit and (i + row) % 2 == 0) else win
                o += f'<rect x="{wx:.1f}" y="{wy:.1f}" width="{ww:.1f}" height="{wh:.1f}" fill="{fc}" stroke="{line}" stroke-width="{sw*0.6:.1f}"/>'
    # acoperișul
    ov = w * 0.06
    if gable:
        pts = [(x - ov, top), (x + w / 2, top - roof_h), (x + w + ov, top)]
    else:
        pts = [(x - ov, top), (x + w * 0.2, top - roof_h), (x + w * 0.8, top - roof_h), (x + w + ov, top)]
    o += f'<polygon points="{P(pts)}" fill="{roof}" stroke="{line}" stroke-width="{sw:.1f}" stroke-linejoin="round"/>'
    if tiles:
        n = max(3, int(roof_h / 9))
        for k in range(1, n):
            yy = top - roof_h * k / n
            t = k / n
            if gable:
                xl = x - ov + (w / 2 + ov) * t; xr = x + w + ov - (w / 2 + ov) * t
            else:
                xl = x - ov + (w * 0.2 + ov) * t; xr = x + w + ov - (w * 0.2 + ov) * t
            o += f'<line x1="{xl+1:.1f}" y1="{yy:.1f}" x2="{xr-1:.1f}" y2="{yy:.1f}" stroke="{line}" stroke-width="{sw*0.45:.1f}" opacity="0.35"/>'
    # lucarnele
    ew = min(w * 0.30, roof_h * 0.9)
    eh = ew * 0.30
    if eyes == 1:
        o += eye_dormer(x + w / 2, top - roof_h * 0.40, ew, eh, roof, line, eye_open, sw)
    elif eyes >= 2:
        for f_ in (0.30, 0.70):
            o += eye_dormer(x + w * f_, top - roof_h * 0.36, ew * 0.85, eh * 0.85, roof, line, eye_open, sw)
        if eyes >= 3:
            o += eye_dormer(x + w / 2, top - roof_h * 0.70, ew * 0.62, eh * 0.62, roof, line, eye_open, sw)
    return o + "\n"


def council_tower(cx, base, w, h, stone, roof, line, dark="#2A231C", sw=1.6, detail=True):
    """Turnul Sfatului, desen simplificat: corp prismatic cu gang boltit la bază, galerie sus,
    acoperiș ascuțit cu patru turnulețe de colț (două vizibile). Returnează SVG."""
    o = ""
    top = base - h
    o += f'<rect x="{cx-w/2:.1f}" y="{top:.1f}" width="{w:.1f}" height="{h:.1f}" fill="{stone}" stroke="{line}" stroke-width="{sw:.1f}"/>'
    # gangul boltit
    aw = w * 0.52; ah = h * 0.20
    o += (f'<path d="M{cx-aw/2:.1f},{base:.1f} V{base-ah*0.62:.1f} A{aw/2:.1f},{aw/2:.1f} 0 0 1 {cx+aw/2:.1f},{base-ah*0.62:.1f} V{base:.1f} Z" '
          f'fill="{dark}" stroke="{line}" stroke-width="{sw*0.8:.1f}"/>')
    if detail:
        # brâuri
        for f_ in (0.30, 0.56):
            yy = base - h * f_
            o += f'<line x1="{cx-w/2:.1f}" y1="{yy:.1f}" x2="{cx+w/2:.1f}" y2="{yy:.1f}" stroke="{line}" stroke-width="{sw*0.7:.1f}"/>'
        # ferestre înguste
        for f_ in (0.40, 0.66):
            yy = base - h * f_
            o += f'<rect x="{cx-w*0.07:.1f}" y="{yy-h*0.05:.1f}" width="{w*0.14:.1f}" height="{h*0.09:.1f}" rx="{w*0.07:.1f}" fill="{dark}"/>'
    # galeria de sus (ușor ieșită)
    gy = top + h * 0.02
    gh = h * 0.16
    o += f'<rect x="{cx-w/2-w*0.06:.1f}" y="{gy:.1f}" width="{w*1.12:.1f}" height="{gh:.1f}" fill="{stone}" stroke="{line}" stroke-width="{sw:.1f}"/>'
    for i in range(3):
        wx = cx - w * 0.36 + i * w * 0.36
        o += (f'<path d="M{wx-w*0.08:.1f},{gy+gh*0.85:.1f} V{gy+gh*0.42:.1f} A{w*0.08:.1f},{w*0.08:.1f} 0 0 1 {wx+w*0.08:.1f},{gy+gh*0.42:.1f} V{gy+gh*0.85:.1f} Z" fill="{dark}"/>')
    # acoperișul ascuțit
    rt = gy
    rh = h * 0.55
    o += f'<polygon points="{P([(cx-w/2-w*0.10, rt), (cx, rt-rh), (cx+w/2+w*0.10, rt)])}" fill="{roof}" stroke="{line}" stroke-width="{sw:.1f}" stroke-linejoin="round"/>'
    o += f'<line x1="{cx:.1f}" y1="{rt-rh:.1f}" x2="{cx:.1f}" y2="{rt:.1f}" stroke="{line}" stroke-width="{sw*0.5:.1f}" opacity="0.45"/>'
    # turnulețele de colț
    for sg in (-1, 1):
        tx = cx + sg * (w / 2 + w * 0.02)
        tw = w * 0.20; th = rh * 0.42
        o += f'<rect x="{tx-tw/2:.1f}" y="{rt-th*0.28:.1f}" width="{tw:.1f}" height="{th*0.28:.1f}" fill="{stone}" stroke="{line}" stroke-width="{sw*0.8:.1f}"/>'
        o += f'<polygon points="{P([(tx-tw/2-1, rt-th*0.28), (tx, rt-th*1.15), (tx+tw/2+1, rt-th*0.28)])}" fill="{roof}" stroke="{line}" stroke-width="{sw*0.8:.1f}" stroke-linejoin="round"/>'
    # vârful
    o += f'<line x1="{cx:.1f}" y1="{rt-rh:.1f}" x2="{cx:.1f}" y2="{rt-rh-h*0.07:.1f}" stroke="{line}" stroke-width="{sw:.1f}"/>'
    o += f'<circle cx="{cx:.1f}" cy="{rt-rh-h*0.035:.1f}" r="{max(1.6, w*0.04):.1f}" fill="{line}"/>'
    return o + "\n"


def lies_bridge(x0, x1, deck, rise, iron, line, rail_h, sw=1.6, stone="#8C8274", stone_h=None, ground=None,
                detail=True, rail_fill=None):
    """Podul Minciunilor, desen simplificat: arc de fontă cu cercuri în timpane, balustradă, culei de piatră."""
    o = ""
    mid = (x0 + x1) / 2
    y0 = deck + rise
    yc = 2 * (deck + 3) - y0
    # culeele de piatră
    gh = stone_h if stone_h else rise + 10
    cw = (x1 - x0) * 0.07
    for xa in (x0 - cw, x1):
        o += f'<rect x="{xa:.1f}" y="{deck:.1f}" width="{cw:.1f}" height="{gh:.1f}" fill="{stone}" stroke="{line}" stroke-width="{sw:.1f}"/>'
        if detail:
            for k in range(1, 4):
                yy = deck + gh * k / 4
                o += f'<line x1="{xa:.1f}" y1="{yy:.1f}" x2="{xa+cw:.1f}" y2="{yy:.1f}" stroke="{line}" stroke-width="{sw*0.5:.1f}" opacity="0.6"/>'
    # cercurile din timpane (mai mari spre capete)
    if detail:
        n = 9
        for i in range(n):
            t = (i + 0.5) / n
            xx = x0 + (x1 - x0) * t
            # înălțimea arcului în acest punct
            ya = (1 - t) ** 2 * y0 + 2 * (1 - t) * t * yc + t ** 2 * y0
            gap = ya - (deck + 6)
            if gap > 7:
                r = min(gap * 0.40, (x1 - x0) / n * 0.38)
                o += f'<circle cx="{xx:.1f}" cy="{deck+6+gap/2:.1f}" r="{r:.1f}" fill="none" stroke="{iron}" stroke-width="{sw*1.1:.1f}"/>'
    # arcul (dublu)
    o += f'<path d="M{x0:.1f},{y0:.1f} Q{mid:.1f},{yc:.1f} {x1:.1f},{y0:.1f}" fill="none" stroke="{iron}" stroke-width="{sw*3.2:.1f}"/>'
    o += f'<path d="M{x0:.1f},{y0-6:.1f} Q{mid:.1f},{yc-6:.1f} {x1:.1f},{y0-6:.1f}" fill="none" stroke="{iron}" stroke-width="{sw*1.2:.1f}"/>'
    # tablierul
    o += f'<rect x="{x0-4:.1f}" y="{deck:.1f}" width="{x1-x0+8:.1f}" height="7" fill="{iron}" stroke="{line}" stroke-width="{sw*0.6:.1f}"/>'
    # balustrada
    ry = deck - rail_h
    if rail_fill:
        o += f'<rect x="{x0:.1f}" y="{ry:.1f}" width="{x1-x0:.1f}" height="{rail_h:.1f}" fill="{rail_fill}"/>'
    o += f'<line x1="{x0-4:.1f}" y1="{ry:.1f}" x2="{x1+4:.1f}" y2="{ry:.1f}" stroke="{iron}" stroke-width="{sw*1.8:.1f}" stroke-linecap="round"/>'
    step = max(6.0, rail_h * 0.55)
    nb = int((x1 - x0) / step)
    for i in range(nb + 1):
        xx = x0 + (x1 - x0) * i / nb
        o += f'<line x1="{xx:.1f}" y1="{ry:.1f}" x2="{xx:.1f}" y2="{deck:.1f}" stroke="{iron}" stroke-width="{sw*0.75:.1f}"/>'
    if detail and rail_h > 9:
        for i in range(nb):
            xx = x0 + (x1 - x0) * (i + 0.5) / nb
            o += f'<circle cx="{xx:.1f}" cy="{ry+rail_h*0.5:.1f}" r="{min(step, rail_h)*0.26:.1f}" fill="none" stroke="{iron}" stroke-width="{sw*0.6:.1f}"/>'
    # stâlpii de capăt
    for xa in (x0 - 4, x1 + 4):
        o += f'<rect x="{xa-3:.1f}" y="{ry-5:.1f}" width="6" height="{rail_h+5:.1f}" fill="{iron}"/>'
    return o + "\n"


def teen(cx, foot, h, fill, arm_up=True):
    """Siluetă simplă de adolescent (cu rucsac), cu o mână ridicată ca la salutul de la final."""
    hr = h * 0.095
    o = f'<circle cx="{cx:.1f}" cy="{foot-h+hr:.1f}" r="{hr:.1f}" fill="{fill}"/>'
    sh = foot - h + hr * 2 + h * 0.03     # umerii
    hip = foot - h * 0.45
    o += f'<path d="M{cx-h*0.12:.1f},{sh+h*0.02:.1f} Q{cx:.1f},{sh-h*0.02:.1f} {cx+h*0.12:.1f},{sh+h*0.02:.1f} L{cx+h*0.10:.1f},{hip:.1f} L{cx-h*0.10:.1f},{hip:.1f} Z" fill="{fill}"/>'
    # rucsacul
    o += f'<rect x="{cx-h*0.20:.1f}" y="{sh+h*0.04:.1f}" width="{h*0.10:.1f}" height="{h*0.22:.1f}" rx="{h*0.03:.1f}" fill="{fill}"/>'
    # picioarele
    o += f'<path d="M{cx-h*0.09:.1f},{hip-1:.1f} L{cx-h*0.10:.1f},{foot:.1f} L{cx-h*0.02:.1f},{foot:.1f} L{cx:.1f},{hip+h*0.08:.1f} L{cx+h*0.02:.1f},{foot:.1f} L{cx+h*0.10:.1f},{foot:.1f} L{cx+h*0.09:.1f},{hip-1:.1f} Z" fill="{fill}"/>'
    # brațele
    aw = max(2.0, h * 0.055)
    o += f'<line x1="{cx-h*0.11:.1f}" y1="{sh+h*0.04:.1f}" x2="{cx-h*0.16:.1f}" y2="{hip+h*0.02:.1f}" stroke="{fill}" stroke-width="{aw:.1f}" stroke-linecap="round"/>'
    if arm_up:
        o += f'<path d="M{cx+h*0.11:.1f},{sh+h*0.04:.1f} L{cx+h*0.25:.1f},{sh-h*0.08:.1f} L{cx+h*0.30:.1f},{sh-h*0.26:.1f}" fill="none" stroke="{fill}" stroke-width="{aw:.1f}" stroke-linecap="round" stroke-linejoin="round"/>'
    else:
        o += f'<line x1="{cx+h*0.11:.1f}" y1="{sh+h*0.04:.1f}" x2="{cx+h*0.16:.1f}" y2="{hip+h*0.02:.1f}" stroke="{fill}" stroke-width="{aw:.1f}" stroke-linecap="round"/>'
    return o + "\n"


def fit(s_, maxw, size, fam="serif", weight=700, italic=False, step=0.5, min_size=8):
    while measure(s_, size, fam, weight, italic) > maxw and size > min_size:
        size -= step
    return size
