def plate(x, y, w, h, kind, name, tool, num=None, name_size=30, num_col=OCRU):
    s = (f'<g filter="url(#sh)"><rect x="{x}" y="{y}" width="{w}" height="{h}" rx="10" fill="{PLATE}" '
         f'stroke="{PLATE_D}" stroke-width="2"/></g>\n')
    s += f'<rect x="{x+5}" y="{y+5}" width="{w-10}" height="{h-10}" rx="7" fill="none" stroke="{VAR}" stroke-width="2.2"/>\n'
    r = (h - 22) / 2
    cx, cy = x + 13 + r, y + h / 2
    s += f'<circle cx="{cx:.1f}" cy="{cy:.1f}" r="{r:.1f}" fill="{PARCH}" stroke="{VAR}" stroke-width="2"/>\n'
    s += G(cx, cy + 1, (r - 4) / 28, TOOLS[tool](), filt=False)
    tx = cx + r + 16
    s += text(tx, y + h * 0.40, kind, 13, "#CFE0F5", "sans", 700, extra='letter-spacing="3"')
    s += text(tx, y + h * 0.40 + name_size * 1.02, name, name_size, VAR, "serif", 700)
    nw = measure(name, name_size, "serif", 700)
    assert tx + nw <= x + w - (44 if num else 14), ("plate name", name)
    if num is not None:
        s += (f'<circle cx="{x + w - 24}" cy="{y + 24}" r="14" fill="{num_col}" stroke="{VAR}" stroke-width="2"/>'
              + text(x + w - 24, y + 30, str(num), 17, BG, "sans", 700, anchor="middle"))
    return s


def mini_plate(x, y, name, tool, num, size=15, h=32):
    """plăcuță mică: [număr][unealtă][nume]; (x,y) = colțul stânga-sus. Returnează (svg, lățime)."""
    nw = measure(name, size, "sans", 700)
    w = 8 + 26 + 4 + 26 + 8 + nw + 12
    s = f'<g filter="url(#shs)"><rect x="{x:.1f}" y="{y:.1f}" width="{w:.1f}" height="{h}" rx="6" fill="{PLATE}" stroke="{PLATE_D}" stroke-width="1.5"/></g>'
    s += f'<rect x="{x+3:.1f}" y="{y+3:.1f}" width="{w-6:.1f}" height="{h-6}" rx="4" fill="none" stroke="{VAR}" stroke-width="1.3"/>'
    s += f'<circle cx="{x+8+13:.1f}" cy="{y+h/2:.1f}" r="11" fill="{OCRU}"/>' + text(x + 21, y + h / 2 + 5, str(num), 14, BG, "sans", 700, anchor="middle")
    s += f'<circle cx="{x+38+13:.1f}" cy="{y+h/2:.1f}" r="12" fill="{PARCH}" stroke="{VAR}" stroke-width="1.2"/>'
    s += G(x + 51, y + h / 2 + 0.5, 9.5 / 28, TOOLS[tool](), filt=False)
    s += text(x + 72, y + h / 2 + size * 0.36, name, size, VAR, "sans", 700)
    return s + "\n", w


def signpost(X, Y, name, tool, num, pole=44, side="right"):
    """indicator pe stâlp: baza stâlpului în (X,Y); plăcuța se întinde spre dreapta sau spre stânga"""
    _, w = mini_plate(0, 0, name, tool, num)
    h = 32
    s = f'<ellipse cx="{X:.1f}" cy="{Y+1:.1f}" rx="9" ry="3" fill="#000" opacity="0.18"/>'
    s += f'<path d="M{X:.1f},{Y:.1f} V{Y-pole-h/2:.1f}" stroke="#5A5F66" stroke-width="4" stroke-linecap="round"/>'
    px, py = (X - 21, Y - pole - h) if side == "right" else (X - w + 21, Y - pole - h)
    svg, _ = mini_plate(px, py, name, tool, num)
    return s + svg, (px, py, px + w, py + h)


def para_prefix(x, y, prefix, text_, size, maxw, pcol=OCRU, tcol=VAR, lh=None, psize=None):
    """paragraf cu un prefix colorat (anul) pe primul rând; returnează (svg, y_următor, nr_rânduri)"""
    lh = lh or round(size * 1.3)
    psize = psize or size
    pw = measure(prefix + " ", psize, "sans", 700)
    words = text_.split()
    lines, cur, first = [], "", True
    for wd in words:
        t = (cur + " " + wd).strip()
        lim = maxw - (pw if first else 0)
        if measure(t, size) <= lim or not cur:
            cur = t
        else:
            lines.append(cur); cur = wd; first = False
    if cur:
        lines.append(cur)
    out = ""
    for i, ln in enumerate(lines):
        if i == 0:
            out += rich(x, y, [(prefix + " ", pcol, 700, False, psize), (ln, tcol, 400, False, size)], size)
        else:
            out += text(x, y, ln, size, tcol)
        y += lh
    return out, y, len(lines)

        t = i / n
        px, py = a0[0] + (a1[0] - a0[0]) * t, a0[1] + (a1[1] - a0[1]) * t
        steps += (f'<line x1="{px - 7 * nx_:.1f}" y1="{py - 7 * ny_:.1f}" x2="{px + 7 * nx_:.1f}" y2="{py + 7 * ny_:.1f}" '
                  f'stroke="{INK2}" stroke-width="1.6"/>')
    for i in range(0, 30):
        t = i / 29
        px, py = a0[0] + (a1[0] - a0[0]) * t, a0[1] + (a1[1] - a0[1]) * t
        B.add("pasaj:pt", px - 6, py - 6, px + 6, py + 6)
    # Piața Aurarilor (contur punctat = poziție orientativă)
    pa = P(395, 135)
    pa_w, pa_h = 64 * k, 52 * k
    steps += (f'<rect x="{pa[0]:.1f}" y="{pa[1]:.1f}" width="{pa_w:.1f}" height="{pa_h:.1f}" rx="5" fill="#FBF7EE" '
              f'stroke="{OCRU_D}" stroke-width="2.4" stroke-dasharray="6 4"/>')
    B.add("PAur:icon", pa[0], pa[1], pa[0] + pa_w, pa[1] + pa_h)

    # indicatoare pe stâlp
    sp = ""
    bx, by = pa[0] - 4, pa[1] + pa_h
    svg, box = signpost(bx, by, "Piața Aurarilor", "aurar", 3, pole=44, side="left")
    sp += svg
    B.add("sign3:icon", *box)
    B.add("sign3:pole", bx - 3, box[3], bx + 3, by)
    mX, mY = P(300, 520)
    svg, box = signpost(mX, mY, "Strada Măcelarilor", "macelar", 1, pole=46)
    sp += svg
    B.add("sign1:icon", *box)
    B.add("sign1:pole", mX - 3, box[3], mX + 3, mY)
    # zona: inel punctat la baza stâlpului
    sp += (f'<ellipse cx="{mX:.1f}" cy="{mY+2:.1f}" rx="30" ry="9" fill="{PETROL}" fill-opacity="0.25" stroke="{PETROL}" '
           f'stroke-width="2" stroke-dasharray="5 4"/>')

    # busolă și mențiune
    CX, CY = MX + MW - 58, MY + 80
    comp = SM.compass(CX, CY, r=24)
    B.add("compass", CX - 46, CY - 48, CX + 46, CY + 46)

    # ---- caseta „poziție de verificat” ----
    QX, QY, QW, QH = MX + 22, 758, MW - 44, 152
    box_svg = (f'<rect x="{QX}" y="{QY}" width="{QW}" height="{QH}" rx="16" fill="#FBF7EE" fill-opacity="0.95" '
               f'stroke="{OCRU_D}" stroke-width="2" stroke-dasharray="9 6"/>\n')
    box_svg += qbadge(QX + 30, QY + 29, 13)
    ttl = "În centrul istoric – poziție de verificat pe o hartă reală"
    box_svg += text(QX + 54, QY + 36, ttl, 21, INK, "serif", 700)
    sub = "Nu sunt în Orașul de Sus: acolo, Măcelarilor e singura stradă cu nume de breaslă."
    box_svg += text(QX + 54, QY + 60, sub, 15, INK2, "sans", 400, True)
    assert QX + 54 + measure(sub, 15, "sans", 400, True) <= QX + QW - 16
    assert QX + 54 + measure(ttl, 21, "serif", 700) <= QX + QW - 16
    col_x = [QX + 30, QX + QW / 2 + 10]
    for i, (num, tool, kind, nm) in enumerate([(2, "brutar", "Strada", "Brutarilor"), (4, "barbier", "Strada", "Bărbierilor"),
                                               (5, "vopsitor", "Strada", "Vopsitorilor"), (6, "croitor", "Strada", "Croitorilor")]):
        cx = col_x[i % 2]; cy = QY + 74 + (i // 2) * 38
        svg, w = mini_plate(cx, cy, f"{kind} {nm}", tool, num, size=16, h=32)
        box_svg += svg
        box_svg += qbadge(cx + w + 16, cy + 16, 10)
        assert cx + w + 30 <= (col_x[1] - 10 if i % 2 == 0 else QX + QW - 10), nm
    assert QY + 74 + 38 + 32 <= QY + QH - 6
    B.add("verifbox", QX, QY, QX + QW, QY + QH)

    # ---- legendă (2 × 2) și mențiunea de scară ----
    GX0, GY0, GW, GH = MX + 22, 922, 636, 66
    leg = (f'<rect x="{GX0}" y="{GY0}" width="{GW}" height="{GH}" rx="12" fill="#FBF7EE" fill-opacity="0.95" '
           f'stroke="#C9B48E" stroke-width="1.5"/>\n')
    items = [
        ("plate", "stradă cu nume de meserie"),
        ("orient", "poziție orientativă"),
        ("zona", "se știe doar zona"),
        ("verif", "de verificat pe o hartă reală"),
    ]
    for i, (kd, lab) in enumerate(items):
        ix = GX0 + 114 + (i % 2) * 262; iy = GY0 + 24 + (i // 2) * 28
        if kd == "plate":
            leg += (f'<rect x="{ix}" y="{iy-9}" width="30" height="16" rx="3" fill="{PLATE}" stroke="{PLATE_D}"/>'
                    f'<rect x="{ix+2}" y="{iy-7}" width="26" height="12" rx="2" fill="none" stroke="{VAR}" stroke-width="1"/>')
        elif kd == "orient":
            leg += f'<rect x="{ix+3}" y="{iy-9}" width="24" height="17" rx="3" fill="#FBF7EE" stroke="{OCRU_D}" stroke-width="2" stroke-dasharray="4 3"/>'
        elif kd == "zona":
            leg += f'<ellipse cx="{ix+15}" cy="{iy-1}" rx="14" ry="6" fill="{PETROL}" fill-opacity="0.25" stroke="{PETROL}" stroke-width="2" stroke-dasharray="4 3"/>'
        else:
            leg += qbadge(ix + 15, iy - 1, 9)
        leg += text(ix + 40, iy + 4, lab, 14, INK2, "sans", 500)
        assert ix + 40 + measure(lab, 14, "sans", 500) <= (GX0 + 114 + 262 - 10 if i % 2 == 0 else GX0 + GW - 8), lab
    leg += text(GX0 + 18, GY0 + 40, "Legendă", 19, INK, "serif", 700)
    B.add("legend", GX0, GY0, GX0 + GW, GY0 + GH)
    nt = text(MX + MW - 24, 962, "Hartă schematică, poziții", 16, INK2, "sans", 400, True, "end")
    nt += text(MX + MW - 24, 983, "aproximative – nu la scară", 16, INK2, "sans", 400, True, "end")
    B.add("note", MX + MW - 28 - measure("Hartă schematică, poziții", 16, "sans", 400, True), 946, MX + MW - 20, 988)
    box_svg += leg

    # ---- etichete ----
    labels = ""

    def put(name, x, y, lines, cands, **kw):
        nonlocal labels
        svg, p_ = place(B, name, x, y, lines, cands, BOUNDS, **kw)
        labels += svg
        return p_

    put("ODS", *P(640, 590), [L("ORAȘUL DE SUS", 22, C_SUS, 700, False, "serif")], [(0, 0, "middle"), (40, 0, "middle"), (-40, 0, "middle")])
    put("ODJ", *P(780, 165), [L("ORAȘUL DE JOS", 22, C_JOS, 700, False, "serif")], [(0, 0, "middle"), (30, 10, "middle"), (-30, 10, "middle")])
    pmx, pmy = P(x1 + w1, y1 + h1 / 2)
    put("PMica", pmx, pmy, [L("Piața Mică", 16, INK, 700)], [(8, 6, "start"), (8, -10, "start"), (8, 22, "start")], hw=4)
    put("Huet", hx, hy, [L("Piața Huet", 14, INK2, 700)], [(0, 36, "middle"), (-24, 6, "end")], hw=4)
    for (gx, gy), nm in gates:
        X, Y = P(gx, gy)
        cand = {"Poarta Cisnădiei": [(0, 34, "middle"), (-22, 6, "end"), (22, 6, "start")],
                "Poarta Ocnei": [(22, 6, "start"), (0, 32, "middle"), (-22, 6, "end")],
                "Poarta Turnului": [(0, 34, "middle"), (22, 6, "start"), (0, -20, "middle")],
                "Poarta Elisabeta": [(0, 34, "middle"), (-22, 6, "end"), (0, -20, "middle")]}[nm]
        put(nm, X, Y, [L(nm, 14, INK2, 600, True)], cand, hw=4)
    # Pasajul Aurarilor
    mpx, mpy = (a0[0] + a1[0]) / 2, (a0[1] + a1[1]) / 2
    put("pasajL", mpx, mpy, [L("Pasajul Aurarilor", 14, INK, 700), L("trepte care coboară", 13, INK2, 400, True),
                             L("din Piața Mică", 13, INK2, 400, True)],
        [(42, -24, "start"), (40, -14, "start"), (46, -34, "start"), (14, 0, "start")], hw=4)
    # Piața Aurarilor
    put("PAurL", pa[0], pa[1] + pa_h / 2, [L("Piața Aurarilor", 17, INK, 700, False, "serif"), L("în Orașul de Jos,", 13, INK2, 400, True),
                                           L("la capătul pasajului", 13, INK2, 400, True), L("poziție orientativă", 13, OCRU_D, 600, True)],
        [(-14, 20, "end"), (-14, 28, "end"), (-24, 20, "end"), (pa_w + 12, 30, "start")])
    # Strada Măcelarilor
    put("macL", mX, mY, [L("Strada Măcelarilor", 18, INK, 700, False, "serif"), L("singura cu nume de breaslă", 14, C_SUS, 600, True),
                         L("în Orașul de Sus", 14, C_SUS, 600, True), L("locul exact: de verificat", 13, OCRU_D, 600, True)],
        [(0, 34, "middle"), (40, 34, "start"), (-40, 34, "end"), (0, -100, "middle")])

    body += icons + steps + sp + labels + box_svg + comp + nt
    body += "</g>\n"
    s += body
    s += f'<rect x="{MX}" y="{MY}" width="{MW}" height="{MH}" rx="22" fill="none" stroke="#2A3845" stroke-width="2"/>\n'

    # ---------------- coloana dreaptă: șase plăcuțe ilustrate ----------------
    RX, CW, CH, GX, GY = 1000, 412, 246, 16, 16
    col = ""
    for i, (num, tool, kind, nm, yr, fact, src, draw, (st, stt)) in enumerate(STREETS):
        x0 = RX + (i % 2) * (CW + GX); y0 = 230 + (i // 2) * (CH + GY)
        col += card(x0, y0, CW, CH, rx=18)
        col += plate(x0 + 18, y0 + 14, CW - 36, 78, kind, nm, tool, num, name_size=29)
        col += text(x0 + 22, y0 + 114, draw, 14, CEATA, "sans", 400, True)
        p, yn, nl = para_prefix(x0 + 22, y0 + 142, yr, fact, 17, CW - 44, lh=21)
        assert nl <= 3, ("fact lines", nm, nl)
        col += p
        ys = yn + 3
        col += text(x0 + 22, ys, src, 13, CEATA, "sans", 400, True)
        assert measure(src, 13, "sans", 400, True) <= CW - 44, ("src", nm)
        yst = y0 + CH - 16
        col += status_icon(st, x0 + 32, yst - 5)
        scol = {"zona": "#5FC3CB", "orient": OCRU, "verif": "#B9A7F5"}[st]
        col += text(x0 + 50, yst, stt, 14, scol, "sans", 600)
        assert x0 + 50 + measure(stt, 14, "sans", 600) <= x0 + CW - 14, ("status", nm)
    s += col

    s += frame_foot()
    write(OUT, s)
    print(OUT)
    bad = check(B, BOUNDS)
    print("overlaps:", len(bad))
