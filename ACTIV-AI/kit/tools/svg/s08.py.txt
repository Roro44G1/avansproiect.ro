    out += f'<path d="{d}" fill="none" stroke="{color}" stroke-width="{w}" stroke-linejoin="round" filter="url(#rough)"/>\n'
    out += f'<path d="{d}" fill="none" stroke="{PARCH}" stroke-width="{w*0.28:.1f}" stroke-dasharray="7 9" opacity="0.9"/>\n'
    return out

def tower(x, y, name, dx=14, dy=6, anchor="start", sub=None, suboff=18):
    out = f'<g filter="url(#shs)"><rect x="{x-11}" y="{y-11}" width="22" height="22" rx="3" fill="{OCRU}" stroke="#3B3326" stroke-width="2"/>'
    out += f'<polygon points="{x-13},{y-11} {x},{y-24} {x+13},{y-11}" fill="{TIGLA}" stroke="#3B3326" stroke-width="1.5"/></g>\n'
    out += text(x + dx, y + dy, name, 17, "#3B3326", "sans", 700, anchor=anchor)
    if sub:
        out += text(x + dx, y + dy + suboff, sub, 15, "#5A4D38", "sans", 400, True, anchor=anchor)
    return out
