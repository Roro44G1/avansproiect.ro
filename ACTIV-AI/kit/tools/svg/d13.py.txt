def flower5(cx, cy, r, col, ctr=OCRU, n=5, rot=-90):
    o = ""
    for k in range(n):
        a = math.radians(rot + k * 360 / n)
        o += f'<circle cx="{cx + r*0.62*math.cos(a):.1f}" cy="{cy + r*0.62*math.sin(a):.1f}" r="{r*0.5:.1f}" fill="{col}"/>'
    o += f'<circle cx="{cx}" cy="{cy}" r="{r*0.32:.1f}" fill="{ctr}"/>\n'
    return o


def leaf(cx, cy, l, w, ang, col=GREEN):
    return (f'<ellipse cx="0" cy="0" rx="{l/2:.1f}" ry="{w/2:.1f}" fill="{col}" '
            f'transform="translate({cx:.1f},{cy:.1f}) rotate({ang})"/>\n')


def ic_snow(cx, cy):
    """Munte cu zăpadă + fulg (Păltiniș)."""
    o = poly([(cx - 22, cy + 16), (cx - 6, cy - 8), (cx + 4, cy + 4), (cx + 12, cy - 4), (cx + 24, cy + 16)], fill="#4E6A86")
    o += poly([(cx - 6, cy - 8), (cx - 12, cy + 1), (cx - 8, cy - 1), (cx - 4, cy + 2), (cx - 1, cy - 1)], fill=VAR)
    o += poly([(cx + 12, cy - 4), (cx + 8, cy + 1), (cx + 12, cy + 0), (cx + 16, cy + 2)], fill=VAR)
    fx, fy, fr = cx + 9, cy - 15, 8
    for k in range(3):
        a = math.radians(k * 60)
        o += (f'<line x1="{fx - fr*math.cos(a):.1f}" y1="{fy - fr*math.sin(a):.1f}" x2="{fx + fr*math.cos(a):.1f}" '
              f'y2="{fy + fr*math.sin(a):.1f}" stroke="{ICE}" stroke-width="2.4" stroke-linecap="round"/>')
    return o + "\n"

def sheep(cx, cy, sc=1.0, face="#7A6656"):
    o = f'<g transform="translate({cx:.1f},{cy:.1f}) scale({sc})">'
    o += f'<rect x="-9" y="6" width="3.5" height="9" fill="{face}"/><rect x="5" y="6" width="3.5" height="9" fill="{face}"/>'
    for dx, dy, r in [(-8, 0, 7), (0, -4, 8), (8, 0, 7), (0, 4, 7), (-4, 2, 6), (5, 3, 6)]:
        o += f'<circle cx="{dx}" cy="{dy}" r="{r}" fill="{VAR}"/>'
    o += f'<ellipse cx="14" cy="-3" rx="5" ry="6.5" fill="{face}" transform="rotate(-15 14 -3)"/>'
    o += f'<ellipse cx="11" cy="-7" rx="3" ry="1.6" fill="{face}" transform="rotate(-30 11 -7)"/>'
    return o + "</g>\n"


def arrow_v(x, y0, y1, col=OCRU):
    up = y1 < y0
    o = f'<line x1="{x}" y1="{y0}" x2="{x}" y2="{y1 + (6 if up else -6)}" stroke="{col}" stroke-width="3"/>'
    o += poly([(x - 5, y1 + (7 if up else -7)), (x, y1), (x + 5, y1 + (7 if up else -7))], fill=col)
    return o

def item_block(x, y, kind, icon, title, expl):
    """Returnează (svg, înălțime) pentru un element al lunii."""
    lines = wrap(expl, TXT_W, 15)
    assert len(lines) <= 3, (title, lines)
    assert measure(title, 18, "sans", 700) <= TXT_W, title
    h = max(60, 24 + 19 * len(lines) + 4)
    o = f'<circle cx="{x+44}" cy="{y+30}" r="27" fill="{DISC}" stroke="{TYPE[kind]}" stroke-width="2.5"/>\n'

def mx(c, frac):

def bar(r, c0, f0, c1, f1, col, label, txtcol=BG, left="round", right="round", icon=None):
    y = ROWY[r] + CH - LANE + 4
    h = 26
    x0, x1 = mx(c0, f0), mx(c1, f1)
    o = f'<rect x="{x0:.1f}" y="{y}" width="{x1-x0:.1f}" height="{h}" rx="13" fill="{col}"/>\n'
    if left == "flat":   # continuă din rândul de sus / de jos: capăt drept, cu un chevron discret
        o += f'<rect x="{x0:.1f}" y="{y}" width="20" height="{h}" fill="{col}"/>\n'
        o += poly([(x0, y), (x0 + 7, y + h / 2), (x0, y + h)], fill=BG, opacity="0.35")
    if right == "arrow":
        o += f'<rect x="{x1-20:.1f}" y="{y}" width="20" height="{h}" fill="{col}"/>\n'
        o += poly([(x1 - 1, y - 4), (x1 + 14, y + h / 2), (x1 - 1, y + h + 4)], fill=col)
    tx = x0 + 16
    if icon:
        o += icon(x0 + 24, y + h / 2)
        tx = x0 + 44
    assert tx + measure(label, 15, "sans", 700) <= x1 - 10, label
    o += text(tx, y + 18, label, 15, txtcol, "sans", 700)
    return o


def mini_flake(cx, cy, col=BG):
    o = ""
    for k in range(3):
        a = math.radians(k * 60 + 90)
        o += (f'<line x1="{cx - 8*math.cos(a):.1f}" y1="{cy - 8*math.sin(a):.1f}" x2="{cx + 8*math.cos(a):.1f}" '
              f'y2="{cy + 8*math.sin(a):.1f}" stroke="{col}" stroke-width="2.2" stroke-linecap="round"/>')
    return o + "\n"
