def ic_star(cx, cy, col, k=1.0):
    import math
    pts = []
    for j in range(10):
        r = 18 if j % 2 == 0 else 7.5
        a = -math.pi / 2 + j * math.pi / 5
        pts.append((r * math.cos(a), r * math.sin(a) + 1))
    o = f'<g transform="translate({cx},{cy}) scale({k})">' + poly(pts, fill=col) + "</g>\n"
    return o
