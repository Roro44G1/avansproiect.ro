def load(p):
    d=json.load(open(f'{R}/{p}')); base=os.path.dirname(p)
    for it in d:
        fs=[]
        for f in it.get('fisiere',[]):
            fs.append(f if '/' in f else f'{base}/{f}')
        it['fisiere']=fs
        for k in ('audio','tts_text'):
            if it.get(k) and '/' not in it[k]: it[k]=f'{base}/{it[k]}'
    return d
