def still(img,out,d=10):
    run(['ffmpeg','-y','-v','error','-loop','1','-t',str(d),'-i',img,'-f','lavfi','-t',str(d),'-i','anullsrc=r=44100:cl=stereo',
         '-vf',f'scale=1920:1080:force_original_aspect_ratio=decrease,pad=1920:1080:(ow-iw)/2:(oh-ih)/2:color=0x0F1720,fps=25,fade=t=in:st=0:d=0.6,fade=t=out:st={d-0.6}:d=0.6,format=yuv420p',
         '-c:v','libx264','-preset','veryfast','-crf','21','-c:a','aac','-b:a','128k','-shortest',out])
def vid(src,out):
    run(['ffmpeg','-y','-v','error','-i',src,'-vf','scale=1920:1080:force_original_aspect_ratio=decrease,pad=1920:1080:(ow-iw)/2:(oh-ih)/2:color=0x0F1720,fps=25,format=yuv420p',
         '-c:v','libx264','-preset','veryfast','-crf','21','-c:a','aac','-b:a','128k','-ar','44100','-ac','2',out])
