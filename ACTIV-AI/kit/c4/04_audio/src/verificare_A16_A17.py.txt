"""Verificare tehnică pentru A16 (peisaj sonor) și A17 (cântec de leagăn – instrumental demo).

Pentru fiecare MP3: ffprobe (format, durată, debit), ffmpeg ebur128 (loudness integrat, LRA,
true peak, vârf de eșantion), eșantioane la limită, liniște la capete, loudness pe secțiuni,
spectrogramă (axă logaritmică) și formă de undă cu loudness pe termen scurt.
A16: pentru fiecare eveniment sonor (bufniță, mierlă, pițigoi, presură, ciocănitoare, pași,
gaiță, veveriță, corb) – energia din banda lui de frecvență în fereastra evenimentului,
comparată cu mediana aceleiași benzi pe tot fișierul (în dB): arată că evenimentul se aude
peste fond în mixul final.
A17: verificarea melodiei-ghid – pentru fiecare notă a solistului, înălțimea dominantă în
registrul vocii, comparată cu nota scrisă.
Ambele .md: ortografie (hunspell ro_RO) și ș/ț cu sedilă.

Rulare: python3 verificare_A16_A17.py -> ../_verificare/raport_A16_A17.json + PNG-uri
"""

def ffprobe(path):
    r = subprocess.run(["ffprobe", "-v", "error", "-show_format", "-show_streams", "-of", "json", path],
                       capture_output=True, text=True, check=True)
    j = json.loads(r.stdout)
    st = j["streams"][0]
    fm = j["format"]
    return {"codec": st["codec_name"], "sample_rate_hz": int(st["sample_rate"]), "canale": st["channels"],
            "bitrate_kbps": round(int(st.get("bit_rate", fm["bit_rate"])) / 1000),
            "durata_s": round(float(fm["duration"]), 2),
            "titlu_id3": fm.get("tags", {}).get("title", "")}


def ebur128(path):
    err = subprocess.run(["ffmpeg", "-nostats", "-i", path, "-filter_complex", "ebur128=peak=true+sample",
                          "-f", "null", "-"], capture_output=True, text=True).stderr
    summ = err[err.rfind("Summary:"):]
    i = float(re.search(r"I:\s+(-?[\d.]+) LUFS", summ).group(1))
    lra = float(re.search(r"LRA:\s+(-?[\d.]+) LU", summ).group(1))
    tp = float(re.search(r"True peak:\s+Peak:\s+(-?[\d.]+) dBFS", summ).group(1))
    sp = float(re.search(r"Sample peak:\s+Peak:\s+(-?[\d.]+) dBFS", summ).group(1))
    return {"loudness_integrat_lufs": i, "lra_lu": lra, "true_peak_dbtp": tp, "sample_peak_dbfs": sp}


def decode(path):
    raw = subprocess.run(["ffmpeg", "-v", "error", "-i", path, "-f", "f32le", "-ac", "2", "-ar", str(SR), "-"],
                         capture_output=True, check=True).stdout
    return np.frombuffer(raw, dtype=np.float32).reshape(-1, 2).T.astype(np.float64)


def short_term(x, win=3.0, hop=0.1):
    y = x
    for b, a in _KW:
        y = signal.lfilter(b, a, y, axis=-1)
    p = (y ** 2).sum(axis=0)
    w, h = int(win * SR), int(hop * SR)
    cs = np.concatenate([[0], np.cumsum(p)])
    st = np.arange(0, len(p) - w, h)
    z = (cs[st + w] - cs[st]) / w
    return (st + w / 2) / SR, -0.691 + 10 * np.log10(z + 1e-20)


def plots(x, marks, title, stem, band=None):
    mono = x.mean(axis=0)
    t_end = len(mono) / SR
    sec = list(marks.items())
    f, t, Z = signal.spectrogram(mono, SR, window="hann", nperseg=4096, noverlap=4096 - 1024, mode="magnitude")
    Zdb = 20 * np.log10(Z + 1e-9)
    Zdb -= Zdb.max()
    fig, ax = plt.subplots(figsize=(16, 7), dpi=110)
    keep = (f >= 30) & (f <= 18000)
    pc = ax.pcolormesh(t, f[keep], Zdb[keep], shading="auto", cmap="magma", vmin=-90, vmax=0)
    ax.set_yscale("log")
    ax.set_ylim(30, 18000)
    ax.set_yticks([50, 100, 200, 400, 800, 1600, 3200, 6400, 12800])
    ax.set_yticklabels(["50", "100", "200", "400", "800", "1,6k", "3,2k", "6,4k", "12,8k"])
    for k, (nm, t0) in enumerate(sec):
        ax.axvline(t0, color="#E2A62B", lw=1.2, alpha=0.9)
        ax.text(t0 + 0.3, [14500, 10500, 7600][k % 3], nm, color="#F3EFE6", fontsize=8.5,
                bbox=dict(facecolor="#0F1720", alpha=0.75, edgecolor="none", pad=2))
    if band:
        lo, hi, lab = band
        ax.axhspan(lo, hi, color="#12A594", alpha=0.12)
        ax.text(0.4, hi * 1.06, lab, color="#F3EFE6", fontsize=9,
                bbox=dict(facecolor="#12A594", alpha=0.8, edgecolor="none", pad=2))
    ax.set_xlim(0, t_end)
    ax.set_xlabel("timp (s)")
    ax.set_ylabel("frecvență (Hz, scară logaritmică)")
    ax.set_title(title)
    fig.colorbar(pc, ax=ax, label="dB (relativ la maxim)")
    fig.tight_layout()
    p1 = os.path.join(VER, f"{stem}_spectrograma.png")
    fig.savefig(p1)
    plt.close(fig)
    fig, (a1, a2) = plt.subplots(2, 1, figsize=(16, 7), dpi=110, sharex=True)
    tt = np.arange(len(mono)) / SR
    step = 200
    a1.plot(tt[::step], x[0, ::step], lw=0.4, color="#0E7C86", label="stânga")
    a1.plot(tt[::step], -x[1, ::step], lw=0.4, color="#6E56CF", alpha=0.7, label="dreapta (inversat)")
    a1.axhline(10 ** (-1 / 20), color="#B9583A", ls="--", lw=0.8)
    a1.axhline(-10 ** (-1 / 20), color="#B9583A", ls="--", lw=0.8)
    a1.set_ylim(-1.05, 1.05)
    a1.set_ylabel("amplitudine")
    a1.legend(loc="lower right", fontsize=8)
    ts, st = short_term(x)
    a2.plot(ts, st, color="#E2A62B", lw=1.4, label="loudness pe termen scurt (3 s)")
    a2.axhline(-14, color="#12A594", ls="--", lw=1, label="țintă −14 LUFS (integrat)")
    a2.set_ylim(-40, -6)
    a2.set_ylabel("LUFS")
    a2.set_xlabel("timp (s)")
    a2.legend(loc="lower right", fontsize=8)
    for ax in (a1, a2):
        for nm, t0 in sec:
            ax.axvline(t0, color="#A9B4C0", lw=0.8, alpha=0.8)
    for k, (nm, t0) in enumerate(sec):
        a1.text(t0 + 0.25, 0.92 if k % 2 == 0 else 0.78, nm, fontsize=8, color="#1B2733")
    a1.set_title(f"{stem.split('_')[0]} – formă de undă (liniile roșii: −1 dBFS) și loudness pe termen scurt")
    fig.tight_layout()
    p2 = os.path.join(VER, f"{stem}_forma_unda.png")
    fig.savefig(p2)
    plt.close(fig)
    return p1, p2


def band_db(mono, lo, hi):
    sos = signal.butter(4, [lo, hi], btype="bandpass", fs=SR, output="sos")
    y = signal.sosfilt(sos, mono)
    return y


def event_check(x, events):
    """Energia benzii evenimentului în fereastra lui vs. mediana benzii (ferestre de 0,25 s) pe tot fișierul."""
    mono = x.mean(axis=0)
    out = []
    cache = {}
    for e in events:
        lo, hi = e["banda_hz"]
        key = (lo, hi)
        if key not in cache:
            y = band_db(mono, lo, hi)
            w = int(0.25 * SR)
            m = len(y) // w
            blocks = np.sqrt(np.mean(y[: m * w].reshape(m, w) ** 2, axis=1)) + 1e-12
            cache[key] = (y, 20 * np.log10(np.median(blocks)))
        y, med = cache[key]
        a = int(e["t"] * SR)
        b = int((e["t"] + max(0.25, e["durata_s"])) * SR)
        r = 20 * np.log10(np.sqrt(np.mean(y[a:b] ** 2)) + 1e-12)
        out.append({"eveniment": e["nume"], "t_s": e["t"], "banda_hz": e["banda_hz"],
                    "peste_mediana_benzii_db": round(r - med, 1)})
    return out


def melody_check(x, notes, lo="C4", hi="F5"):
    mono = x.mean(axis=0)
    cands = list(range(midi(lo), midi(hi) + 1))
    ok, bad = 0, []
    for nt in notes:
        a = int((nt["t"] + 0.04) * SR)
        b = int((nt["t"] + min(nt["d"] - 0.03, 0.4)) * SR)
        seg = mono[a:b] * np.hanning(b - a)
        N = 1 << 16
        sp = np.abs(np.fft.rfft(seg, N))
        fr = np.fft.rfftfreq(N, 1 / SR)

        def band(f):
            m = (fr > f * 2 ** (-0.3 / 12)) & (fr < f * 2 ** (0.3 / 12))
            return sp[m].max() if np.any(m) else 0.0

        score = {m: band(mhz(m)) + 0.8 * band(2 * mhz(m)) + 0.6 * band(3 * mhz(m)) for m in cands}
        best = max(score, key=score.get)
        if best == midi(nt["nota"]):
            ok += 1
        else:
            bad.append({"t": nt["t"], "scris": nt["nota"], "detectat_midi": best})
    return {"note_verificate": len(notes), "note_corecte": ok,
            "procent": round(100 * ok / max(1, len(notes)), 1), "nepotriviri": bad}


def hunspell(path):
    txt = open(path, encoding="utf-8").read()
    txt = re.sub(r"`[^`]*`", "", txt)
    r = subprocess.run(["hunspell", "-d", "ro_RO", "-i", "utf-8", "-l"], input=txt, capture_output=True, text=True)
    words = sorted(set(r.stdout.split()))
    sed = len(re.findall("[şţŞŢ]", txt))
    return words, sed


def common(path, marks, title, stem, band=None, dur_range=(60, 100)):
    probe = ffprobe(path)
    loud = ebur128(path)
    x = decode(path)
    env = np.abs(x).max(axis=0)
    nz = np.nonzero(env > 10 ** (-60 / 20))[0]
    names = list(marks.keys())
    tt = list(marks.values()) + [x.shape[1] / SR]
    per_sec = {nm: round(lufs(x[:, int(tt[k] * SR):int(tt[k + 1] * SR)]), 1) for k, nm in enumerate(names)}
    p1, p2 = plots(x, marks, title.format(d=probe["durata_s"], i=loud["loudness_integrat_lufs"]), stem, band)
    rep = {
        "fisier": "04_audio/" + os.path.basename(path),
        "ffprobe": probe,
        "ebur128": loud,
        "esantioane_la_limita_0dBFS": int(np.sum(np.abs(x) >= 0.999)),
        "inceput_sunet_s": round(nz[0] / SR, 3), "sfarsit_sunet_s": round(nz[-1] / SR, 3),
        "loudness_pe_sectiuni_lufs": per_sec,
        "conditii": {
            f"durata_{dur_range[0]}_{dur_range[1]}_s": dur_range[0] <= probe["durata_s"] <= dur_range[1],
            "loudness_-14_plus_minus_0_5": abs(loud["loudness_integrat_lufs"] + 14) <= 0.5,
            "true_peak_sub_-1_dBTP": loud["true_peak_dbtp"] <= -1.0,
            "mp3_192_kbps": probe["codec"] == "mp3" and probe["bitrate_kbps"] == 192,
            "esantionare_44100_hz": probe["sample_rate_hz"] == 44100,
        },
        "grafice": ["04_audio/_verificare/" + os.path.basename(p1), "04_audio/_verificare/" + os.path.basename(p2)],
    }
    return rep, x


if __name__ == "__main__":
    rep = {}
    a16 = os.path.join(ROOT, "A16_dimineata_in_dumbrava.mp3")
    if os.path.exists(a16):
        m = json.load(open(os.path.join(HERE, "A16_marcaje.json"), encoding="utf-8"))
        r, x = common(a16, m["marcaje_s"], "A16 „Între bufniță și soare” – spectrograma MP3 ({d:.1f} s, {i:.1f} LUFS)",
                      "A16_dimineata_in_dumbrava", dur_range=(70, 100))
        r["evenimente_in_mix"] = event_check(x, m["evenimente"])
        vals = [e["peste_mediana_benzii_db"] for e in r["evenimente_in_mix"]]
        r["conditii"]["toate_evenimentele_peste_fond_cu_minim_3_dB"] = bool(min(vals) >= 3.0)
        rep["A16"] = r
    a17 = os.path.join(ROOT, "A17_cantec_de_leagan_cibin.mp3")
    if os.path.exists(a17):
        m = json.load(open(os.path.join(HERE, "A17_marcaje.json"), encoding="utf-8"))
        r, x = common(a17, m["marcaje_s"], "A17 „Optzeci și doi de kilometri” – spectrograma MP3 ({d:.1f} s, {i:.1f} LUFS)",
                      "A17_cantec_de_leagan_cibin", band=(mhz(midi("E4")) * 0.97, mhz(midi("D5")) * 1.03,
                                                          "registrul melodiei-ghid (E4–D5)"), dur_range=(60, 90))
        r["melodia_ghid"] = melody_check(x, m["note"])
        r["conditii"]["melodia_ghid_peste_95_la_suta"] = r["melodia_ghid"]["procent"] >= 95
        rep["A17"] = r
    rep["ortografie_hunspell_ro_RO"] = {}
    for fn in ("A16_dimineata_in_dumbrava.md", "A17_cantec_de_leagan_cibin.md", "_fapte_noi_A16_A17.md"):
        p = os.path.join(ROOT, fn)
        if os.path.exists(p):
            w, sed = hunspell(p)
            rep["ortografie_hunspell_ro_RO"][fn] = {"cuvinte_nerecunoscute": w, "s_t_cu_sedila": sed}
    out = os.path.join(VER, "raport_A16_A17.json")
    json.dump(rep, open(out, "w", encoding="utf-8"), ensure_ascii=False, indent=2)
    print(json.dumps(rep, ensure_ascii=False, indent=2))
