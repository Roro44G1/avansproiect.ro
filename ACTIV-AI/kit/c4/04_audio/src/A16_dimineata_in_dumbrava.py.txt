# evenimente punctuale (pentru verificarea benzilor de frecvență în mix)
EVENTS = []


def ev(name, t, lo, hi, dur):
    EVENTS.append(dict(nume=name, t=round(float(t), 3), banda_hz=[lo, hi], durata_s=round(float(dur), 3)))


# ------------------------------------------------------------------ utilitare de ton
def tone(fcurve, env, harm=(1.0,), phase0=0.0):
    """Ton cu frecvență variabilă (vector) și anvelopă; armonice opționale."""
    ph = 2 * np.pi * (phase0 + np.cumsum(fcurve) / SR)
    x = np.zeros(len(fcurve))
    for k, a in enumerate(harm, start=1):
        if a:
            x += a * np.sin(k * ph)
    return x * env


def smooth_env(n, att, rel, shape=1.0):
    """Anvelopă netedă: urcare sinusoidală, palier, coborâre sinusoidală."""
    e = np.ones(n)
    a, r = max(1, min(n // 2, ns(att))), max(1, min(n // 2, ns(rel)))
    e[:a] = np.sin(np.linspace(0, np.pi / 2, a)) ** 2
    e[n - r:] *= np.cos(np.linspace(0, np.pi / 2, r)) ** 2
    return e ** shape


def contour(n, pts):
    """Curbă de frecvență prin puncte (fracțiune din durată, Hz), interpolare cosinus."""
    u = np.linspace(0, 1, n)
    xs = np.array([p[0] for p in pts])
    ys = np.array([p[1] for p in pts])
    out = np.zeros(n)
    for k in range(len(xs) - 1):
        m = (u >= xs[k]) & (u <= xs[k + 1])
        w = (u[m] - xs[k]) / max(1e-9, xs[k + 1] - xs[k])
        w = 0.5 - 0.5 * np.cos(np.pi * w)
        out[m] = ys[k] * (1 - w) + ys[k + 1] * w
    return out


def distance(x, far):
    """far 0..1: filtrare trece-jos crescătoare (aerul absoarbe înaltele) și atenuare."""
    if far <= 0:
        return x
    return lp(x, 12000 - 8500 * far, order=2) * (1 - 0.45 * far)


# ------------------------------------------------------------------ vânt în coroane + foșnet
def wind_env(n):
    t = t_axis(n)
    base = 0.22 + 0.10 * np.clip((t - 30) / 40, 0, 1)
    gusts = [(11.0, 6.0, 0.35), (39.0, 7.0, 0.45), (61.0, 6.5, 0.4), (77.0, 11.0, 0.5)]
    e = base.copy()
    for tc, d, a in gusts:
        u = np.clip((t - tc) / d, 0, 1)
        e += a * np.sin(np.pi * u) ** 1.8
    return e


def wind_layer(rng):
    n = ns(DUR)
    env = wind_env(n)
    buf = stereo(DUR)
    for ch in range(2):
        nz = colored_noise(n, rng_of(1600 + ch), 1.2)
        body = bp(nz, 220, 1300, order=2) * (0.3 + env)
        hiss = bp(nz, 1300, 4000, order=2) * env ** 1.6 * 0.7
        # foșnetul frunzelor de stejar: zgomot înalt, modulat de „pâlpâiri” rapide și neregulate
        w = rng_of(1700 + ch).standard_normal(n)
        flick = np.abs(lp(rng_of(1800 + ch).standard_normal(n), 28, order=2))
        flick = np.minimum(flick / (np.percentile(flick, 99) + 1e-12), 1.0)
        leaves = bp(w, 2500, 9000, order=2) * flick ** 1.5 * env ** 1.4 * 1.1
        buf[ch] = body + hiss + leaves
    buf[1] = np.roll(buf[1], ns(0.013))
    return hp(buf, 150, order=2)


# ------------------------------------------------------------------ apa lacurilor
def water_layer(rng):
    n = ns(DUR)
    buf = stereo(DUR)
    # clipocit la mal: zgomot jos, cu „valuri” lente
    for ch in range(2):
        nz = colored_noise(n, rng_of(2100 + ch), 1.0)
        lap_env = smooth_random(n, rng_of(2200 + ch), 0.45, 0.0, 1.0) ** 2
        buf[ch] += lp(bp(nz, 180, 1100, order=2), 800) * (0.25 + lap_env)
    # pârâiașul care drenează lacurile: bule mici (frecvență care urcă, anvelopă scurtă)
    t = rng.uniform(0, 0.05)
    while t < DUR - 0.1:
        f0 = np.exp(rng.uniform(np.log(380), np.log(1700)))
        d = rng.uniform(0.012, 0.04)
        m = ns(d)
        tt = t_axis(m)
        f = f0 * (1 + rng.uniform(8, 25) * tt)
        x = np.sin(2 * np.pi * np.cumsum(f) / SR) * np.exp(-tt / (d / 3)) * (1 - np.exp(-tt / 0.0015))
        add(buf, x, t, gain=rng.uniform(0.15, 0.6), pan=rng.uniform(0.25, 0.85))
        t += rng.exponential(1 / 22.0)
    return hp(lp(buf, 4200, order=2), 90, order=2)


# ------------------------------------------------------------------ bufnița
def owl_hoot(rng, f0=305.0):
    """„Hu-hu” grav: prima silabă mai lungă și accentuată, a doua mai scurtă, puțin mai joasă."""
    parts = []
    for d, f, a in ((0.5, f0, 1.0), (0.32, f0 * 0.93, 0.7)):
        m = ns(d)
        fc = f * (1.02 - 0.05 * np.linspace(0, 1, m)) * (1 + 0.004 * np.sin(2 * np.pi * 6 * t_axis(m)))
        e = smooth_env(m, 0.09, 0.16)
        x = tone(fc, e, harm=(1.0, 0.12, 0.04))
        x += bp(rng.standard_normal(m), 200, 700) * e * 0.06
        parts.append(a * x)
    gap = np.zeros(ns(0.22))
    return np.concatenate([parts[0], gap, parts[1]])


def owl_layer(rng):
    buf = stereo(DUR)
    for t0, g, pan in ((2.5, 1.0, -0.55), (8.8, 0.9, -0.6), (15.2, 0.65, -0.65)):
        x = distance(owl_hoot(rng, 305 * rng.uniform(0.98, 1.02)), 0.35)
        add(buf, x, t0, gain=g, pan=pan)
        ev("bufniță", t0, 250, 350, 0.5)
    return buf


# ------------------------------------------------------------------ mierla
def blackbird_phrase(rng):
    """Frază fluierată, „jos” pentru o pasăre (aprox. 1,5–3 kHz), cu o mică încheiere ciripită."""
    notes = []
    for _ in range(rng.integers(4, 8)):
        d = rng.uniform(0.09, 0.26)
        kind = rng.integers(0, 4)
        a = rng.uniform(1500, 2600)
        if kind == 0:
            pts = [(0, a), (1, a * rng.uniform(1.15, 1.35))]
        elif kind == 1:
            pts = [(0, a * 1.2), (1, a * rng.uniform(0.8, 0.95))]
        elif kind == 2:
            pts = [(0, a), (0.5, a * 1.25), (1, a * 1.05)]
        else:
            pts = [(0, a), (1, a * 1.02)]
        m = ns(d)
        fc = contour(m, pts) * (1 + 0.012 * np.sin(2 * np.pi * rng.uniform(18, 30) * t_axis(m)))
        x = tone(fc, smooth_env(m, 0.015, 0.03), harm=(1.0, 0.18, 0.05))
        notes.append((x, rng.uniform(0.03, 0.09), 1.0))
    if rng.uniform() < 0.75:  # încheiere ciripită, înaltă și mai slabă
        for _ in range(rng.integers(3, 7)):
            d = rng.uniform(0.02, 0.045)
            m = ns(d)
            f1 = rng.uniform(4200, 6800)
            fc = contour(m, [(0, f1), (1, f1 * rng.uniform(0.75, 1.25))])
            notes.append((tone(fc, smooth_env(m, 0.004, 0.008), harm=(1.0, 0.1)), rng.uniform(0.01, 0.03), 0.35))
    out = []
    for x, gap, g in notes:
        out += [g * x, np.zeros(ns(gap))]
    return np.concatenate(out)


def blackbird_layer(rng):
    buf = stereo(DUR)
    birds = [  # (început, sfârșit, pauză medie între fraze, pan, depărtare)
        (17.0, DUR - 3.0, 6.5, -0.3, 0.25),
        (46.0, DUR - 4.0, 8.0, 0.65, 0.55),
    ]
    for t0, t1, gap, pan, far in birds:
        t = t0
        while t < t1:
            x = distance(blackbird_phrase(rng), far)
            dens = 0.7 if t > MARKS["Răsăritul: corul plin"] else 1.0
            add(buf, x, t, gain=rng.uniform(0.75, 1.0), pan=pan + rng.uniform(-0.08, 0.08))
            if far < 0.4:
                ev("mierlă", t, 1500, 3200, min(1.2, len(x) / SR))
            t += len(x) / SR + gap * dens * rng.uniform(0.6, 1.3)
    return buf


# ------------------------------------------------------------------ pițigoiul
def tit_song(rng):
    """Două note repetate (prima înaltă, a doua mai joasă) – ca „ți-ciu, ți-ciu”."""
    hi_f, lo_f = rng.uniform(5300, 6100), rng.uniform(3400, 3900)
    out = []
    for _ in range(rng.integers(3, 6)):
        m1, m2 = ns(0.075), ns(0.105)
        a = tone(contour(m1, [(0, hi_f * 1.04), (1, hi_f * 0.95)]), smooth_env(m1, 0.008, 0.02), harm=(1.0, 0.06))
        b = tone(contour(m2, [(0, lo_f * 0.97), (1, lo_f * 1.03)]), smooth_env(m2, 0.01, 0.03), harm=(1.0, 0.08))
        out += [a, np.zeros(ns(0.035)), b * 0.9, np.zeros(ns(0.11))]
    return np.concatenate(out)


def tit_layer(rng):
    buf = stereo(DUR)
    for t0, t1, gap, pan, far in ((24.0, DUR - 3, 5.0, 0.5, 0.2), (60.0, DUR - 4, 7.0, -0.75, 0.5)):
        t = t0
        while t < t1:
            x = distance(tit_song(rng), far)
            dens = 0.6 if t > MARKS["Răsăritul: corul plin"] else 1.0
            add(buf, x, t, gain=rng.uniform(0.8, 1.0), pan=pan)
            if far < 0.4:
                ev("pițigoi", t, 3300, 6300, len(x) / SR)
            t += len(x) / SR + gap * dens * rng.uniform(0.7, 1.3)
    return buf


# ------------------------------------------------------------------ presura
def bunting_song(rng):
    """Șir de note scurte, tot mai tari, urmate de două note mai lungi."""
    out = []
    k = rng.integers(7, 11)
    f = rng.uniform(5400, 6200)
    for i in range(k):
        m = ns(0.045)
        x = tone(contour(m, [(0, f * 1.08), (1, f * 0.9)]), smooth_env(m, 0.006, 0.015), harm=(1.0, 0.1))
        out += [x * (0.45 + 0.55 * i / (k - 1)), np.zeros(ns(0.075))]
    m = ns(0.22)
    out += [tone(contour(m, [(0, f * 1.05), (1, f * 1.0)]), smooth_env(m, 0.02, 0.05), harm=(1.0, 0.1)),
            np.zeros(ns(0.04))]
    m = ns(0.55)
    tt = t_axis(m)
    buzz = 1 + 0.25 * np.sin(2 * np.pi * 42 * tt)
    out.append(tone(contour(m, [(0, f * 0.78), (1, f * 0.74)]), smooth_env(m, 0.03, 0.15) * buzz,
                    harm=(1.0, 0.12)))
    return np.concatenate(out)


def bunting_layer(rng):
    buf = stereo(DUR)
    t = 30.5
    while t < DUR - 4:
        x = distance(bunting_song(rng), 0.45)
        add(buf, x, t, gain=rng.uniform(0.85, 1.0), pan=-0.65)
        ev("presură", t, 4000, 6800, len(x) / SR)
        t += len(x) / SR + rng.uniform(6.0, 9.5) * (0.75 if t > MARKS["Răsăritul: corul plin"] else 1.0)
    return buf


# ------------------------------------------------------------------ ciocănitoarea
def wood_strike(rng):
    fr = [380 * rng.uniform(0.97, 1.03), 905, 1640, 2750]
    x = modal(fr, [1.0, 0.6, 0.35, 0.2], [0.11, 0.07, 0.045, 0.03], 0.16, rng=rng, attack=0.0004)
    m = len(x)
    x += bp(rng.standard_normal(m), 1500, 7000) * env_exp(m, 0.004) * 0.5
    return x / (np.max(np.abs(x)) + 1e-12)


def drum_roll(rng, k=None):
    k = k or int(rng.integers(13, 18))
    m = ns(0.06 * k + 0.4)
    out = np.zeros(m)
    t = 0.0
    for i in range(k):
        g = min(1.0, 0.55 + 0.15 * i) * (1 - 0.5 * max(0, i - (k - 5)) / 5)
        s = wood_strike(rng)
        i0 = ns(t)
        out[i0:i0 + len(s)] += g * s[: m - i0]
        t += 0.058 - 0.0012 * i + rng.uniform(-0.002, 0.002)
    return out


def woodpecker_layer(rng):
    buf = stereo(DUR)
    for t0, pan, far in ((36.0, 0.35, 0.3), (41.5, 0.35, 0.3), (57.5, 0.4, 0.35), (80.5, 0.45, 0.45)):
        x = distance(drum_roll(rng), far)
        add(buf, x, t0, gain=1.0, pan=pan)
        ev("ciocănitoare", t0, 350, 1000, 0.9)
    return buf


# ------------------------------------------------------------------ pași pe frunze uscate
def crunch(rng, length=0.12, density=420):
    """Strivirea frunzelor uscate: sute de micro-pocnete filtrate, într-un nor scurt."""
    m = ns(length + 0.02)
    x = np.zeros(m)
    k = rng.poisson(density * length)
    times = rng.exponential(length / 3, k)
    for tt in times[times < length]:
        g = ns(rng.uniform(0.0006, 0.003))
        i0 = ns(tt)
        if i0 + g >= m:
            continue
        x[i0:i0 + g] += rng.standard_normal(g) * rng.uniform(0.2, 1.0) * np.hanning(g)
    x = bp(x, 900, 9000, order=2)
    thump = lp(rng.standard_normal(m), 220) * env_ar(m, 0.002, 0.05) * 0.6
    return x + thump


def step(rng):
    a = crunch(rng, 0.10, 380)
    b = crunch(rng, 0.13, 460)
    out = np.zeros(len(a) + ns(0.07) + len(b))
    out[: len(a)] += 0.8 * a
    out[ns(0.07): ns(0.07) + len(b)] += b
    return out / (np.max(np.abs(out)) + 1e-12)


def steps_layer(rng):
    buf = stereo(DUR)
    t0, t1 = 44.0, 56.5
    t = t0
    while t < t1:
        u = (t - t0) / (t1 - t0)
        amp = 0.3 + 0.7 * np.sin(np.pi * u) ** 1.2
        x = step(rng)
        if u > 0.7:
            x = lp(x, 6000 - 4000 * (u - 0.7) / 0.3)
        add(buf, x, t, gain=amp, pan=-0.8 + 1.6 * u)
        t += 0.62 + rng.uniform(-0.04, 0.04)
    ev("pași pe frunze", 49.5, 2000, 8000, 1.5)
    return buf


# ------------------------------------------------------------------ gaița
def jay_screech(rng):
    d = rng.uniform(0.38, 0.5)
    m = ns(d)
    tt = t_axis(m)
    rasp = 0.55 + 0.45 * np.sign(np.sin(2 * np.pi * rng.uniform(75, 95) * tt + rng.uniform(0, 6)))
    rasp = lp(rasp, 600)
    nz = rng.standard_normal(m)
    body = bp(nz, 1100, 5200, order=2) + 0.6 * bp(nz, 2200, 3200, order=2)
    f0 = 980 * (1.05 - 0.15 * tt / d)
    harm = tone(f0, np.ones(m), harm=(0.5, 0.6, 0.45, 0.3, 0.2))
    x = (body * 1.2 + 0.35 * harm) * rasp * smooth_env(m, 0.02, 0.12)
    return x / (np.max(np.abs(x)) + 1e-12)


def jay_layer(rng):
    buf = stereo(DUR)
    for t0, pan in ((50.5, 0.55), (51.45, 0.6), (52.6, 0.62), (71.0, -0.4)):
        x = distance(jay_screech(rng), 0.2 if t0 < 60 else 0.5)
        add(buf, x, t0, gain=1.0 if t0 < 60 else 0.6, pan=pan)
        ev("gaiță", t0, 1100, 5200, 0.4)
    return buf


# ------------------------------------------------------------------ veverița
def scratch_burst(rng, k):
    m = ns(0.03 * k + 0.05)
    x = np.zeros(m)
    for i in range(k):
        g = ns(rng.uniform(0.008, 0.02))
        i0 = ns(i * rng.uniform(0.022, 0.035))
        if i0 + g >= m:
            break
        x[i0:i0 + g] += rng.standard_normal(g) * np.hanning(g) * rng.uniform(0.5, 1.0)
    return bp(x, 1800, 8500, order=2)


def squirrel_layer(rng):
    buf = stereo(DUR)
    t = 59.5
    seq = 0
    while t < 66.5:
        u = (t - 59.5) / 7.0
        x = scratch_burst(rng, int(rng.integers(4, 9)))
        x = lp(x, 9000 - 4500 * u)  # urcă pe trunchi și se îndepărtează
        add(buf, x, t, gain=0.9 - 0.4 * u, pan=0.15 + 0.35 * u)
        if seq == 0:
            ev("veveriță", t, 2000, 8000, 0.4)
        seq += 1
        t += len(x) / SR + (rng.uniform(0.6, 1.1) if rng.uniform() < 0.4 else rng.uniform(0.05, 0.2))
    # câteva bucățele de scoarță care cad pe frunze
    for t0 in (61.4, 62.0, 64.3):
        add(buf, crunch(rng, 0.03, 200) * 0.5, t0, gain=0.5, pan=0.2)
    return buf


# ------------------------------------------------------------------ corbul
def raven_croak(rng):
    """Croncănit grav, rezonant, cu „r” rulat la început („prruk”)."""
    d = rng.uniform(0.26, 0.34)
    m = ns(d)
    tt = t_axis(m)
    f0 = rng.uniform(190, 230) * (1.06 - 0.12 * tt / d)
    from audiolib import saw
    src = saw(f0)
    roll = 1 - 0.55 * (tt < d * 0.45) * (0.5 + 0.5 * np.sin(2 * np.pi * 28 * tt))
    x = (peaking(src, 720, 9.0, 2.0) + 0.6 * bp(src, 1100, 1600) + 0.3 * bp(src, 2300, 2900))
    x = lp(x, 3200) * roll * smooth_env(m, 0.015, 0.09)
    return x / (np.max(np.abs(x)) + 1e-12)


def raven_layer(rng):
    buf = stereo(DUR)
    calls = [(67.5, 0.75), (68.25, 0.6), (69.6, 0.2), (70.4, 0.0), (72.2, -0.45), (73.1, -0.6)]
    for i, (t0, pan) in enumerate(calls):
        far = 0.15 + 0.08 * abs(i - 3)
        add(buf, distance(raven_croak(rng), far), t0, gain=1.0 - 0.06 * abs(i - 3), pan=pan)
        ev("corb", t0, 500, 1200, 0.3)
    return buf


# ------------------------------------------------------------------ mix
def render(seed=1754):
    rng = rng_of(seed)
    ir_near = make_ir(t60=1.1, predelay=0.012, seed=31, hf_ratio=0.5)
    ir_far = make_ir(t60=2.0, predelay=0.03, seed=32, hf_ratio=0.35)
    layers = {  # nume: (semnal, LUFS țintă, IR, wet, factor de vârf maxim în dB)
        "vant": (wind_layer(rng), -26.0, ir_near, 0.05, 13.0),
        "apa": (water_layer(rng), -30.0, ir_near, 0.12, 14.0),
        "bufnita": (owl_layer(rng), -26.5, ir_far, 0.55, 10.0),
        "mierla": (blackbird_layer(rng), -24.5, ir_far, 0.35, 12.0),
        "pitigoi": (tit_layer(rng), -27.5, ir_far, 0.3, 12.0),
        "presura": (bunting_layer(rng), -29.0, ir_far, 0.35, 12.0),
        "ciocanitoare": (woodpecker_layer(rng), -26.5, ir_far, 0.45, 9.0),
        "pasi": (steps_layer(rng), -26.5, ir_near, 0.12, 10.0),
        "gaita": (jay_layer(rng), -25.0, ir_far, 0.3, 10.5),
        "veverita": (squirrel_layer(rng), -29.0, ir_near, 0.15, 14.0),
        "corb": (raven_layer(rng), -24.0, ir_far, 0.4, 10.0),
    }
    mix = stereo(DUR)
    for name, (x, target, ir, wet, crest) in layers.items():
        # vârfurile foarte scurte (pocnete, lovituri) sunt îmblânzite înainte de mix,
        # ca limitatorul final să nu „turtească” tot peisajul
        x = match_lufs(x, target)
        x = match_lufs(limiter(x, ceiling_db=target + crest, look_ms=3.0, release_ms=40.0), target)
        report(x, name)
        mix += reverb(x, ir, wet=wet, dry=1.0)
    mix = peaking(mix, 4500, -1.0, 0.8)
    return mix
